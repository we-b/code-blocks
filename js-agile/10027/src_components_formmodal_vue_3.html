<script setup>
import GoodsBody from './GoodsBody.vue'
import GenreBody from './GenreBody.vue'
import DetailBody from './DetailBody.vue'
import { computed } from 'vue'

const props = defineProps({
  isOpen: Boolean,
  body: String,
  goods: {
    type: Object,
    required: false,
    default: () => ({})
  }
})

const emit = defineEmits(['close-modal']); 
const closeModal = () => {
  emit('close-modal')
}

const component = computed(() => {
  if (props.body === 'goods') {
    return GoodsBody;
  }else if (props.body === 'genre') {
    return GenreBody;
  }else if (props.body === 'details'){
    return DetailBody;
  }
})
</script>

<template>
  <div v-if="isOpen">
    <component :is="component" :goods="goods" @close-modal="closeModal"/>
  </div>
</template>

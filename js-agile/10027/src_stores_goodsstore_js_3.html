  async function changeGoodsStatus(selectedGoods) {
    try{
      const response = await api.put(`/goods/${selectedGoods.id}/status`, {statusId: selectedGoods.statusId});

      // responseの商品idと合致する商品をgoods内から探してIndexを取得する
      const index = goods.value.findIndex(g => g.id === response.data.id)
      // Indexが取得できたら、該当データのstatusIdを更新する。
      if(index !== -1) {
        goods.value[index].statusId = selectedGoods.statusId
      }
    }catch(error){
      console.log('ステータス変更に失敗しました', error);
    }
  }
  return {goods, fetchAllGoods, addGoods, updateGoods, removeGoods, changeGoodsStatus}

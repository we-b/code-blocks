<script setup>
import { computed } from 'vue';
const props = defineProps({
  tasks: Array,
  users: Array
})

const formattedMakerName = (task) => {
  const makerId = task.makerId;
  const maker = props.users.find(u => u.uid === makerId);
  return maker ? maker.displayName : '不明なユーザー';
}

const formattedAssigneeName = (task) => {
  const assigneeId = task.assigneeId;
  const assignee = props.users.find(u => u.uid === assigneeId);
  return assignee ? assignee.displayName : '不明なユーザー';
}
</script>

<template>
  <div>
    <ul class="task-lists"> 
      <li class="task-list" v-for="task in tasks" :key="task.id">
        <p class="task-title">作成者:{{ formattedMakerName(task) }}</p>
        <p>内容:{{ task.content }}</p>
        <p>担当者:{{ formattedAssigneeName(task) }}</p>
      </li>
    </ul>
  </div>
</template>

# ネズミの合計を計算するメソッド
def cal_animal_count(adult, child)
  (adult + child) * 2
end


adult = 0 # 大人のネズミ
child = 0 # 子供のネズミ
# 成獣のネズミ１組投入
adult += 1
puts "----------#{0}ヶ月目----------"
puts "ネズミは合計#{cal_animal_count(adult, child)}匹です"

# 2年経過するように繰り返し処理をする
1.upto(1) do |i|
  puts "----------#{i * 3}ヶ月目----------"
  # ここに繁殖する処理をかく

  puts "前回存在した幼獣0組が成獣になり、成獣は合計で1組になりました"
  puts "前回存在した成獣1組が幼獣を産み、幼獣は合計で1組になりました"
  puts "ネズミは合計4匹です"
end

p cal_animal_count(1, 1)

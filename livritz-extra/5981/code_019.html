# 100円で1ポイント
# 1000円以上からは、100円で2ポイント
# 5000円以上からは100円で4ポイント

def caliculate_point
  paid_money       = gets.to_i
  normal_point     = (paid_money * 1/100).floor
  beyond1000_point = ((paid_money - 1000) * 1/50).floor
  beyond5000_point = ((paid_money - 1000) * 1/25).floor
  
  if paid_money <= 1000
    point =  normal_point
  elsif paid_money < 5000
    point = normal_point + beyond1000_point
  else
    point = normal_point + beyond1000_point + beyond5000_point
  end
end

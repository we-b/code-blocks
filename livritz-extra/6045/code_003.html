class FileManager
  def initialize(path)
    # path名のファイルを作成。
    @file = File.open(path, "w")
    @line_number = 1
  end

  def write_line_with_number(text)
    @file.print("#{@line_number}:#{text}")
    @file.print("\n")
    @line_number += 1
  end

  def write_line_with_blank(text)
    @file.print(text)
    @file.print("\n\n")
  end
end

line_file = FileManager.new('line.txt')
line_file.write_line_with_number('1行目へ記述するテキストです。')
line_file.write_line_with_number('2行目へ記述するテキストです。')
line_file.write_line_with_number('3行目へ記述するテキストです。')

blank_file = FileManager.new('blank.txt')
blank_file.write_line_with_blank('1行目に記述し、次は1行空行にします。')
blank_file.write_line_with_blank('3行目に記述し、次は1行空行にします。')
blank_file.write_line_with_blank('5行目に記述し、次は1行空行にします。')

# forwardableモジュールを読み込む記述。
require 'forwardable'

class FileManager
 # 中略

 def write_line(line)
   @file.print(line)
   @file.print("\n")
 end

end

class FileManageDecorator
 # 処理を委譲したいクラスで、forwardableをextendで読み込む。
 extend Forwardable

 # 第一引数にオブジェクトを参照している属性、第二引数に委譲させたいメソッド名を記載
 def_delegator :@real_writer, :write_line
 # 中略
end

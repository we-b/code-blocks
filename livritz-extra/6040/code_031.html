# 今回のコンテンツデータ
class ContentsData
  attr_accessor :data_title, :data, :created_at
  def initialize
    @data_title = 'コンテンツのタイトル'
    @data = 'コンテンツの内容'
    @created_at = Time.new(2021, 1, 1)
  end


  def replace_date_jp
    @date_jp = @created_at.strftime("%Y年%-m月%-d日")

  end

  def replace_date_us
    @date_us = @created_at.strftime("%-m/%-d/%Y")

  end
end

# プリンター
class Printer
  def initialize
    @data = ContentsData.new
  end

  def print(num)
    # 部数分を出力する。
    num.times do
      puts @data.data
    end
  end

  def replace_date_jp
    @data.replace_date_jp
  end

  def replace_date_us
    @data.replace_date_us
  end
end


printer = Printer.new
printer.print(5)

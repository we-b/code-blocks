# 従業員クラス
class Employee
  attr_reader :name, :salary

  # 初期化時に、名前と給料と、担当上司を設定
  def initialize(name, salary, manager)
    @name = name
    @salary = salary
    @manager = manager
  end

  # 給料を更新するメソッド。上司への通知もここで行う
  def salary=(new_salary)
    @salary = new_salary
    @manager.update(self) # 情報を上司に送信
  end
end

# 上司クラス
class Maneger
  def update(changed_employee)
    puts "#{changed_employee.name}さんの給料は#{changed_employee.salary}円です！"
  end
end

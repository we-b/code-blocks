class PCShop
  attr_reader :individual, :corporate


  def initialize(individual, corporate)
    @individual = individual
    @corporate = corporate
  end

  def notice_software_update(software_update)
    puts "#{individual.kind_of_customer}の顧客にソフトウェアアップデートを通知します"
    individual.download_software_update(software_update)
    puts "#{corporate.kind_of_customer}の顧客にソフトウェアアップデートを通知します"
    corporate.download_software_update(software_update)
  end
end

class Customer
  def initialize
    @updates = []
  end

  def kind_of_customer
    raise '顧客の種類をサブクラスで定義してください'
  end

  def download_software_update(software_update)
    @updates << software_update
    puts "新しい更新情報:#{software_update}"
  end
end

class Individual < Customer # 個人
  def kind_of_customer
    '個人'
  end
end

class Corporate < Customer # 法人
  def kind_of_customer
    '法人'
  end
end

pc_shop = PCShop.new(Individual.new, Corporate.new)
pc_shop.notice_software_update('ソフトウェアver1.02')

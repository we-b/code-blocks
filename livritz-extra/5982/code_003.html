lottery_elements = ['prize' , 'blank' , 'blank' , 'blank' , 'blank']
lottery = lottery_elements.shuffle

while lottery.length > 0 do
 length = lottery.length
 puts "#{[[*0..(length - 1)]].join(" , ")}から番号を選んで下さい"
 target = gets.to_i
 if lottery[target] == 'prize'
   puts '当たり'

   exit
 elsif lottery[target] == 'blank'
   puts 'はずれ'

 else
   puts '正しい番号を選んでください'
 end
 lottery.delete_at(target)
end

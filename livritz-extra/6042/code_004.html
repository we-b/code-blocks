class Mail
  def initialize
    @name = '田中 太郎'
    @mail_addr = 'sample@test.com'
    @company = '株式会社テックキャンプ'
    @addr = '東京都渋谷区1-1-1'
    @line = '■----------------------------------■'
  end
  def output_mail(format)
    puts "お世話になっております、#{@name}です。"
    if(format == '契約完了')
      puts 'この度は、ご契約ありがとうございました。'

    elsif (format == '商品紹介')
      puts "お世話になっております、#{@name}です。"
      puts 'この度は、商品のご紹介です。弊社で取り扱っている「〇〇」は多くのお客様にご使用されています。'
      puts '体験会を10/10 10:00より行っておりますので、是非ご参加ください。'

    elsif (format == '会社紹介')
      puts '先日は突然のご挨拶に対応していただき、誠にありがとうございました。'
      puts 'この度は、弊社サービスが御社の〇〇事業でお役に立てると思い、ご連絡を差し上げた次第です。'
      puts 'よろしければ、ぜひご連絡ください。'

    elsif (format == '退職の挨拶')
      puts 'このたび、一身上の都合により３月末で退社することになり、本日が最終出社日となりました。'
      puts '本来であれば直接ご挨拶をすべきところ、メールでの挨拶にて失礼いたします。'
      puts '在籍中はいたらぬ点もあったかと思いますが、お世話になりありがとうございました。'
      puts '退職後の連絡先は下記になりますので、今後ともご連絡いただけると幸いです。'
      puts 'メール：hoge@mail.com'
      puts '携帯：080-XXXX-XXXX'
    end
    puts @line
    puts "#{@name}\n#{@mail_addr}\n#{@company}\n#{@addr}"
    puts @line
  end
end
mail = Mail.new
mail.output_mail('契約完了')
mail.output_mail('商品紹介')
mail.output_mail('会社紹介')
mail.output_mail('退職の挨拶')

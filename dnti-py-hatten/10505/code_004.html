from django.db import transaction
from .models import BankAccount

def transfer_money(sender_account_number, receiver_account_number, amount):
    try:
        with transaction.atomic():
            sender = BankAccount.objects.select_for_update().get(account_number=sender_account_number)
            receiver = BankAccount.objects.select_for_update().get(account_number=receiver_account_number)

            if sender.balance < amount:
                raise ValueError("送金元の口座の残高が不足しています。")

            sender.balance -= amount
            sender.save()

            receiver.balance += amount
            receiver.save()
    except BankAccount.DoesNotExist:
        print("送金先または送金元の口座が存在しません。")
    except Exception as e:
        print(f"送金に失敗しました: {e}")

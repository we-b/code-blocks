app.get('/search', async (req, res) => {
  const query = req.query.q || '';

  try {
    const tasks = await prisma.task.findMany({
      where: {
        name: {
          contains: query,
          mode: 'insensitive'
        }
      },
      orderBy: {
        id: 'asc'
      }
    });

    res.json(tasks);
  } catch (error) {
    console.error("検索処理に失敗しました:", error);
    res.status(500).json({ message: "検索処理に失敗しました" });
  }
});

// 「健康状態」の状態管理をするreducer関数

const reducer = (state, action) => {
  switch (action.type) {
    case '食事':
      return {weight: state.weight + 5, fat: state.fat + 5 };
    case 'サウナ':
      return {weight: state.weight - 5, fat: state.fat - 5 };
    case '筋トレ':
      return {weight: state.weight + 5, fat: state.fat - 5 };
  }
}

// reducer関数に「現在の状態」と「アクション」を渡すと、新しい状態が返ってくる
let health = {weight: 60, fat: 15}

health = reducer(health, {type: '食事'})
console.log('食事後: ', health)
// => {weight: 65, fat: 20}

health = reducer(health, {type: 'サウナ'})
console.log('サウナ後: ', health)
// => {weight: 60, fat: 15}

health = reducer(health, {type: '筋トレ'})
console.log('筋トレ後: ', health)
// => {weight: 65, fat: 10}

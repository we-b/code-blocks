import { useReducer } from "react";
import { TaskType } from "../interfaces/TaskType";
import { GenreType } from "../interfaces/GenreType";

export type dataAction = {
  type: "tasksUpdate" | "genresUpdate";
  【ここを編集】
};

export type Data = {
  tasksData: TaskType[];
  genresData: GenreType[];
};

export const useDataReducer = (): any => {
  const initialData = {
    tasksData: [],
    genresData: []
  }

  const reducer = (state: Data, action: dataAction) => {
    switch (action.type) {
      case "tasksUpdate":
        return { ...state, tasksData: 【ここを編集】 || state.tasksData };
      case "genresUpdate":
        return { ...state, genresData: 【ここを編集】 || state.genresData};
    }
  };

  const [data, dispatch] = useReducer(reducer, initialData)
  return [data, dispatch]
};

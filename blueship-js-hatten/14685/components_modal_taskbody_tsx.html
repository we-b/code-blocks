import React, { useContext, useState } from "react";
import { DataContext } from "../../pages/home";
import { Select } from "../select";
import { taskRequest } from "../../requests/taskRequest";
import "./style.css";

interface Props {
  handleClose: () => void;
}

export const TaskBody = (props: Props) => {
  const { data, dispatch } = useContext(DataContext);
  const [taskName, setTaskName] = useState("");
  const [taskExplanation, setTaskExplanation] = useState("");
  const [taskDeadlineDate, setTaskDeadlineDate] = useState("");
  const [taskGenreId, setTaskGenreId] = useState(0);

  const handleCreate = async () => {
    if (taskName.trim() === "") return;
    await taskRequest("createTasks", {
      data: {
        name: taskName,
        explanation: taskExplanation,
        deadlineDate: taskDeadlineDate,
        status: 0,
        genreId: taskGenreId,
      },
    });
    const tasksResponse = await taskRequest("fetchTasks");
    dispatch({
      type: "tasksUpdate",
      payload: { tasksData: tasksResponse, genresData: [] },
    });
    setTaskName("");
    setTaskExplanation("");
    setTaskDeadlineDate("");
    setTaskGenreId(0);
    props.handleClose();
  };

  const changeSelectGenreId = (e: React.ChangeEvent<HTMLSelectElement>) => {
    setTaskGenreId(Number(e.target.value));
  };

  return (
    <form className="modal_body">
      <h2 className="input_menu">タスクを追加</h2>
      <div>
        <h4 className="input_title">ジャンル</h4>
        <div className="task_genre">
          <Select genres={data.genresData} changeSelect={changeSelectGenreId} />
        </div>
        <h4 className="input_title">タイトル</h4>
        <input
          type="text"
          value={taskName}
          onChange={(e) => setTaskName(e.target.value)}
        />
        <h4 className="input_title">説明</h4>
        <textarea
          value={taskExplanation}
          onChange={(e) => setTaskExplanation(e.target.value)}
        />
        <h4 className="input_title">期限</h4>
        <input
          className="input_date"
          type="date"
          value={taskDeadlineDate}
          onChange={(e) => setTaskDeadlineDate(e.target.value)}
        />
      </div>
      <input
        className="input_submit"
        type="button"
        value="送信"
        onClick={handleCreate}
      />
    </form>
  );
};

useEffect(() => {
    const fetchData = async () => {
      const genresResponse = await genreRequest("fetchGenres");
      dispatch({ type: "genresUpdate", payload: { tasksData: [], genresData: genresResponse } });
      const tasksResponse = await taskRequest("fetchTasks");
      dispatch({ type: "tasksUpdate", payload: { tasksData: tasksResponse, genresData: [] } });
    };
    fetchData();
  }, []);

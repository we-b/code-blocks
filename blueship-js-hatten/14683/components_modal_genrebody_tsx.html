import React, { useContext, useState } from "react";
import { DataContext } from "../../pages/home";
import { GenreType } from "../../interfaces/GenreType";
import { genreRequest } from "../../requests/genreRequest";
import CancelIcon from "@mui/icons-material/Cancel";
import "./style.css";

export const GenreBody = () => {
  const { data, dispatch } = useContext(DataContext);
  const [genreName, setGenreName] = useState("");

  const handleCreate = async () => {
    if (genreName.trim() === "") return;
    await genreRequest("createGenres", { data: { name: genreName } });
    const genresResponse = await genreRequest("fetchGenres");
    dispatch({
      type: "genresUpdate",
      payload: { tasksData: [], genresData: genresResponse },
    });
    setGenreName("");
  };

  return (
    <div className="modal_body">
      <h2 className="input_menu">ジャンル編集</h2>
      <ul>
        {data.genresData.map((genre: GenreType) => (
          <li className="genre_title" key={genre.id}>
            <span>{genre.name}</span>
            <CancelIcon />
          </li>
        ))}
      </ul>
      <input
        type="text"
        value={genreName}
        onChange={(e) => setGenreName(e.target.value)}
      />
      <input
        className="input_submit"
        type="button"
        value="追加"
        onClick={handleCreate}
      />
    </div>
  );
};

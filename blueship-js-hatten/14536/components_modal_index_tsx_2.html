import React, { useState } from "react";
import Modal from "react-modal";

const customStyles = {
  top: "10%",
  left: "60%",
  right: "50%",
  height: "75vh",
  width: "20vw",
  marginLeft: "-30vw",
  padding: "2vw 10vw",
};

export const FormModal = () => {
  const [isOpen, setIsOpen] = useState(false);
  const handleOpen = () => {
    setIsOpen(true);
  };
  const handleClose = () => {
    setIsOpen(false);
  };

  Modal.setAppElement("#root");

  return (
    <div>
      <button onClick={handleOpen}>開くボタン</button>
      <Modal
        isOpen={isOpen}
        style={customStyles}
      >
      </Modal>
    </div>
  );
};

public class Curry {
  protected String description = "Basic Curry";
  protected int baseCost = 500;

  public String getDescription() {
    return description;
  }

  public int cost() {
    return baseCost;
  }
}

public class CheeseCurry extends Curry {
  public CheeseCurry() {
    description = "Basic Curry, with Cheese";
  }

  @Override
  public int cost() {
    return super.cost() + 200;
  }
}

public class VegetableCurry extends Curry {
  public VegetableCurry() {
    description = "Basic Curry, with Vegetables";
  }

  @Override
  public int cost() {
    return super.cost() + 150;
  }
}

public class CheeseVegetableCurry extends Curry {
  public CheeseVegetableCurry() {
    description = "Basic Curry, with Cheese and Vegetables";
  }

  @Override
  public int cost() {
    return super.cost() + 350;
  }
}


public class Main {
  public static void main(String[] args) {
    Curry basicCurry = new Curry();
    System.out.println(basicCurry.getDescription() + " Cost: " + basicCurry.cost());

    Curry cheeseCurry = new CheeseCurry();
    System.out.println(cheeseCurry.getDescription() + " Cost: " + cheeseCurry.cost());

    Curry vegetableCurry = new VegetableCurry();
    System.out.println(vegetableCurry.getDescription() + " Cost: " + vegetableCurry.cost());

    Curry cheeseVegetableCurry = new CheeseVegetableCurry();
    System.out.println(cheeseVegetableCurry.getDescription() + " Cost: " + cheeseVegetableCurry.cost());
  }
}

import java.util.ArrayList;
import java.util.List;

public class BulletinBoard {
  private static BulletinBoard instance;
  private List<String> messages;

  private BulletinBoard() {
    messages = new ArrayList<>();
  }

  public static BulletinBoard getInstance() {
    if (instance == null) {
      instance = new BulletinBoard();
    }
    return instance;
  }

  public void postMessage(String message) {
    messages.add(message);
    System.out.println("新しいメッセージ: " + message);
  }

  public void displayMessages() {
    System.out.println("メッセージ一覧:");
    for (String message : messages) {
      System.out.println("- " + message);
    }
  }
}

public class Organization {
  private String name;

  public Organization(String name) {
    this.name = name;
  }

  public void postMessage(String message) {
    BulletinBoard board = BulletinBoard.getInstance();
    board.postMessage(name + " : " + message);
  }
}

public class Main {
  public static void main(String[] args) {
    Organization salesOrg = new Organization("営業部");
    Organization developmentOrg = new Organization("開発部");

    salesOrg.postMessage("売上が前年同期比で10%増加しました！");
    developmentOrg.postMessage("新しいプロジェクトがスタートしました！");

    BulletinBoard.getInstance().displayMessages();
  }
}

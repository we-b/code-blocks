import React, { useState } from 'react';

// フォーマット戦略を定義する関数
const normalFormat = (value) => new Intl.NumberFormat().format(value);
const percentFormat = (value) => `${(value * 100).toFixed(2)}%`;
const currencyFormat = (value) => `￥${new Intl.NumberFormat().format(value)}`;

// Strategyパターンのコンテキストとなるコンポーネント
const ValueDisplay = ({ value, formatStrategy }) => {
  return <div>{formatStrategy(value)}</div>;
};

// メインコンポーネント
const App = () => {
  const [format, setFormat] = useState(normalFormat);

  const handleFormatChange = (event) => {
    const selectedFormat = event.target.value;
    
    switch (selectedFormat) {
      case 'percent':
        setFormat(() => percentFormat);
        break;
      case 'currency':
        setFormat(() => currencyFormat);
        break;
      default:
        setFormat(() => normalFormat);
    }
  };

  return (
    <div>
      <h1>Strategy Pattern Example</h1>
      <select onChange={handleFormatChange}>
        <option value="normal">Normal</option>
        <option value="percent">Percent</option>
        <option value="currency">Currency</option>
      </select>
      <ValueDisplay value={1500} formatStrategy={format} />
    </div>
  );
};

export default App;

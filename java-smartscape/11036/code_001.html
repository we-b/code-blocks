// yearが閏年かどうかを判定する関数
const getYear = (year) => {
  // yearが4で割り切れる（year % 4 == 0）場合
  if (year % 4 == 0){ 
    // yearが100で割り切れ、なおかつ（&&）、400で割り切れない（year % 400 != 0）場合
    if (year % 100 == 0 && year % 400 != 0){
      console.log(`${year}年は閏年ではありません`);
    } else {
      console.log(`${year}年は閏年です`);
    };
  }else {
    console.log(`${year}年は閏年ではありません`);
  };
};

public class Sample {
    public static void main(String[] args) {
        int num = 60;

        if (num % 15 == 0) {
            System.out.println(num + "は3と5の倍数です");
        } else if (num % 3 == 0) {
            System.out.println(num + "は3の倍数です");
        } else if (num % 5 == 0) {
            System.out.println(num + "は5の倍数です");
        } else {
            System.out.println(num + "は3の倍数でも、5の倍数でもありません");
        }
    }
}

interface Swimmable {
  void swim();
}

interface Flyable {
  void fly();
}

public class Animal {
  void eat() {
    // 食べる処理
  }
}

public class Duck extends Animal implements Swimmable, Flyable {
  @Override
  public void swim() {
    // 泳ぐ処理
  }
  
  @Override
  public void fly() {
    // 飛ぶ処理
  }
}

public class Main {
  public static void main(String[] args) {
    Duck duck = new Duck();
    duck.eat();
    duck.swim();
    duck.fly();
  }
}

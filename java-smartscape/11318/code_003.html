public abstract class Cooking {
  public final void prepareDish() {
    gatherIngredients();
    cook();
    serve();
  }

  private void gatherIngredients() {
    System.out.println("材料を集める");
  }

  protected abstract void cook();

  private void serve() {
    System.out.println("盛り付ける");
  }
}

public class CurryCooking extends Cooking {
  @Override
  protected void cook() {
    System.out.println("食材を煮込む");
  }
}

public class PastaCooking extends Cooking {
  @Override
  protected void cook() {
    System.out.println("パスタを茹でる");
    System.out.println("トッピングと和える");
  }
}

public class Mein {
  public static void main(String[] args) {
    Cooking curryCooking = new CurryCooking();
    curryCooking.prepareDish();
    
    Cooking pastaCooking = new PastaCooking();
    pastaCooking.prepareDish();
  }
}

// ConcreteObserverクラス
public abstract class Customer implements Observer {
  protected List<String> updates;

  public Customer() {
    updates = new ArrayList<>();
  }

  @Override
  public void downloadSoftwareUpdate(String softwareUpdate) {
    updates.add(softwareUpdate);
    System.out.println(kindOfCustomer() + " が新しい更新情報をダウンロードしました: " + softwareUpdate);
  }

  @Override
  public abstract String kindOfCustomer();
}

// ConcreteObserverクラス（個人）
public class Individual extends Customer {
  @Override
  public String kindOfCustomer() {
    return "個人";
  }
}

// ConcreteObserverクラス（法人）
public class Corporate extends Customer {
  @Override
  public String kindOfCustomer() {
    return "法人";
  }
}

import { useForm } from "react-hook-form";
import api from "../api/axios";

type RegisterFormData = {
  nickname: string;
  email: string;
  password: string;
  password_confirmation: string;
};

const UserForm = () => {
  const { register, handleSubmit, formState: { errors } } = useForm<RegisterFormData>();

  const onSubmit = async (data: RegisterFormData) => {
    try {
      const response = await api.post("/register", data);
      console.log("登録成功:", response.data);
      alert("ユーザーの登録が完了しました");
    } catch (error) {
      console.error("登録失敗:", error);
    }
  };

  return (
    <>
      <h2>ユーザー登録</h2>
      <form onSubmit={handleSubmit(onSubmit)} noValidate>
        <label htmlFor="nickname">ニックネーム</label>
        <input type="text" id="nickname" className="input-field" {...register("nickname")} />

        <label htmlFor="email">メールアドレス</label>
        <input type="email" id="email" className="input-field" {...register("email")} />

        <label htmlFor="password">パスワード</label>
        <input type="password" id="password" className="input-field" {...register("password")} />

        <label htmlFor="password-confirmation">確認用パスワード</label>
        <input type="password" id="password-confirmation" className="input-field" {...register("password_confirmation")} />

        <input type="submit" value="登録" id="submit-btn" />
      </form>
    </>
  );
};

export default UserForm;

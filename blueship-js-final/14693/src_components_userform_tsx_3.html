import { useForm } from "react-hook-form";
import api from "../api/axios";

type RegisterFormData = {
  nickname: string;
  email: string;
  password: string;
  password_confirmation: string;
};

const UserForm = () => {
  const { register, handleSubmit, formState: { errors }, watch } = useForm<RegisterFormData>();

  const onSubmit = async (data: RegisterFormData) => {
    try {
      const response = await api.post("/register", data);
      console.log("登録成功:", response.data);
      alert("ユーザーの登録が完了しました");
    } catch (error) {
      console.error("登録失敗:", error);
    }
  };

  return (
    <>
      <h2>ユーザー登録</h2>
      <form onSubmit={handleSubmit(onSubmit)} noValidate>
        <label htmlFor="nickname">ニックネーム</label>
        {/* 必須(required)のバリデーション追加 */}
        <input type="text" id="nickname" className="input-field"
          {...register("nickname", {
            required: "Nickname is required"
          })}
        />

        <label htmlFor="email">メールアドレス</label>
        {/* 必須(required)、emailパターンに関するバリデーション追加 */}
        <input type="email" id="email" className="input-field"
          {...register("email", {
            required: "Email is required",
            pattern: {
              value: /^[^\s@]+@[^\s@]+\.[^\s@]+$/,
              message: "Invalid email address"
            }
          })}
        />

        <label htmlFor="password">パスワード</label>
        {/* 必須(required)、最小文字数に関するバリデーション追加 */}
        <input type="password" id="password" className="input-field"
          {...register("password", {
            required: "Password is required",
            minLength: {
              value: 6,
              message: "Password must be at least 6 characters long"
            }
          })}
        />

        <label htmlFor="password-confirmation">確認用パスワード</label>
        {/* 必須(required)、passwordとの一致に関するバリデーション追加 */}
        <input type="password" id="password-confirmation" className="input-field"
          {...register("password_confirmation", {
            required: "Password Confirmation is required",
            validate: (value) =>
              value === watch("password") || "Password confirmation does not match password"
          })}
        />

        <input type="submit" value="登録" id="submit-btn" />
      </form>
    </>
  );
};

export default UserForm;

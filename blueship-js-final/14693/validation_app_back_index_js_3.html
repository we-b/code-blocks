app.post(
  '/register',
  [
  // ニックネームのバリデーション
  body('nickname').notEmpty().withMessage('Nickname is required')
  .matches(/^[ぁ-んァ-ヴー一-龥]+$/).withMessage('Nickname should be in full-width Hiragana, Katakana, or Kanji characters.'),

  // emailが必須(emailの形式)であること
  body('email').notEmpty().withMessage('Email is required').isEmail().withMessage('Invalid email address'),

  // passwordのバリデーションチェック(6文字以上であること)
  body('password').notEmpty().withMessage('Password is required').isLength({ min: 6 }).withMessage('Password must be at least 6 characters long'),

  // 確認用passwordのバリデーション(passwordと確認用passwordが一致すること)
  body('password_confirmation').notEmpty().withMessage('Password Confirmation is required').custom((value, { req }) => {
    if ( value !== req.body.password) {
      throw new Error("Password confirmation does not match password");
    }
    return true;
  })
],
//(省略)

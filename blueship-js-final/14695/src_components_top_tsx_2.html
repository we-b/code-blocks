import React, { useState, useEffect } from "react";
import axios from "axios";
import { useNavigate } from "react-router-dom";
import { auth, signOut } from "../firebase";
import { TaskList } from "./TaskList";

type FirebaseUser = {
  uid: string;
  displayName: string;
};

export const Top: React.FC = () => {
  const navigate = useNavigate();
  const displayName = auth.currentUser?.displayName;
  const [allUsers, setAllUsers] = useState<FirebaseUser[]>([]);

  useEffect(() => {
    const fetchUsers = async () => {
      try {
        const response = await axios.get("http://localhost:3100/users");
        setAllUsers(response.data);
      } catch (error) {
        console.log("ユーザーデータの取得ができませんでした", error);
      }
    };
    fetchUsers();
  }, []);

  const handleSignOut = async () => {
    // (省略)
  };

  return (
    <div>
      <div className="header">
        <h1>タスク管理アプリ</h1>
        <button className="logout" onClick={handleSignOut}>
          ログアウト
        </button>
      </div>
      <div className="user-name">
        <span>こんにちは{displayName}さん</span>
      </div>
      <form className="task-form">
        <div className="input-area">
          <label htmlFor="content">タスク内容</label>
          <textarea id="content" placeholder="タスクを入力してください" />
        </div>
        <div className="input-area">
          <label htmlFor="assignee">対応者</label>
          <select>
            <option value="">対応者を選択</option>
            {allUsers.map((user) => (
              <option key={user.uid} value={user.uid}>
                {user.displayName}
              </option>
            ))}
          </select>
        </div>
        <button type="submit" className="submitBtn">
          保存
        </button>
      </form>
      <TaskList tasks={[]} users={allUsers} />
    </div>
  );
};

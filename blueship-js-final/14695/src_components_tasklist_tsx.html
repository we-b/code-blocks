import React from "react";

type Task = {
  id: number;
  content: string;
  makerId: string;
  assigneeId: string;
};

type User = {
  uid: string;
  displayName: string;
};

type TaskListProps = {
  tasks: Task[];
  users: User[];
};

export const TaskList: React.FC<TaskListProps> = ({ tasks, users }) => {
  const formattedMakerName = (task: Task): string => {
    const maker = users.find((u) => u.uid === task.makerId);
    return maker ? maker.displayName : "不明なユーザー";
  };

  const formattedAssigneeName = (task: Task): string => {
    const assignee = users.find((u) => u.uid === task.assigneeId);
    return assignee ? assignee.displayName : "不明なユーザー";
  };

  return (
    <div>
      <ul className="task-lists">
        {tasks.map((task) => (
          <li className="task-list" key={task.id}>
            <p className="task-title">作成者: {formattedMakerName(task)}</p>
            <p>内容: {task.content}</p>
            <p>担当者: {formattedAssigneeName(task)}</p>
          </li>
        ))}
      </ul>
    </div>
  );
};

export const Top: React.FC = () => {
  // (省略)

  // handleSignOutの後に追記
  // タスクの保存処理
  const submitTask = async (e: React.FormEvent) => {
    e.preventDefault();

    // ログインユーザーの確認
    const currentUser = auth.currentUser;
    if (!currentUser) {
      console.log("ログインが必要です");
      return;
    }

    // makerIdに現在ログイン中のユーザーのuidを設定
    const makerId = currentUser.uid;

    try {
      const response = await axios.post("http://localhost:3100/tasks", {
        content,
        makerId,
        assigneeId,
      });
      // 取得したtaskデータをtasksの配列に追加
      setTasks((prev) => [...prev, response.data]);

      // 入力フォームのリセット
      setContent("");
      setAssigneeId("");
    } catch (error) {
      console.log("タスク保存中にエラーが発生しました", error);
    }
  };

  return (
    <div>
      {/* (省略) */}
      <form className="task-form" onSubmit={submitTask}>
        {/* (省略) */}
      </form>
      {/* (省略) */}
    </div>
  );
};

import { useState } from "react";
import Modal from "react-modal";
import { Chat, Comment } from "../types";
import CommentItem from "./CommentItem";

Modal.setAppElement("#root");

type ChatModalProps = {
  isOpen: boolean;
  chat: Chat | null;
  comments: Comment[];
  onClose: () => void;
  onAddComment: (content: string, chatId: number) => void;
};

const ChatModal = ({ isOpen, chat, comments, onClose, onAddComment }: ChatModalProps) => {
  const [commentContent, setCommentContent] = useState("");

  if (!chat) return null;

  const handleSubmitComment = (e: React.FormEvent) => {
    e.preventDefault();
    onAddComment(commentContent, chat.id);
    setCommentContent("");
  };

  return (
    <Modal
      isOpen={isOpen}
      onRequestClose={onClose}
      className="modal-content"
      overlayClassName="modal-overlay"
    >
      <div className="modal-header">
        <h2>チャット詳細</h2>
        <button className="modal-close" onClick={onClose}>
          &times;
        </button>
      </div>
      <div className="chat-detail">
        <div className="chat-detail-section">
          <h3>タイトル</h3>
          <p>{chat.title}</p>
        </div>
        <div className="chat-detail-section">
          <h3>内容</h3>
          <p>{chat.content}</p>
        </div>
      </div>
      <form className="comment-form" onSubmit={handleSubmitComment}>
        <label htmlFor="comment">コメント</label>
        <textarea
          id="comment"
          placeholder="コメントを入力してください"
          value={commentContent}
          onChange={(e) => setCommentContent(e.target.value)}
        ></textarea>
        <div className="button-container">
          <button type="submit">送信</button>
        </div>
      </form>
      <h3>コメント一覧</h3>
      <div className="comment-list">
        {/* TODO: コメントの一覧表示はこの後実装します */}
      </div>
    </Modal>
  );
};

export default ChatModal;

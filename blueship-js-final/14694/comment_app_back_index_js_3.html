// コメント保存の処理
app.post("/comments", async (req, res) => {
  try {
    const { content, chatId } = req.body;
    const savedData = await prisma.comment.create({ data: { content, chatId } });
    res.json(savedData);
  } catch (error) {
    res.status(500).send("コメントの保存に失敗しました");
  }
});

app.listen(3100, () => {
  console.log("サーバーがポート3100で起動しました");
});

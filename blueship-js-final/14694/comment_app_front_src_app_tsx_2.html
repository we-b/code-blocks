import { useState, useEffect } from "react";
import { Chat } from "./types";
import api from "./api/axios";
import ChatForm from "./components/ChatForm";
import ChatCard from "./components/ChatCard";
import ChatModal from "./components/ChatModal";
import "./App.css";

function App() {
  const [chats, setChats] = useState<Chat[]>([]);

  // チャット一覧を取得
  const fetchChats = async () => {
    try {
      const response = await api.get("/chats");
      setChats(response.data);
    } catch (error) {
      console.error("チャットの取得に失敗しました", error);
    }
  };

  // 初回マウント時にチャットを取得
  useEffect(() => {
    fetchChats();
  }, []);

  // チャットを追加
  const addChat = async (title: string, content: string) => {
    try {
      const response = await api.post("/chats", { title, content });
      const addedChat: Chat = response.data;
      setChats((prev) => [...prev, addedChat]);
    } catch (error) {
      console.error("チャットの保存に失敗しました", error);
    }
  };

  return (
    <div className="app">
      <h1>Comment App</h1>
      <ChatForm onSubmit={addChat} />
      <div className="chat-list">
        {chats.map((chat) => (
          <ChatCard key={chat.id} chat={chat} onClickChat={() => {}} />
        ))}
      </div>
    </div>
  );
}

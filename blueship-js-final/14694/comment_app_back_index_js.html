app.post("/chats", async (req, res) => {
  try {
    const { title, content } = req.body;
    const savedData = await prisma.chat.create({ data: { title, content } });
    res.json(savedData);
  } catch (error) {
    res.status(500).send("チャットの保存に失敗しました");
  }
});

app.listen(3100, () => {
  console.log("サーバーがポート3100で起動しました");
});

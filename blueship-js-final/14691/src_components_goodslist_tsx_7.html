  const handleStatusChange = async (goods: Goods, newStatusId: number) => {
    try {
      const response = await api.put(`/goods/${goods.id}/status`, {
        statusId: newStatusId,
      });
      dispatch({ type: "UPDATE_GOODS_STATUS", payload: response.data });
    } catch (error) {
      console.error("ステータス変更ができませんでした", error);
    }
  };

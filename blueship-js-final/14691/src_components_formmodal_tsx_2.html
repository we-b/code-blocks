import React from "react";
import Modal from "react-modal";
import { GoodsBody } from "./GoodsBody";
import { GenreBody } from "./GenreBody";
import { DetailBody } from "./DetailBody";
import { Goods } from "../types";

Modal.setAppElement("#root");

type Props = {
  isOpen: boolean;
  body: "goods" | "genre" | "details";
  onClose: () => void;
  goods?: Goods | null;
};

export const FormModal: React.FC<Props> = ({ isOpen, body, onClose, goods }) => {
  const renderBody = () => {
    if (body === "goods") {
      return <GoodsBody onClose={onClose} />;
    } else if (body === "genre") {
      return <GenreBody onClose={onClose} />;
    } else if (body === "details") {
      return <DetailBody onClose={onClose} goods={goods!} />;
    }
    return null;
  };

  return (
    <Modal
      isOpen={isOpen}
      onRequestClose={onClose}
      className="modal-content"
      overlayClassName="modal-overlay"
    >
      {renderBody()}
    </Modal>
  );
};

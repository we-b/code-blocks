import React, { useState, useContext } from "react";
import api from "../api/axios";
import { DataContext } from "../App";
import { Genre } from "../types";

type Props = {
  onClose: () => void;
};

export const GenreBody: React.FC<Props> = ({ onClose }) => {
  const { state, dispatch } = useContext(DataContext);
  const [genreName, setGenreName] = useState("");

  const submitGenre = async () => {
    try {
      const response = await api.post("/genres", { name: genreName });
      const addedGenre: Genre = response.data;
      dispatch({ type: "ADD_GENRE", payload: addedGenre });
      // 入力フォームのリセット
      setGenreName("");
    } catch (error) {
      console.error("ジャンルの保存に失敗しました", error);
    }
  };

  return (
    <div className="modal_body">
      <h2 className="input_menu">ジャンル編集</h2>
      <form>
        <input
          className="input_genre"
          type="text"
          placeholder="ジャンル名"
          value={genreName}
          onChange={(e) => setGenreName(e.target.value)}
        />
        <input
          className="input_submit"
          type="button"
          value="追加"
          onClick={submitGenre}
        />
      </form>
    </div>
  );
};

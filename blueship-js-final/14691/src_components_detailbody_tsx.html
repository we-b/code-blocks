import React, { useState, useContext } from "react";
import { DataContext } from "../App";
import { FormModal } from "./FormModal";
import { Goods, Genre } from "../types";

type Props = {
  onClose: () => void;
  goods: Goods;
};

export const DetailBody: React.FC<Props> = ({ onClose, goods }) => {
  const { state } = useContext(DataContext);
  const [showEditModal, setShowEditModal] = useState(false);

  // stateから最新のgoodsデータを取得
  const latestGoods =
    state.goods.find((g: Goods) => g.id === goods.id) || goods;

  const genreName = (() => {
    const genre = state.genres.find(
      (genre: Genre) => genre.id === latestGoods.genreId
    );
    return genre ? genre.name : "不明";
  })();

  return (
    <div className="detail_body">
      <div className="detail_header">
        <h2>リスト詳細</h2>
        <div className="buttons">
          <span
            className="btn edit"
            onClick={() => setShowEditModal(true)}
          >
            編集
          </span>
          <span className="btn">削除</span>
        </div>
      </div>
      <div className="detail_contents">
        <h3 className="detail_title">品目</h3>
        <div>{latestGoods.name}</div>
        <h3 className="detail_title">個数</h3>
        <div>{latestGoods.quantity}</div>
        <h3 className="detail_title">ジャンル</h3>
        <div>{genreName}</div>
      </div>
      {showEditModal && (
        <FormModal
          isOpen={showEditModal}
          body="goods"
          goods={latestGoods}
          onClose={() => setShowEditModal(false)}
        />
      )}
    </div>
  );
};

import React, { useState } from "react";
import { FormModal } from "./FormModal";

export const Header: React.FC = () => {
  const [showModal, setShowModal] = useState(false);
  const [bodyType, setBodyType] = useState<"goods" | "genre">("goods");

  const openModal = (type: "goods" | "genre") => {
    setBodyType(type);
    setShowModal(true);
  };

  return (
    <div className="header-list">
      <button className="add-list" onClick={() => openModal("goods")}>
        リスト追加
      </button>
      <button onClick={() => openModal("genre")}>ジャンル追加</button>
      <FormModal
        isOpen={showModal}
        body={bodyType}
        onClose={() => setShowModal(false)}
      />
    </div>
  );
};

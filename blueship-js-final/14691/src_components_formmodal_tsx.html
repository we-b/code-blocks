import React from "react";
import Modal from "react-modal";
import { GoodsBody } from "./GoodsBody";
import { GenreBody } from "./GenreBody";

Modal.setAppElement("#root");

type Props = {
  isOpen: boolean;
  body: "goods" | "genre";
  onClose: () => void;
};

export const FormModal: React.FC<Props> = ({ isOpen, body, onClose }) => {
  return (
    <Modal
      isOpen={isOpen}
      onRequestClose={onClose}
      className="modal-content"
      overlayClassName="modal-overlay"
    >
      {body === "goods" ? (
        <GoodsBody onClose={onClose} />
      ) : (
        <GenreBody onClose={onClose} />
      )}
    </Modal>
  );
};

// 送信処理
  const goodsSubmit = async () => {
    try {
      if (isEditing) {
        // 編集処理
        const updateData = {
          name,
          quantity,
          genreId: Number(genreId),
          statusId,
        };
        const response = await api.put(
          `/goods/${initialGoods!.id}`,
          updateData
        );
        dispatch({ type: "UPDATE_GOODS", payload: response.data });
        onClose();
      } else {
        const newGoods = {
          name,
          quantity: Number(quantity),
          genreId: Number(genreId),
          statusId,
        };
        const response = await api.post("/goods", newGoods);
        dispatch({ type: "ADD_GOODS", payload: response.data });
        setName("");
        setQuantity("");
        setGenreId("");
        setStatusId(0);
        onClose();
      }
    } catch (error) {
      console.error(
        `${isEditing ? "更新" : "保存"}に失敗しました。`,
        error
      );
    }
  };

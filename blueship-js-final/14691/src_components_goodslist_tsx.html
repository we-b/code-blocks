import React, { useState, useContext } from "react";
import { DataContext } from "../App";
import { FormModal } from "./FormModal";
import { Goods } from "../types";

export const GoodsList: React.FC = () => {
  const { state } = useContext(DataContext);
  const [showModal, setShowModal] = useState(false);
  const [selectedGoods, setSelectedGoods] = useState<Goods | null>(null);

  const showDetail = (goods: Goods) => {
    setSelectedGoods(goods);
    setShowModal(true);
  };

  const statusList = ["未購入", "購入済み"];

  return (
    <table border={1}>
      <thead>
        <tr>
          <th>品目</th>
          <th>個数</th>
          <th>ステータス</th>
        </tr>
      </thead>
      <tbody>
        {state.goods.map((goods: Goods) => (
          <tr key={goods.id}>
            <td onClick={() => showDetail(goods)}>{goods.name}</td>
            <td>{goods.quantity}</td>
            <td>
              <select className="form_input" value={goods.statusId}>
                {statusList.map((status, index) => (
                  <option key={index} value={index}>
                    {status}
                  </option>
                ))}
              </select>
            </td>
          </tr>
        ))}
      </tbody>
      <FormModal
        isOpen={showModal}
        body="details"
        goods={selectedGoods}
        onClose={() => setShowModal(false)}
      />
    </table>
  );
};

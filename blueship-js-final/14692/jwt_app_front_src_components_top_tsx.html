import React, { useState, useEffect } from "react";
import { ShowList } from "./ShowList";
import api from "../api/axios";
import { auth, onAuthStateChanged } from "../firebase";
import { User } from "firebase/auth";

type Book = {
  id: number;
  title: string;
  review: string;
  createdAt: string;
};

export const Top: React.FC = () => {
  const [book, setBook] = useState({ title: "", review: "" });
  const [books, setBooks] = useState<Book[]>([]);
  const [currentUser, setCurrentUser] = useState<User | null>(null);

  useEffect(() => {
    const unsubscribe = onAuthStateChanged(auth, (user) => {
      if (user) {
        setCurrentUser(user);
      } else {
        setCurrentUser(null);
      }
    });
    return () => unsubscribe();
  }, []);

  const registerBook = async () => {
    if (!currentUser) {
      console.log("ログインしていません");
      return;
    }
    const uid = currentUser.uid;
    const bookWithUid = { ...book, uid };

    try {
      const response = await api.post("/books", bookWithUid);
      setBooks([...books, response.data]);
      setBook({ title: "", review: "" });
    } catch (error) {
      console.log("レビューの保存ができませんでした", error);
    }
  };

  return (
    <div>
      <div>
        <form>
          <div className="form_element">
            <label htmlFor="title">タイトル</label>
            <br />
            <input
              type="text"
              id="title"
              value={book.title}
              onChange={(e) => setBook({ ...book, title: e.target.value })}
            />
          </div>
          <div className="form_element">
            <label htmlFor="review">感想</label>
            <br />
            <textarea
              id="review"
              value={book.review}
              onChange={(e) => setBook({ ...book, review: e.target.value })}
            />
          </div>
          <input type="button" value="送信" onClick={registerBook} />
        </form>
        <ShowList books={books} />
      </div>
    </div>
  );
};

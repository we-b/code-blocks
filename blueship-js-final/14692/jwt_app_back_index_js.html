// レビュー保存の処理
app.post("/books", async(req, res) => {
  try {
   // bookテーブルへreq.bodyの値を保存する
    const { title, review, uid } = req.body;
    const book = await prisma.book.create({ data: { title, review, uid } })
    res.json(book)
  } catch(error) {
    res.status(500).send("データの保存に失敗しました");
  }
})

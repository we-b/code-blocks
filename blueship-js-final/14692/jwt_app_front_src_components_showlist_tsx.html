import React from "react";

type Book = {
  id: number;
  title: string;
  review: string;
};

type Props = {
  books: Book[];
};

export const ShowList: React.FC<Props> = ({ books }) => {
  return (
    <div>
      <h2>レビュー一覧</h2>
      <div>
        {books.map((book) => (
          <ul className="review-lists" key={book.id}>
            <li className="review-list">
              <h2 className="review-title">{book.title}</h2>
              <p>{book.review}</p>
            </li>
          </ul>
        ))}
      </div>
    </div>
  );
};

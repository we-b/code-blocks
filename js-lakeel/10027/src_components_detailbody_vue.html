<script setup>
import { useGenreStore } from '../stores/GenreStore'
import { computed } from 'vue'

const props = defineProps({
  goods: Object
})

const genreStore = useGenreStore();
const genreName = computed(() => {
  const genre = genreStore.genres.find(genre => genre.id == props.goods.genreId);
  return genre.name;
})
</script>

<template>
  <div class="detail_body">
    <div class="detail_header">
      <h2>リスト詳細</h2>
      <div class="buttons">
        <span class="btn edit" @click="showModal = true">編集</span>
        <span class="btn">削除</span>
      </div>
    </div>
    <div class="detail_contents">
      <h3 class="detail_title">品目</h3>
      <div>{{ goods.name }}</div>
      <h3 class="detail_title">個数</h3>
      <div>{{ goods.quantity}}</div>
      <h3 class="detail_title">ジャンル</h3>
      <div>{{ genreName }}</div>
    </div>
  </div>
</template>

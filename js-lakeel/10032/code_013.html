import { getAuth, updateProfile } from "firebase/auth";
const auth = getAuth();

// ログイン中のユーザーのdisplayNameを更新
  try {
    await updateProfile(auth.currentUser, {
      displayName: "UpdateName",
    });
    // プロフィール更新が成功した際の処理
    console.log("Profileが更新されました");
  } catch (error) {
    // エラー発生時の処理
    console.error("更新に失敗しました", error);
  }

from google import genai

# 1. CSVファイルを読み込む
with open("sales.csv", "r", encoding="utf-8") as f:
    csv_text = f.read()

print("=== 読み込んだCSVデータ ===")
print(csv_text)

# 2. AIに分析を依頼する
client = genai.Client(api_key="あなたのAPIキー")

prompt = f"""以下のCSVデータは、ある会社の購入履歴です。
このデータを分析して、以下の3点をまとめてください。

1. 売上合計（単価 x 個数の合計）
2. カテゴリ別の売上と、その割合
3. この会社への提案（どのカテゴリに注力すべきか）

CSVデータ:
{csv_text}
"""

response = client.models.generate_content(
    model="gemini-3-flash-preview",
    contents=prompt
)

print("=== AIの分析結果 ===")
print(response.text)

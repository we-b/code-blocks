import random

print("最初はグー、じゃんけんぽん！")
player_hand = input("あなたの手を選んでください（0:グー、1:チョキ、2:パー）: ")
player_hand = int(player_hand)
cpu_hand = random.randint(0, 2)

if player_hand == cpu_hand: 
    print("あいこです")
elif player_hand == 0 and cpu_hand == 1:
    print("あなたの勝ちです")
elif player_hand == 1 and cpu_hand == 2:
    print("あなたの勝ちです")
elif player_hand == 2 and cpu_hand == 0:
    print("あなたの勝ちです")
else:
    print("あなたの負けです")

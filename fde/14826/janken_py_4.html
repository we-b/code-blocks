import random

hands = ["グー", "チョキ", "パー"]
win_table = {
    "グー": "チョキ",
    "チョキ": "パー",
    "パー": "グー",
}

def get_cpu_hand():
    num = random.randint(0, 2)
    return hands[num]


def get_player_hand():
    num = int(input("あなたの手を選んでください（0:グー、1:チョキ、2:パー）: "))
    return hands[num]


def judge(player_hand, cpu_hand):
    if player_hand == cpu_hand:
        return "draw"
    elif win_table[player_hand] == cpu_hand:
        return "win"
    else:
        return "lose"


is_playing = True

while is_playing:
    print("最初はグー、じゃんけんぽん！")
    player_hand = get_player_hand()
    cpu_hand = get_cpu_hand()
    print(f"あなたの手: {player_hand}")
    print(f"コンピュータの手: {cpu_hand}")
    result = judge(player_hand, cpu_hand)

    if result == "draw":
        print("あいこです。もう一度勝負！")
    elif result == "win":
        print("あなたの勝ちです")
        is_playing = False
    else:
        print("あなたの負けです")
        is_playing = False

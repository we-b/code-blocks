import random

hands = ["グー", "チョキ", "パー"]
win_table = {
    "グー": "チョキ",
    "チョキ": "パー",
    "パー": "グー",
}

is_playing = True

while is_playing:
    print("最初はグー、じゃんけんぽん！")
    player_hand_num = int(input("あなたの手を選んでください（0:グー、1:チョキ、2:パー）: "))
    cpu_hand_num = random.randint(0, 2)

    player_hand = hands[player_hand_num]
    cpu_hand = hands[cpu_hand_num]

    print(f"あなたの手: {player_hand}")
    print(f"コンピュータの手: {cpu_hand}")

    if player_hand == cpu_hand:
        print("あいこです。もう一度勝負！")
    elif win_table[player_hand] == cpu_hand:
        print("あなたの勝ちです")
        is_playing = False
    else:
        print("あなたの負けです")
        is_playing = False

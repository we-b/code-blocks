import random

hands = ["グー", "チョキ", "パー"]

is_playing = True

while is_playing:
    print("最初はグー、じゃんけんぽん！")
    player_hand = input("あなたの手を選んでください（0:グー、1:チョキ、2:パー）: ")
    player_hand = int(player_hand)
    cpu_hand = random.randint(0, 2)

    print(f"あなたの手: {hands[player_hand]}")
    print(f"コンピュータの手: {hands[cpu_hand]}")

    if player_hand == cpu_hand:
        print("あいこです。もう一度勝負！")
    elif player_hand == 0 and cpu_hand == 1:
        print("あなたの勝ちです")
        is_playing = False
    elif player_hand == 1 and cpu_hand == 2:
        print("あなたの勝ちです")
        is_playing = False
    elif player_hand == 2 and cpu_hand == 0:
        print("あなたの勝ちです")
        is_playing = False
    else:
        print("あなたの負けです")
        is_playing = False

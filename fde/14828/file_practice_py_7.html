# 読み込み
with open("memo.txt", "r", encoding="utf-8") as f:
    lines = f.readlines()

for line in lines:
    clean_line = line.strip()
    if clean_line.startswith("・"):
        print(clean_line)

print("---")

# 書き込み
with open("result.txt", "w", encoding="utf-8") as f:
    f.write("抽出結果：\n")
    for line in lines:
        clean_line = line.strip()
        if clean_line.startswith("・"):
            f.write(clean_line + "\n")

print("result.txt に書き込みました。")

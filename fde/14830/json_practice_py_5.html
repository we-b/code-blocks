import json

with open("company.json", "r", encoding="utf-8") as f:
    data = json.load(f)

print(f"会社名: {data['company']}")
print()

for dept in data["departments"]:
    print(f"【{dept['name']}】")
    for member in dept["members"]:
        print(f"  {member['name']}（{member['role']}）")
    print()

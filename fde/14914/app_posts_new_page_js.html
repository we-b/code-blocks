'use client';

import { useState } from 'react';
import Link from 'next/link';

export default function NewPostPage() {
  const [count, setCount] = useState(0);

  return (
    <div>
      <h1>新規投稿</h1>
      <nav>
        <Link href="/">トップ</Link>
        {' | '}
        <Link href="/posts">投稿一覧</Link>
      </nav>
      <p>ボタンを押した回数: {count}</p>
      <button onClick={() => setCount(count + 1)}>カウントアップ</button>
    </div>
  );
}

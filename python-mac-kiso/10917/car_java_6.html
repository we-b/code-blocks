public class Car {
  private int rideNumber;
  private String name;
  private String model;
  private int capacity;

  public Car(String carName, String carModel, int capacity) {
    name = carName;
    this.model = carModel;
    this.capacity = capacity;
  }

  public static void run() {
    System.out.println("走行します");
  }

  public void rideCar(int rideNumber) {
    this.rideNumber = rideNumber;
    if (rideNumber > this.capacity) {
      System.out.println(this.name + "は、乗車定員が" + this.capacity + "人なので、" + rideNumber + "人は乗れません");
    } else {
      System.out.println(rideNumber + "人乗車できました");
    }
  }
}

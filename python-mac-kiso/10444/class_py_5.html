class Car:
    def __init__(self, car_name, car_model, capacity):
        self.name = car_name
        self.model = car_model
        self.capacity = capacity

    @classmethod
    def run(cls):
        print('走行します')

    def ride_car(self, ride_number):
        self.ride_number = ride_number
        if ride_number > self.capacity:
            print(f"{self.name}は、乗車定員が{self.capacity}人なので、{ride_number}人は乗れません。")
        else:
            print(f"{ride_number}人乗車できました。")

fire_truck = Car('消防車', '大型車', 5)
fire_truck.ride_car(3)

Car.run()

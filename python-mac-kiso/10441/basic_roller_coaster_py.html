ride_count = 0
friends = []

for i in range(3):
    friend = {}
    print("お友達の名前は？")
    friend["name"] = input()
    print("お友達の身長は？")
    friend["height"] = int(input())
    friends.append(friend)
    if friend["height"] >= 130:
        print(f"{friend['name']}くんは乗車できます")
        ride_count += 1
    else:
        print(f"{friend['name']}くんは乗車できません！")

print(f"乗車するのは{ride_count}人です")

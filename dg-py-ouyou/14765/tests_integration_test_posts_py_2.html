class TestPost(LiveServerTestCase):
    @classmethod
    def setUpClass(cls):
        super().setUpClass()
        # ChromeDriverの設定
        service = Service(ChromeDriverManager().install())
        cls.driver = webdriver.Chrome(service=service)
        cls.driver.implicitly_wait(10)

    @classmethod
    def tearDownClass(cls):
        cls.driver.quit()
        super().tearDownClass()

from django.test import LiveServerTestCase
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from selenium.webdriver.common.by import By
from selenium.webdriver.support.wait import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from webdriver_manager.chrome import ChromeDriverManager
from posts.models import Post

class TestPost(LiveServerTestCase):
    @classmethod
    def setUpClass(cls):
        super().setUpClass()
        service = Service(ChromeDriverManager().install())
        cls.driver = webdriver.Chrome(service=service)
        cls.driver.implicitly_wait(10)

    @classmethod
    def tearDownClass(cls):
        cls.driver.quit()
        super().tearDownClass()

class TestPostCreateSuccessCase(TestPost):
    def test_post_can_create_with_valid_information(self):
        # トップページに移動する
        self.driver.get(f"{self.live_server_url}/posts/")

        # トップページに新規投稿ページへ遷移するリンクがあることを確認する
        create_link = self.driver.find_element(By.LINK_TEXT, "新規投稿")
        self.assertTrue(create_link.is_displayed())

        # 新規投稿ページへ移動する
        self.driver.get(f"{self.live_server_url}/posts/create/")

        # タイトルと内容を入力する
        self.driver.find_element(By.NAME, "title").send_keys("Seleniumテスト投稿")
        self.driver.find_element(By.NAME, "content").send_keys("これはSeleniumによる自動テストの投稿です")

        # 投稿ボタンを押すとPostモデルのカウントが1上がることを確認する
        post_count = Post.objects.count()
        self.driver.find_element(By.CSS_SELECTOR, "input[type='submit']").click()
        self.assertEqual(Post.objects.count(), post_count + 1)

        # トップページへ遷移することを確認する
        WebDriverWait(self.driver, 10).until(
            EC.url_to_be(f"{self.live_server_url}/posts/")
        )

        # 投稿した内容がトップページに表示されていることを確認する
        self.assertIn("Seleniumテスト投稿", self.driver.page_source)

class TestPostCreateFailureCase(TestPost):
    def test_post_cannot_create_with_empty_information(self):
        # 新規投稿ページへ移動する
        self.driver.get(f"{self.live_server_url}/posts/create/")

        # 何も入力せずに投稿ボタンを押す
        post_count = Post.objects.count()
        self.driver.find_element(By.CSS_SELECTOR, "input[type='submit']").click()

        # 投稿ボタンを押してもPostモデルのカウントは上がらないことを確認する
        self.assertEqual(Post.objects.count(), post_count)

        # 投稿ページに留まっていることを確認する
        current_url = self.driver.current_url
        self.assertIn("/posts/create/", current_url)

        # エラーメッセージが表示されていることを確認する
        error_area = self.driver.find_element(By.CLASS_NAME, "error-summary")
        self.assertTrue(error_area.is_displayed())

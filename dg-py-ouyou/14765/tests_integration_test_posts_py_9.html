class TestPostCreateFailureCase(TestPost):
    # 空の情報では投稿ができずに投稿ページへ戻ってくる
    def test_post_cannot_create_with_empty_information(self):
        # 新規投稿ページへ移動する
        self.driver.get(f"{self.live_server_url}/posts/create/")

        # 何も入力せずに投稿ボタンを押す
        post_count = Post.objects.count()
        self.driver.find_element(By.CSS_SELECTOR, "input[type='submit']").click()

        # 投稿ボタンを押してもPostモデルのカウントは上がらないことを確認する
        self.assertEqual(Post.objects.count(), post_count)

        # 投稿ページに留まっていることを確認する
        current_url = self.driver.current_url
        self.assertIn("/posts/create/", current_url)

        # エラーメッセージが表示されていることを確認する
        error_area = self.driver.find_element(By.CLASS_NAME, "error-summary")
        self.assertTrue(error_area.is_displayed())

def youtube_url(id):
    return f"https://www.youtube.com/embed/{id}"

def tiktok_url(id):
    return f"https://www.tiktok.com/@{id}"

movies = ["19883950", "81293482", "80931242"]
movie_urls = []
for id in movies:
    movie_urls.append(youtube_url(id))
print("moviesに格納されたYouTubeの動画を表示するURLを表示します")
print(movie_urls)

movies = ["31290843", "19841902", "92103929"]

    movie_urls.append(tiktok_url(id))
print("moviesに格納されたTikTokの動画を表示するURLを表示します")
print(movie_urls)

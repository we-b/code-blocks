def youtube_url(id):
    return f"https://www.youtube.com/embed/{id}"

def tiktok_url(id):
    return f"https://www.tiktok.com/@{id}"

youtube_movies = ["19883950", "81293482", "80931242"]
youtube_movie_urls = []
for id in youtube_movies:
    youtube_movie_urls.append(youtube_url(id))
print("youtube_moviesに格納されたYouTubeの動画を表示するURLを表示します")
print(youtube_movie_urls)

tiktok_movies = ["31290843", "19841902", "92103929"]
tiktok_movie_urls = []
for id in tiktok_movies:
    tiktok_movie_urls.append(tiktok_url(id))
print("tiktok_moviesに格納されたTikTokの動画を表示するURLを表示します")
print(tiktok_movie_urls)

import random
import secrets

# ランダムのアルファベットを生成
def random_string():
    return ''.join(random.choices('ABCDEFGHIJKLMNOPQRSTUVWXYZ', k=5))

# ランダムのhex15桁を生成
def password():
    return secrets.token_hex(15)

# ランダムのアドレスを生成
def mail_address():
    return f"{secrets.token_hex(10)}@sample.com"

image_url = 'https://tech-master.s3.amazonaws.com/uploads/curriculums/images/Rails1-4/sample.jpg'

# user_create
for _ in range(10):
    params = {
        'nickname': random_string(),
        'password': password(),
        'email': mail_address()
    }
    User.create(params)

# tweet_create
for _ in range(10):
    params = {
        'text': random_string(),
        'image': image_url,
        'user_id': 1
    }
    Tweet.create(params)

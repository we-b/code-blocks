import random
import secrets
from abc import ABC, abstractmethod

# Factoryの親クラス
class Factory(ABC):
    @abstractmethod
    def create(self):
        pass

    def random_string(self):
        return ''.join(random.choices('ABCDEFGHIJKLMNOPQRSTUVWXYZ', k=5))

class UserFactory(Factory):
    def password(self):
        return secrets.token_hex(15)

    def mail_address(self):
        return f"{secrets.token_hex(10)}@sample.com"

    def create(self):
        params = {
            'nickname': self.random_string(),
            'password': self.password(),
            'email': self.mail_address()
        }
        return User.create(params)

class TweetFactory(Factory):
    def create(self):
        image_url = 'https://tech-master.s3.amazonaws.com/uploads/curriculums/images/Rails1-4/sample.jpg'
        params = {
            'text': self.random_string(),
            'image': image_url,
            'user_id': 1
        }
        return Tweet.create(params)

# メインファイル
if __name__ == '__main__':
    user_factory = UserFactory()
    tweet_factory = TweetFactory()

    for _ in range(10):
        user_factory.create()
        tweet_factory.create()

def choose_action(self):
    book = self.choose_title()
    print('本を読む場合は0、書評を読む場合は1を選択してください')
    while True:
        input_value = int(input())
        if input_value == 0:
            print(f"{book['name']}を読みます")
            self.read_book(book)
            break
        elif input_value == 1:
            print(f"{book['name']}の書評を読みます")
            self.read_book_review(book)
            break
        else:
            print('本を読む場合は0、書評を読む場合は1を選択してください')

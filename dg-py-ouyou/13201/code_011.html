account_data = {'name': '佐藤', 'deposit': 100000}

def main(account_data):
    query_deposit(account_data)
    withdraw(account_data)
    query_deposit(account_data)  # 引き落とし後の残高を表示する

def query_deposit(account_data):
    print(f"残高は{account_data['deposit']}円です")

def withdraw(account_data):
    print('引き落とし額を数字で入力してください')
    while True:
        withdrawal = int(input())
        if withdrawal < 1000:
            print("引き落としは1000円以上からです")
            print('もう1度、引き落とし額を数字で入力してください')
        else:
            account_data['deposit'] = account_data['deposit'] - withdrawal
            break
    print(f"{withdrawal}円を正常に下ろしました")

main(account_data)

class Vehicle:
    def __init__(self, type):
        self.type = type

    def speed_limit(self):
        if self.type == '車':
            print('制限速度:60km')
        elif self.type == '自転車':
            print('制限速度:15km')
        elif self.type == 'バイク':
            print('制限速度:30km')
        else:
            print('制限速度:不明')

    def run(self):
        if self.type == '車':
            print('車が走ります')
        elif self.type == '自転車':
            print('自転車が走ります')
        elif self.type == 'バイク':
            print('バイクが走ります')
        else:
            print('走れません')

car = Vehicle('車')
car.speed_limit()
car.run()

bicycle = Vehicle('自転車')
bicycle.speed_limit()
bicycle.run()

bike = Vehicle('バイク')
bike.speed_limit()
bike.run()

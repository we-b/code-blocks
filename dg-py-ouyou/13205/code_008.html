# templatetags/custom_tags.py
from django import template
from django.utils.safestring import mark_safe

register = template.Library()

@register.simple_tag
def ad(ad_type, image_count):
    if ad_type == "gif":
        ad_instance = Gif()
    elif ad_type == "image":
        ad_instance = Image()
    elif ad_type == "movie":
        ad_instance = Movie()
    return ad_instance.return_html(ad_type, image_count)

class AdTemplate(ABC):
    def __init__(self):
        self.header = '<header>株式会社〇〇</header>'
        self.footer = '<footer>詳しくはコチラ</footer>'
        self.body = ''

    @abstractmethod
    def return_body(self, ad_type, image_count):
        raise NotImplementedError("Call Abstract Method: return_body")

    def return_html(self, ad_type, image_count):
        self.return_body(ad_type, image_count)
        html = ''
        html = self.header + self.body + self.footer
        return mark_safe(html)

class Gif(AdTemplate):
    def return_body(self, ad_type, image_count):
        if image_count == 2:
            self._return_src_2()
        else:
            self._return_src_1()

    def _return_src_2(self):
        # ２枚パターン
        self.body += '<iframe id="player1" type="text/html" width="700px" height="350px" src="/assets/ad_gif1.png" frameborder="0"></iframe>'
        self.body += '<iframe id="player2" type="text/html" width="700px" height="350px" src="/assets/ad_gif2.png" frameborder="0"></iframe>'

    def _return_src_1(self):
        # １枚パターン
        self.body += '<iframe id="player" type="text/html" width="700px" height="700px" src="/assets/ad_gif1.png" frameborder="0"></iframe>'

class Image(AdTemplate):
    def return_body(self, ad_type, image_count):
        self._return_src()

    def _return_src(self):
        self.body += '<iframe id="player" type="text/html" width="700px" height="700px" src="/assets/ad_image.png" frameborder="0"></iframe>'

class Movie(AdTemplate):
    def return_body(self, ad_type, image_count):
        self._return_src()

    def _return_src(self):
        self.body += '<iframe id="player" type="text/html" width="500px" height="500px" src="/assets/ad_movie.png" frameborder="0"></iframe>'

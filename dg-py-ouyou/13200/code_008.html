class TweetsController:
    def index(self):
        self.tweets = Tweet.all()

    def new(self):
        self.tweet = Tweet.new()

    def create(self):
        Tweet.create(self.tweet_params())

    def destroy(self):
        tweet = Tweet.find(self.params['id'])
        tweet.destroy()

    def edit(self):
        self.tweet = Tweet.find(self.params['id'])

    def update(self):
        tweet = Tweet.find(self.params['id'])
        tweet.update(self.tweet_params())

    def show(self):
        self.tweet = Tweet.find(self.params['id'])

    def tweet_params(self):
        return self.params.require('tweet').permit('name', 'image', 'text')

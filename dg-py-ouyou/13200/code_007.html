def is_admin():
    return user.permission == 'admin'

def is_editor():
    return user.permission == 'editor'

current_editor = user_signed_in and is_editor()

if is_admin():
    redirect_to(admin_url)
elif current_editor:
    redirect_to(editor_url)
elif user_signed_in:
    redirect_to(user_url)
else:
    redirect_to(login_url, alert="ログインできません")

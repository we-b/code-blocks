import random

lottery_elements = ['prize', 'blank', 'blank', 'blank', 'blank']
lottery = lottery_elements.copy()
random.shuffle(lottery)

while len(lottery) > 0:
    length = len(lottery)
    print(f"{list(range(length))}から番号を選んで下さい")
    target = int(input())
    if lottery[target] == 'prize':
        print('当たり')
        exit()
    elif lottery[target] == 'blank':
        print('はずれ')
    else:
        print('正しい番号を選んでください')
    lottery.pop(target)

movie_urls = []
movies = [
    {"genre": "youtube", "ids": ["youtube_sample_id", "youtube_sample_id", "youtube_sample_id"]},
    {"genre": "tiktok", "ids": ["tiktok_sample_id", "tiktok_sample_id", "tiktok_sample_id"]}
]

for movie in movies:
    if movie["genre"] == "youtube":
        for id in movie["ids"]:
            movie_urls.append(f"https://www.youtube.com/embed/{id}")
    elif movie["genre"] == "tiktok":
        for id in movie["ids"]:
            movie_urls.append(f"https://www.tiktok.com/@{id}")

print(movie_urls)

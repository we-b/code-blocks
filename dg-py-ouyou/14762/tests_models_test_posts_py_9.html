    def test_title_cannot_be_blank(self):
        test_post = Post(title="テスト投稿", content="テスト内容")
        test_post.title = ""

        with self.assertRaises(ValidationError) as cm:
            test_post.full_clean()

        self.assertIn('title', cm.exception.message_dict)
        self.assertEqual(cm.exception.message_dict['title'], ["このフィールドは空ではいけません。"])

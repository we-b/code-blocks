# ストラテジの親クラス
class MailStrategy(ABC):
    @abstractmethod
    def output_contents(self):
        pass

# 各ストラテジクラス
class Agreement(MailStrategy):
    def output_contents(self):
        pass

class IntroductionProduct(MailStrategy):
    def output_contents(self):
        pass

class InfoVacation(MailStrategy):
    def output_contents(self):
        pass

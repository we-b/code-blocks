import datetime

# 文書データとその操作を持つクラス
class Content:
    def __init__(self):
        self.title = 'Pythonの基本'
        self.body = 'これはPythonについてのドキュメントです。'
        self.created_at = datetime.datetime(2025, 6, 11)

    # 日本語形式の日付文字列を返す関数
    def format_date_jp(self):
        return self.created_at.strftime("%Y年%m月%d日")

    # アメリカ形式の日付文字列を返す関数
    def format_date_us(self):
        return self.created_at.strftime("%m/%d/%Y")

# 印刷機クラス
class Printer:
    def __init__(self):
        # Printerは内部にContentのインスタンスを持つ（コンポジション）
        self.content = Content()

    # 本文を部数分、出力する
    def print_body(self, num):
        print(f"--- 本文を{num}部印刷します ---")
        for _ in range(num):
            print(self.content.body)
        print("--- 印刷完了 ---")

    # この関数が「委譲」を行っている
    def get_formatted_date_jp(self):
        # 日付のフォーマット処理を、自身では行わない
        # 内部に持つ self.content オブジェクトに処理を「任せる」
        return self.content.format_date_jp()

    # こちらも委譲
    def get_formatted_date_us(self):
        return self.content.format_date_us()

# --- 実行 ---
printer = Printer()

# Printer本来の機能（本文の印刷）を実行
printer.print_body(3)
print()

# Printerが委譲を使って取得した日付データを表示
jp_date = printer.get_formatted_date_jp()
print(f"レポートの作成日 (日本形式): {jp_date}")

us_date = printer.get_formatted_date_us()
print(f"レポートの作成日 (米国形式): {us_date}")

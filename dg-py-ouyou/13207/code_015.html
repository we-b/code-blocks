class Mail:
    def __init__(self, elements):
        # これらの属性は、インスタンス化された後、外部からアクセス可能です。
        self.name = '田中 太郎'
        self.mail_addr = 'sample@test.com'
        self.company = '株式会社パイソン'
        self.addr = '東京都渋谷区1-1-1'
        self.line = '■----------------------------------■'
        self.elements = elements
    
    # ... (他の関数は省略) ...

# --- 使い方の例 ---
# 空のリストを渡してインスタンスを作成
mail_instance = Mail([])

# 属性の読み取り (Rubyのattr_readerに相当)
print(f"名前: {mail_instance.name}")

# 属性の書き換え (Rubyのattr_accessorに相当)
print(f"変更前のelements: {mail_instance.elements}")
mail_instance.elements = ['新しい要素']
print(f"変更後のelements: {mail_instance.elements}")

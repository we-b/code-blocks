class Animal:
    """動物の基本クラス"""
    def speak(self):
        # 親クラスでは一般的な鳴き声を定義
        print("動物が鳴いています。")

class Dog(Animal):
    """犬クラス"""
    # 親クラスのspeak関数をオーバーライド
    def speak(self):
        print("ワン！")

class Cat(Animal):
    """猫クラス"""
    # 親クラスのspeak関数をオーバーライド
    def speak(self):
        print("ニャー！")

# DogとCatのインスタンスを作成
dog = Dog()
cat = Cat()

# animal_listには異なるクラスのインスタンスが混在している
animal_list = [dog, cat]

# --- ポリモーフィズムの確認 ---
# ループで各動物を「鳴かせる」
for animal in animal_list:
    # animalのクラスがDogかCatかを意識せず、単にspeak()を呼び出す
    animal.speak()

from abc import ABC, abstractmethod

# 親クラス（テンプレートの骨格を定義）
class Mail(ABC):
    def __init__(self):
        """初期化関数で、共通の属性を定義します。"""
        self.name = "田中 太郎"
        self.mail_addr = "sample@test.com"
        self.company = "株式会社パイソン"
        self.addr = "東京都渋谷区1-1-1"
        self.line = "■----------------------------------■"

    # この関数が「テンプレート関数」です。
    # 処理のアルゴリズム（骨格）を定義します。
    def output_mail(self):
        self.output_greeting()
        self.output_body()  # この部分の具体的な処理はサブクラスに任せます。
        self.output_info()

    def output_greeting(self):
        """共通の挨拶文を出力します。"""
        print(f"お世話になっております、{self.name}です。")

    @abstractmethod
    def output_body(self):
        """本文を出力する関数。サブクラスでの実装を強制します。"""
        raise NotImplementedError('サブクラスでoutput_bodyを実装してください。')

    def output_info(self):
        """共通の署名を出力します。"""
        print(self.line)
        print(f"{self.name}\n{self.mail_addr}\n{self.company}\n{self.addr}")
        print(self.line)

# 具体的な処理を定義するサブクラス①
class DoneMail(Mail):
    def output_body(self):
        """契約完了メールの本文を実装します。"""
        print("この度は、ご契約ありがとうございました。")

# 具体的な処理を定義するサブクラス②
class IntroductionMail(Mail):
    def output_body(self):
        """商品紹介メールの本文を実装します。"""
        print("この度は、商品のご紹介です。弊社で取り扱っている「〇〇」は多くのお客様にご使用されています。")
        print("体験会を2025年10月10日 10:00より行っておりますので、是非ご参加ください。")

# --- 実行部分 ---
print("--- 契約完了メール ---")
done_mail = DoneMail()
done_mail.output_mail()

print("\n" + "="*20 + "\n") # 区切り線

print("--- 商品紹介メール ---")
introduction_mail = IntroductionMail()
introduction_mail.output_mail()

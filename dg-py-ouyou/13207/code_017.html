# (from abc import ... と MailStrategy, Agreement, IntroductionProduct の定義は上記と同じ)

class InfoVacation(MailStrategy):
    def output_contents(self):
        print("\n\n年末年始の休暇のお知らせ")
        print("2024年12月29日 ～ 2025年1月3日")
        print("新年の営業は1月4日午前9時からとなります。")

class InfoCampaign(MailStrategy):
    def __init__(self, term):
        self.term = term
        
    def output_contents(self):
        print("\nまた、弊社では1月4日より「新年キャンペーン」を開催いたします。")
        print("この機会に、ぜひお買い求めください。")
        print("\nキャンペーンの詳細")
        print(f"開催期間：{self.term}")
        print("対象商品：〇〇 30％OFF(他のキャンペーンとは併用不可)")

class Mail:
    def __init__(self, elements):
        self.name = '田中 太郎'
        self.mail_addr = 'sample@test.com'
        self.company = '株式会社パイソン'
        self.addr = '東京都渋谷区1-1-1'
        self.line = '■----------------------------------■'
        # この時点では、ストラテジを辞書(dict)で受け取る想定
        self.elements = elements

    def output_mail(self):
        """メール全体を出力するテンプレート"""
        print(f"お世話になっております、{self.name}です。")
        self._construct_sentence() # 本文の組み立てと出力を委譲
        print(self.line)
        print(f"{self.name}\n{self.mail_addr}\n{self.company}\n{self.addr}")
        print(self.line)

    def _construct_sentence(self):
        """
        保持しているストラテジを使って本文を組み立てる。
        この時点では辞書(dict)の 'main' と 'sub' キーを元に処理を呼び出す。
        """
        self.elements['main'].output_contents()
        if 'sub' in self.elements:
            self.elements['sub'].output_contents()

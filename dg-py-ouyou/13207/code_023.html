# --- 実行部分 ---

# 1. ストラテジに渡す引数を定義
term = "2025年1月4日(土) 〜 1月8日(水)"

# 2. 組み合わせたいストラテジのインスタンスをリストに格納
#    (契約完了 + キャンペーン + 休暇のお知らせ)
strategies1 = [
    AgreementStrategy(), 
    InfoCampaignStrategy(term),
    InfoVacationStrategy()
]

# 3. ストラテジのリストを渡して、Mail（コンテキスト）のインスタンスを作成
mail = Mail(strategies1)
# 4. メールを出力
mail.output_mail()


print("\n" + "="*40 + "\n") # 区切り線


# 5. 同じmailオブジェクトのストラテジを動的に入れ替える
#    (商品紹介のみ)
mail.strategies = [IntroductionStrategy()]
# 6. 再度メールを出力
mail.output_mail()

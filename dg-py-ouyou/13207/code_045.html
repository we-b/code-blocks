# -----------------------------------------------
# 1. ストラテジを「関数」として定義する
# -----------------------------------------------

def agreement_strategy():
    """契約完了の文章を出力するストラテジ関数。"""
    print("この度は、ご契約ありがとうございました。")

def introduction_strategy():
    """商品紹介の文章を出力するストラテジ関数。"""
    print("この度は、商品のご紹介です。弊社で取り扱っている「〇〇」は多くのお客様にご使用されています。")
    print("体験会を2025年10月10日 10:00より行っておりますので、是非ご参加ください。")

def vacation_info_strategy():
    """休暇のお知らせを出力するストラテジ関数。"""
    print("\n\n年末年始の休暇のお知らせ")
    print("2024年12月29日 ～ 2025年1月3日")
    print("新年の営業は1月4日午前9時からとなります。")

def create_campaign_strategy(term):
    """
    期間(term)を引数に取り、その期間を記憶した「新しい関数」を返す。
    クロージャの仕組みを利用している。
    """
    def campaign_strategy():
        print("\nまた、弊社では1月4日より「新年キャンペーン」を開催いたします。")
        print("この機会に、ぜひお買い求めください。")
        print("\nキャンペーンの詳細")
        print(f"開催期間：{term}")
        print("対象商品：〇〇 30％OFF(他のキャンペーンとは併用不可)")
    
    return campaign_strategy

# -----------------------------------------------
# 2. コンテキストを定義する (ストラテジが関数であることを前提とする)
# -----------------------------------------------

class Mail:
    def __init__(self, strategies):
        self.name = '田中 太郎'
        self.mail_addr = 'sample@test.com'
        self.company = '株式会社パイソン'
        self.addr = '東京都渋谷区1-1-1'
        self.line = '■----------------------------------■'
        # 利用するストラテジ（関数オブジェクトのリスト）
        self.strategies = strategies

    def output_mail(self):
        """メール全体を出力する。"""
        print(f"お世話になっております、{self.name}です。")
        self._construct_sentence()
        print(self.line)
        print(f"{self.name}\n{self.mail_addr}\n{self.company}\n{self.addr}")
        print(self.line)

    def _construct_sentence(self):
        """保持しているストラテジ（関数）を順番に実行する。"""
        for strategy_func in self.strategies:
            # 関数オブジェクトを () を付けて実行する
            strategy_func()

# -----------------------------------------------
# 3. 実行する
# -----------------------------------------------

# キャンペーン期間を定義
term = "2025年1月4日(土) 〜 1月8日(水)"

# 引数が必要なストラテジは、先にファクトリ関数を呼び出して生成しておく
campaign_func = create_campaign_strategy(term)

# 利用したいストラテジ（関数）をリストにまとめる
strategies = [ introduction_strategy, campaign_func ]

# Mailクラスにストラテジのリストを渡してインスタンス化
mail = Mail(strategies)
mail.output_mail()

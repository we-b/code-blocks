# Djangoのmodelsモジュールをインポート
from django.db import models

# models.Modelクラスを継承して、Articleクラスを定義
class Article(models.Model):
    # このクラスに独自の属性（フィールド）を定義するだけでよい
    title = models.CharField(max_length=100)
    content = models.TextField()

# これだけで、Articleクラスはデータベース操作の全機能を持つ
# article = Article(title="新しい記事", content="...")
# article.save()  # 親クラスから継承したsave関数を使える

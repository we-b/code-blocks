from abc import ABC, abstractmethod

# -----------------------------------------------
# ストラテジの定義 (各文章パーツ)
# -----------------------------------------------

class MailStrategy(ABC):
    """全てのストラテジが従うべき共通のインターフェースを定義する親クラス。"""
    @abstractmethod
    def output_contents(self):
        pass

class AgreementStrategy(MailStrategy):
    """契約完了の文章を出力するストラテジ。"""
    def output_contents(self):
        print("この度は、ご契約ありがとうございました。")

class IntroductionStrategy(MailStrategy):
    """商品紹介の文章を出力するストラテジ。"""
    def output_contents(self):
        print("この度は、商品のご紹介です。弊社で取り扱っている「〇〇」は多くのお客様にご使用されています。")
        print("体験会を2025年10月10日 10:00より行っておりますので、是非ご参加ください。")

class VacationInfoStrategy(MailStrategy):
    """休暇のお知らせを出力するストラテジ。"""
    def output_contents(self):
        print("\n\n年末年始の休暇のお知らせ")
        print("2024年12月29日 ～ 2025年1月3日")
        print("新年の営業は1月4日午前9時からとなります。")

class CampaignInfoStrategy(MailStrategy):
    """キャンペーンのお知らせを出力するストラテジ。"""
    def __init__(self, term):
        self.term = term
        
    def output_contents(self):
        print("\nまた、弊社では1月4日より「新年キャンペーン」を開催いたします。")
        print("この機会に、ぜひお買い求めください。")
        print("\nキャンペーンの詳細")
        print(f"開催期間：{self.term}")
        print("対象商品：〇〇 30％OFF(他のキャンペーンとは併用不可)")

# -----------------------------------------------
# コンテキストの定義 (メール全体)
# -----------------------------------------------

class Mail:
    def __init__(self, strategies):
        # メール全体の共通情報
        self.name = '田中 太郎'
        self.mail_addr = 'sample@test.com'
        self.company = '株式会社パイソン'
        self.addr = '東京都渋谷区1-1-1'
        self.line = '■----------------------------------■'
        # 利用するストラテジのリスト
        self.strategies = strategies

    def output_mail(self):
        """メール全体を出力する。"""
        print(f"お世話になっております、{self.name}です。")
        self._construct_sentence()
        print(self.line)
        print(f"{self.name}\n{self.mail_addr}\n{self.company}\n{self.addr}")
        print(self.line)

    def _construct_sentence(self):
        """保持しているストラテジを使って本文を組み立てる。"""
        for strategy in self.strategies:
            strategy.output_contents()

# -----------------------------------------------
# 実行部分
# -----------------------------------------------

# パターン1：契約完了＋キャンペーン＋休暇のお知らせ
print("--- パターン1 ---")
term = "2025年1月4日(土) 〜 1月8日(水)"
strategies1 = [
    AgreementStrategy(), 
    CampaignInfoStrategy(term),
    VacationInfoStrategy()
]
mail = Mail(strategies1)
mail.output_mail()

print("\n" + "="*40 + "\n") # 区切り線

# パターン2：商品紹介のみ（ストラテジを動的に入れ替え）
print("--- パターン2 ---")
mail.strategies = [IntroductionStrategy()]
mail.output_mail()

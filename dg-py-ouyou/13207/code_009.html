class Engine:
    def start(self):
        print("エンジンが始動しました。")

class Car:
    def __init__(self):
        # Carの初期化と同時に、内部でEngineのインスタンスを生成する
        self.engine = Engine()

    def drive(self):
        print("車の運転を開始します。")
        self.engine.start()

# Carのインスタンスを作成するだけで、内部のエンジンも自動的に作られる
my_car = Car()
my_car.drive()

# CarオブジェクトがEngineオブジェクトを内部に持っていることを確認
print(f"my_carのインスタンス: {my_car}")
print(f"my_carが持つengine: {my_car.engine}")

from abc import ABC, abstractmethod

# ----------------------------------------------------
# 1. 親クラスと、第一階層の子クラス
# ----------------------------------------------------
class Mail(ABC):
    """メールの骨格を定義する親クラス（抽象クラス）"""
    def __init__(self):
        self.name = "田中 太郎"
        self.mail_addr = "sample@test.com"
        self.company = "株式会社パイソン"
        self.addr = "東京都渋谷区1-1-1"
        self.line = "■----------------------------------■"

    def output_mail(self):
        self.output_greeting()
        self.output_body()
        self.output_info()

    def output_greeting(self):
        print(f"お世話になっております、{self.name}です。")

    @abstractmethod
    def output_body(self):
        raise NotImplementedError('サブクラスでoutput_bodyを実装してください。')

    def output_info(self):
        print(self.line)
        print(f"{self.name}\n{self.mail_addr}\n{self.company}\n{self.addr}")
        print(self.line)

class DoneMail(Mail):
    """契約完了メールの本文を定義するクラス"""
    def output_body(self):
        print("この度は、ご契約ありがとうございました。")

class IntroductionMail(Mail):
    """商品紹介メールの本文を定義するクラス"""
    def output_body(self):
        print("この度は、商品のご紹介です。弊社で取り扱っている「〇〇」は多くのお客様にご使用されています。")
        print("体験会を2025年10月10日 10:00より行っておりますので、是非ご参加ください。")

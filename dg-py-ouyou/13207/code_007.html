class Engine:
    def start(self):
        print("エンジンが始動しました。")

class Car:
    def __init__(self, engine):
        # CarはEngineのインスタンスを部品として持つ
        self.engine = engine

    def drive(self):
        print("車の運転を開始します。")
        self.engine.start()

# 1. まずEngineのインスタンスを作成
my_engine = Engine()

# 2. 作成したエンジンを渡してCarのインスタンスを作成
my_car = Car(my_engine)

# 3. Carの機能を使うと、内部のEngineの機能も呼び出される
my_car.drive()

# CarオブジェクトがEngineオブジェクトを内部に持っていることを確認
print(f"my_carのインスタンス: {my_car}")
print(f"my_carが持つengine: {my_car.engine}")

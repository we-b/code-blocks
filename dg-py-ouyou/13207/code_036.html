# 引数が必要な場合は、関数を返す関数（高階関数）を使う
def create_campaign_strategy(term):
    def campaign_strategy():
        print("\nまた、弊社では1月4日より「新年キャンペーン」を開催いたします。")
        # ...(中略)...
        print(f"開催期間：{term}")
    # 中で定義した関数を返す
    return campaign_strategy

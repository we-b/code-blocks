# ファイルへの書き込みを担当するクラス
class FileManager:
    def __init__(self, path: str):
        # path名のファイルを書き込みモード("w")で開く
        # encoding='utf-8' は文字化けを防ぐための推奨設定
        self.file = open(path, "w", encoding="utf-8")

    def write_line(self, line: str):
        # f-stringを使って、渡された文字列に改行を加えて書き込む
        self.file.write(f"{line}\n")
    
    def close(self):
        # ファイルを確実に閉じるためのメソッド
        self.file.close()

# FileManagerを装飾(Decorate)して、箇条書きの機能を追加するクラス
class BulletWriter:
    def __init__(self, real_writer):
        # ↓装飾したいオブジェクトをインスタンス変数として保持する
        self.real_writer = real_writer

    def write_line(self, text: str):
        # 受け取ったテキストの先頭に "- " を付けてから、
        # 保持しているwriterオブジェクトのwrite_lineメソッドを呼び出す
        self.real_writer.write_line(f"- {text}")

    def close(self):
        # close処理も、保持しているオブジェクトに委譲する
        self.real_writer.close()

# --- 実行コード ---
# try...finallyブロックを使うことで、途中でエラーが発生しても
# 確実に .close() メソッドが呼ばれ、ファイルが安全に閉じられる
writer = BulletWriter(FileManager('bullet.txt'))
try:
    writer.write_line('箇条書きの1つ目です。')
    writer.write_line('箇条書きの2つ目です。')
    writer.write_line('箇条書きの3つ目です。')
finally:
    writer.close()

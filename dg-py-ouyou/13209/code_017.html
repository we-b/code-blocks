# --- 書き込みの本体となるクラス ---
class FileManager:
    def __init__(self, path: str):
        # path名のファイルを作成し、インスタンス変数 self.file に保持する
        self.file = open(path, "w", encoding="utf-8")

    # (write_line, closeメソッドは省略)


# --- 全てのデコレーターの基底となるクラス ---
class FileManagerDecorator:
    def __init__(self, real_writer):
        # 装飾対象のオブジェクトをインスタンス変数 self._real_writer に保持する
        self._real_writer = real_writer

    # (write_line, closeメソッドは省略)


# --- 行番号を付与するデコレーター ---
class LineNumberWriter(FileManagerDecorator):
    def __init__(self, real_writer):
        # 親クラスの__init__を呼び出し、装飾対象をセットする
        super().__init__(real_writer)
        # 行番号の初期値をインスタンス変数 self.line_number に保持する
        self.line_number = 1


# --- 空白行を無視するデコレーター ---
class BlankLineWriter(FileManagerDecorator):
    # このデコレーターは追加の属性を持たないため、__init__の実装は不要
    # (親クラスの__init__が自動的に呼ばれる)
    pass

from abc import ABC, abstractmethod

# 装飾される対象となる基本クラス (Component)
class TextComponent:
    def __init__(self, text=""):
        self.text = text

    def get_text(self):
        return self.text

# Decoratorの親クラス (ABC: Abstract Base Class)
# Componentと同じインターフェース (get_text) を持つ
class Decorator(ABC):
    def __init__(self, component: TextComponent):
        # 【 ① 】: 装飾対象のコンポーネントを保持する
        
    @abstractmethod
    def get_text(self):
        pass

# 具体的なDecorator A
class PlainTextDecorator(Decorator):
    # 【 ② 】: 親クラスの抽象メソッドを実装する
    def get_text(self):
        return f"--- Plain Text ---\n{self.component.get_text()}"

# 具体的なDecorator B
class HtmlDecorator(Decorator):
    # 【 ③ 】: 親クラスの抽象メソッドを実装する
    def get_text(self):
        base_text = self.component.get_text()
        return f"<html><body><p>{base_text}</p></body></html>"

# --- 実行部分 ---
# 元のテキストオブジェクトを作成
original_text = TextComponent("これはDecoratorパターンのテストです。")

# PlainTextDecoratorで装飾
plain_decorated = PlainTextDecorator(original_text)
print(plain_decorated.get_text())
print("-" * 20)

# HtmlDecoratorで装飾
html_decorated = HtmlDecorator(original_text)
print(html_decorated.get_text())
print("-" * 20)

# Decoratorを連結（装飾を重ねがけ）することも可能
# original_text -> PlainTextDecorator -> HtmlDecorator
chained_decoration = HtmlDecorator(PlainTextDecorator(original_text))
print(chained_decoration.get_text())

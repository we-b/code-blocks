# ファイル操作のためのモジュールをインポート
from abc import ABC, abstractmethod

class FileManager:
    def __init__(self, path):
        # path名のファイルを作成
        self.file = open(path, "w")

    def write_line(self, line):
        self.file.write(line + "\n")

class FileManageDecorator(ABC):
    def __init__(self, real_writer):
        self.real_writer = real_writer

    @abstractmethod
    def write_line(self, line):
        pass

class LineNumberWriter(FileManageDecorator):
    def __init__(self, real_writer):
        super().__init__(real_writer)
        self.line_number = 1

    def write_line(self, text):
        self.real_writer.write_line(f"{self.line_number}: {text}")
        self.line_number += 1

class BlankLineWriter(FileManageDecorator):
    def write_line(self, text):
        self.real_writer.write_line(text)
        self.real_writer.write_line("")

# 使用例
line_writer = LineNumberWriter(FileManager('line.txt'))
line_writer.write_line('1行目へ記述するテキストです。')
line_writer.write_line('2行目へ記述するテキストです。')
line_writer.write_line('3行目へ記述するテキストです。')

blank_writer = BlankLineWriter(FileManager('blank.txt'))
blank_writer.write_line('1行目に記述し、次は1行空行にします。')
blank_writer.write_line('3行目に記述し、次は1行空行にします。')
blank_writer.write_line('5行目に記述し、次は1行空行にします。')

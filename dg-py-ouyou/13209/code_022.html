class FileManager:
    # ... __init__とcloseは省略 ...
    def write_line(self, line: str):
        # 渡された文字列(line)の末尾に改行を加えてファイルに書き込む
        self.file.write(f"{line}\n")


class FileManagerDecorator:
    # ... __init__とcloseは省略 ...
    def write_line(self, text: str):
        # 保持しているオブジェクトのwrite_lineメソッドを呼び出す
        self._real_writer.write_line(text)


class LineNumberWriter(FileManagerDecorator):
    # ... __init__は省略 ...
    def write_line(self, text: str):
        # 行番号とテキストを連結した文字列を作成し、書き込み処理を委譲する
        self._real_writer.write_line(f"{self.line_number}: {text}")
        # 次の行のために、行番号を1つ増やす
        self.line_number += 1


class BlankLineWriter(FileManagerDecorator):
    # ... __init__は省略 ...
    def write_line(self, text: str):
        # 渡されたテキストを書き込んだ後、さらに空行を1行書き込む
        self._real_writer.write_line(text)
        self._real_writer.write_line("")

from django.db import models

# Schema Information (スキーマ情報)
#
# Table name: your_app_user (テーブル名は自動的に 'アプリ名_モデル名' になる)
#
# id:     bigint, not null, primary key (自動で作成される)
# name:   varchar(255)
# gender: integer, not null, default: 0

class User(models.Model):
    """
    Userモデル
    """
    # 性別の選択肢をクラス内で定義 (IntegerChoices)
    class Gender(models.IntegerChoices):
        UNDEFINED = 0, '未定義'
        MALE = 1, '男性'
        FEMALE = 2, '女性'

    name = models.CharField(max_length=255, blank=True, null=True)
    gender = models.IntegerField(
        choices=Gender.choices,
        default=Gender.UNDEFINED,
        null=False
    )

    def __str__(self):
        return self.name

    class Meta:
        # (任意) テーブル名を明示的に指定する場合
        db_table = 'users'

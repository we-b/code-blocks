from django.shortcuts import render
from .models import User
from .decorators import UserDecorator

def user_profile(request, user_id):
    # 1. データベースからUserオブジェクトを取得
    user_object = User.objects.get(id=user_id)
    
    # 2. UserオブジェクトをDecoratorでラップする
    decorated_user = UserDecorator(user_object)
    
    # 3. Decoratorをテンプレートに渡す
    # テンプレートでは decorated_user.name_with_honorific() や
    # decorated_user.name のように両方のメソッド/属性が使える
    return render(request, 'profile.html', {'user': decorated_user})

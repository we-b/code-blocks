from .models import User, Gender

class UserDecorator:
    def __init__(self, user: User):
        # 1. コンストラクタでUserモデルのインスタンスを受け取る
        self.user = user

    def name_with_honorific(self):
        """ 2. 追加したい機能（敬称付きの名前を返す）を実装する """
        if self.user.gender == Gender.MALE:
            prefix = "Mr."
        elif self.user.gender == Gender.FEMALE:
            prefix = "Ms."
        else:
            prefix = ""
        
        return f"{prefix} {self.user.name}"

    def __getattr__(self, name):
        """
        3. Decoratorにない属性やメソッドは、元のuserオブジェクトに問い合わせる
        これにより、decorated_user.name や decorated_user.id のように
        Userモデルが持つ属性に直接アクセスできる。
        """
        return getattr(self.user, name)

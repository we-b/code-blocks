# 印刷機のシミュレーション
# このコードは印刷機の基本的な機能を模倣したものです

class Printer:
    def __init__(self, jobs):
        self.jobs = jobs

    # ① 印刷待ちリストに追加された順に印刷できるように末尾に追加するようにしている
    def add_job(self, job):
        self.jobs.append(job)
        print(f"印刷待ちリストに '{job}' を追加しました")

    # ② 印刷待ちリストに追加された順に印刷できるように印刷待ちリストの先頭を印刷
    def print(self):
        if not self.jobs:
            print("印刷待ちのジョブはありません")
            return
        job = self.jobs.pop(0)
        # ③ pop(0)メソッドを使用しないと元々のリストの先頭が消えない
        print(f"'{job}' を印刷中...")
        print(f"印刷が完了しました: {job}")

    def display_jobs(self):
        # ④ TODO: 画面に表示させるフォーマットを整える
        print("\n現在の印刷待ちリスト:")
        if not self.jobs:
            print("印刷待ちのジョブはありません")
        else:
            for i, job in enumerate(self.jobs, 1):
                print(f"{i}. {job}")

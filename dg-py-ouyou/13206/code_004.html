class Mail:
    def __init__(self):
        self.name = '田中 太郎'
        self.mail_addr = 'sample@test.com'
        self.company = '株式会社テックキャンプ'
        self.addr = '東京都渋谷区1-1-1'
        self.line = '■----------------------------------■'

    def output_mail(self, format):
        print(f"お世話になっております、{self.name}です。")
        if format == '契約完了':
            print('この度は、ご契約ありがとうございました。')
        elif format == '商品紹介':
            print(f"お世話になっております、{self.name}です。")
            print('この度は、商品のご紹介です。弊社で取り扱っている「〇〇」は多くのお客様にご使用されています。')
            print('体験会を10/10 10:00より行っておりますので、是非ご参加ください。')
        elif format == '会社紹介':
            print('先日は突然のご挨拶に対応していただき、誠にありがとうございました。')
            print('この度は、弊社サービスが御社の〇〇事業でお役に立てると思い、ご連絡を差し上げた次第です。')
            print('よろしければ、ぜひご連絡ください。')
        elif format == '退職の挨拶':
            print('このたび、一身上の都合により３月末で退社することになり、本日が最終出社日となりました。')
            print('本来であれば直接ご挨拶をすべきところ、メールでの挨拶にて失礼いたします。')
            print('在籍中はいたらぬ点もあったかと思いますが、お世話になりありがとうございました。')
            print('退職後の連絡先は下記になりますので、今後ともご連絡いただけると幸いです。')
            print('メール：hoge@mail.com')
            print('携帯：080-XXXX-XXXX')
        print(self.line)
        print(f"{self.name}\n{self.mail_addr}\n{self.company}\n{self.addr}")
        print(self.line)

mail = Mail()
mail.output_mail('契約完了')
mail.output_mail('商品紹介')
mail.output_mail('会社紹介')
mail.output_mail('退職の挨拶')

class PCShop:
    def __init__(self, individual, corporate):
        self.individual = individual
        self.corporate = corporate

    def notice_software_update(self, software_update):
        print(f"{self.individual.kind_of_customer}の顧客にソフトウェアアップデートを通知します")
        self.individual.download_software_update(software_update)
        print(f"{self.corporate.kind_of_customer}の顧客にソフトウェアアップデートを通知します")
        self.corporate.download_software_update(software_update)

class Customer:
    def __init__(self):
        self.updates = []

    def kind_of_customer(self):
        raise NotImplementedError('顧客の種類をサブクラスで定義してください')

    def download_software_update(self, software_update):
        self.updates.append(software_update)
        print(f"新しい更新情報:{software_update}")

class Individual(Customer):  # 個人
    def kind_of_customer(self):
        return '個人'

class Corporate(Customer):  # 法人
    def kind_of_customer(self):
        return '法人'

pc_shop = PCShop(Individual(), Corporate())
pc_shop.notice_software_update('ソフトウェアver1.02')

from django.db import models
from django.db.models.signals import post_save
from django.dispatch import receiver
from slack_sdk import WebClient

class User(models.Model):
    email = models.EmailField()
    password = models.CharField(max_length=128)

@receiver(post_save, sender=User)
def notify_password_update(sender, instance, **kwargs):
    if instance.password != instance._original_password:
        AdminMailer.password_update_email(instance.email).send()
        # Slack通知用のクライアントを作成
        client = WebClient(token=settings.SLACK_API_TOKEN)
        # Slackに通知する
        client.chat_postMessage(
            channel="#password-updates",
            text=f"{instance.email}さんがパスワードを変更しました！"
        )

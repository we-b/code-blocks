# 従業員クラス
class Employee:
    def __init__(self, name, salary, manager):
        self.name = name
        self.salary = salary
        self.manager = manager

    @property
    def salary(self):
        return self._salary

    @salary.setter
    def salary(self, new_salary):
        self._salary = new_salary
        self.manager.update(self)  # 情報を上司に送信

# 上司クラス
class Manager:
    def update(self, changed_employee):
        print(f"{changed_employee.name}さんの給料は{changed_employee.salary}円です！")

takashi = Employee('takashi', 10000, Manager())
takashi.salary = 9999999

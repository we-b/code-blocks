# 従業員クラス
class Employee:
    def __init__(self, name, salary):
        self.name = name
        self.salary = salary
        # observerの一覧
        self.observers = []

    def add_observer(self, observer):
        self.observers.append(observer)

    @property
    def salary(self):
        return self._salary

    @salary.setter
    def salary(self, new_salary):
        self._salary = new_salary

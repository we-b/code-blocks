from pubsub import pub

class Employee:
    def __init__(self, name, salary):
        self.name = name
        self.salary = salary

    @property
    def salary(self):
        return self._salary

    @salary.setter
    def salary(self, new_salary):
        self._salary = new_salary
        pub.sendMessage('salary_changed', employee=self)

class Manager:
    def __init__(self):
        pub.subscribe(self.update, 'salary_changed')

    def update(self, employee):
        print(f"上司：{employee.name}さんの給料は{employee.salary}円です！")

class Accountant:
    def __init__(self):
        pub.subscribe(self.update, 'salary_changed')

    def update(self, employee):
        print(f"経理部門：{employee.name}さんの給料は{employee.salary}円です!")

takashi = Employee('takashi', 10000)
manager = Manager()
accountant = Accountant()
takashi.salary = 9999999

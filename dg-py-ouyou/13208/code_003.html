# 従業員クラス
class Employee:
    def __init__(self, name, salary, manager, accountant):
        self.name = name
        self.salary = salary
        self.manager = manager
        # この行が追加されている
        self.accountant = accountant

    @property
    def salary(self):
        return self._salary

    @salary.setter
    def salary(self, new_salary):
        self._salary = new_salary
        self.manager.update(self)
        self.accountant.update(self)

# 上司クラス
class Manager:
    def update(self, changed_employee):
        print(f"{changed_employee.name}さんの給料は{changed_employee.salary}円です！")

# 経理担当クラス
class Accountant:
    def update(self, changed_employee):
        print(f"{changed_employee.name}さんの給料は{changed_employee.salary}円です!")

takashi = Employee('takashi', 10000, Manager(), Accountant())
takashi.salary = 9999999

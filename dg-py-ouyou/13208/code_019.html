# observers.py
class UserObserver:
    def __init__(self):
        self.observers = []

    def add_observer(self, observer):
        self.observers.append(observer)

    def notify_observers(self, user):
        for observer in self.observers:
            observer.update(user)

# models.py
from django.db import models
from django.db.models.signals import post_save
from django.dispatch import receiver
from .observers import UserObserver

class User(models.Model):
    email = models.EmailField()
    password = models.CharField(max_length=128)

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._original_password = self.password
        self.observer = UserObserver()

    def save(self, *args, **kwargs):
        if self.password != self._original_password:
            self.observer.notify_observers(self)
        super().save(*args, **kwargs)

from abc import ABC, abstractmethod

# 抽象基底クラスHuman
class Human(ABC):
    @abstractmethod
    def talk(self):
        pass

    @abstractmethod
    def walk(self):
        pass

    @abstractmethod
    def run(self):
        pass

# 具象クラスJapaneseは、抽象クラスHumanを継承
class Japanese(Human):
    def talk(self):
        print("話します")

    def walk(self):
        print("歩きます")

    def run(self):
        print("走ります")

    # 食事前の挨拶は、日本人独自の関数として定義
    def greeting_before_meals(self):
        print("いただきます")

mr_tanaka = Japanese()
mr_tanaka.talk()
mr_tanaka.greeting_before_meals()

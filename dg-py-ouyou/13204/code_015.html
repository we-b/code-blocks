class A:
    def __put_hoge(self):  # ダブルアンダースコアで疑似private
        print("Hoge")

    def call_private_method(self):
        self.__put_hoge() # クラス内からは呼び出し可能

a = A()
a.call_private_method() # 出力: Hoge

# 外部から直接呼び出そうとするとエラーになる
# a.__put_hoge() # AttributeError: 'A' object has no attribute '__put_hoge'

class ContentsData:
    def __init__(self):
        self.data_title = 'コンテンツのタイトル'
        self.data = 'コンテンツの内容'
        import datetime
        self.created_at = datetime.datetime(2021, 1, 1)

    def replace_date_jp(self):
        self.date_jp = self.created_at.strftime("%Y年%-m月%-d日")

    def replace_date_us(self):
        self.date_us = self.created_at.strftime("%-m/%-d/%Y")

class Printer:
    def __init__(self):
        self.data = ContentsData()

    def print(self, num):
        for _ in range(num):
            print(self.data.data)

    def replace_date_jp(self):
        self.data.replace_date_jp()

    def replace_date_us(self):
        self.data.replace_date_us()

printer = Printer()
printer.print(5)

from abc import ABC, abstractmethod

class Product(ABC):
    def __init__(self, name='', price=0, weight=0):
        self.name = name
        self.price = price
        self.weight = weight

    @abstractmethod
    def unique_info(self):
        pass

class PC(Product):
    def __init__(self, info, parts):
        super().__init__(**info)
        self.parts = parts

    def unique_info(self):
        self.spec_sheet = f"""
        スペック表\nCPU:{self.parts.cpu}\nRAM:{self.parts.ram}GB\nROM:{self.parts.rom}GB
        """
        return self.spec_sheet

class PCParts:
    def __init__(self, cpu='', ram='', rom=''):
        self.cpu = cpu
        self.ram = ram
        self.rom = rom

pc_info = { 'name': 'tcbook', 'price': 120_000, 'weight': 1200 }
pc_spec = { 'cpu': 'core i5', 'ram': 8, 'rom': 256 }
pc_parts = PCParts(**pc_spec)
pc = PC(pc_info, pc_parts)

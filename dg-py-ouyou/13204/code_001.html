class Curriculum:
    def __init__(self, title, content):
    # 属性を設定
        self.title = title
        self.content = content

    def put_attr(self):
        print(self.title)
        print(self.content)

title = "クラス図とは"
content = "クラス図とは…のことです。"
curriculum = Curriculum(title, content)
curriculum.put_attr()

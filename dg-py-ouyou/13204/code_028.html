class PC:
    def __init__(self, data, config):
        self.data = data
        self.config = config

class Printer:
    def __init__(self, pc, num):
        self.data = pc.data
        self.config = pc.config
        self.num = num

    def print(self):
        print(f"{self.data}を{self.config}で{self.num}部印刷しました。")

pc = PC("会議資料", "モノクロ")
printer = Printer(pc, 100)
printer.print()
# => "会議資料をモノクロで100部印刷しました。"

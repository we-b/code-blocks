class A:
    def _protected_method(self):  # アンダースコア1つでprotected
        print("This is a protected method")

class B(A):
    def call_protected(self):
        self._protected_method()  # サブクラス内からは呼び出し可能

# サブクラスのインスタンスを作成
b = B()
# サブクラスの関数経由でprotected関数を呼び出す
b.call_protected()  # 出力: This is a protected method

# 外部から直接アクセスすることも可能だが、非推奨
a = A()
a._protected_method() # 出力: This is a protected method

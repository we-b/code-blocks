from abc import ABC, abstractmethod

class Nation(ABC):
    def __init__(self, country_name, currency):
        print(f"{country_name}の通貨は{currency}です。")

class Japan(Nation):
    def __init__(self, country_name, currency):
        self.country_name = country_name
        self.currency = currency
        super().__init__(self.country_name, self.currency)

class USA(Nation):
    pass

Japan("日本", "日本円")
USA("アメリカ", "米ドル")

# プロダクトの抽象クラス
class Product:
    def __init__(self, name='', price=0, weight=0):
        self.name = name
        self.price = price
        self.weight = weight

# 抽象クラスであるProductによって生成するパソコンの具象クラス
class PC(Product):
    def __init__(self, info, parts):
        super().__init__(**info)
        self.parts = parts

# パソコンのパーツ
class PCParts:
    def __init__(self, cpu='', ram='', rom=''):
        self.cpu = cpu
        self.ram = ram
        self.rom = rom

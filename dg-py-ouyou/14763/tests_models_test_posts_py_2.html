from django.test import TestCase
from django.core.exceptions import ValidationError
from ..factories.posts import PostFactory

class PostModelTestCase(TestCase):
    def setUp(self):
        self.test_post = PostFactory.build()

    def test_content_cannot_be_blank(self):
        self.test_post.content = ""

        with self.assertRaises(ValidationError) as cm:
            self.test_post.full_clean()

        self.assertIn('content', cm.exception.message_dict)
        self.assertEqual(cm.exception.message_dict['content'], ["このフィールドは空ではいけません。"])

    def test_title_cannot_be_blank(self):
        self.test_post.title = ""

        with self.assertRaises(ValidationError) as cm:
            self.test_post.full_clean()

        self.assertIn('title', cm.exception.message_dict)
        self.assertEqual(cm.exception.message_dict['title'], ["このフィールドは空ではいけません。"])

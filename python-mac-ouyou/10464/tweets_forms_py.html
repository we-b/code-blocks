from .models import Tweet
from django import forms

class TweetForm(forms.ModelForm):
    class Meta:
      model = Tweet
      fields = ['image', 'text'] # nameを削除
      widgets = {
         'name': forms.TextInput(attrs={'placeholder': 'Nickname'}),# この行を削除
         'text': forms.Textarea(attrs={'placeholder': 'Text', 'rows': 10}),
      }
      labels = {
         'name':'Nickname',# この行を削除
         'text': 'テキスト',
      }

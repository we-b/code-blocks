'use client';

import { useState, useEffect } from 'react';
import { useParams, useRouter } from 'next/navigation';
import { findTweetById, updateTweet } from '../../../api/tweets';
import Header from '../../../_components/Header';
import Footer from '../../../_components/Footer';
import TweetForm from '../../../_components/TweetForm';
import AuthGuard from '../../../_components/AuthGuard';

interface TweetFormData {
  text: string;
  image: string;
}

export default function EditTweetPage() {
  const params = useParams();
  const router = useRouter();
  const tweetId = Number(params.tweetId);
  const [initialData, setInitialData] = useState<TweetFormData>({ text: '', image: '' });
  const [errorMessages, setErrorMessages] = useState<string[]>([]);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    const fetchTweet = async () => {
      try {
        const tweet = await findTweetById(tweetId);
        setInitialData({ text: tweet.text, image: tweet.image || '' });
      } catch (error) {
        console.error('ツイートの取得に失敗しました:', error);
      } finally {
        setIsLoading(false);
      }
    };
    fetchTweet();
  }, [tweetId]);

  const handleSubmit = async (data: TweetFormData) => {
    try {
      setErrorMessages([]);
      await updateTweet(tweetId, data);
      router.push(`/tweets/${tweetId}`);
    } catch (error: unknown) {
      if (error && typeof error === 'object' && 'response' in error) {
        const axiosError = error as { response?: { data?: { messages?: string[] } } };
        const messages = axiosError.response?.data?.messages;
        setErrorMessages(messages && messages.length > 0 ? messages : ['更新に失敗しました']);
      } else {
        setErrorMessages(['更新に失敗しました']);
      }
    }
  };

  if (isLoading) {
    return <div className="loading">Loading...</div>;
  }

  return (
    <AuthGuard>
      <Header />
      <div className="contents row">
        <div className="container">
          <h3>編集する</h3>
          <TweetForm
            initialData={initialData}
            errorMessages={errorMessages}
            onSubmit={handleSubmit}
          />
        </div>
      </div>
      <Footer />
    </AuthGuard>
  );
}

'use client';
import { useEffect } from 'react';
import { useForm } from 'react-hook-form';

interface TweetFormData {
  text: string;
  image: string;
}

interface TweetFormProps {
  initialData: TweetFormData;
  errorMessages: string[];
  onSubmit: (data: TweetFormData) => void;
}

const TweetForm = ({ initialData, errorMessages, onSubmit }: TweetFormProps) => {
  const { register, handleSubmit, reset, formState: { errors } } = useForm<TweetFormData>({
    defaultValues: initialData,
  });

  useEffect(() => {
    reset(initialData);
  }, [initialData, reset]);

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      {errorMessages.length > 0 && (
        <div className="error-messages">
          {errorMessages.map((msg, index) => (
            <p key={index} className="error-message">{msg}</p>
          ))}
        </div>
      )}

      {errors.image && <span className="error-message">{errors.image.message}</span>}
      <input
        type="text"
        {...register('image')}
        placeholder="Image Url"
      />

      {errors.text && <span className="error-message">{errors.text.message}</span>}
      <textarea
        {...register('text', { required: 'テキストを入力してください' })}
        placeholder="text"
        rows={10}
      />

      <input type="submit" value="SEND" />
    </form>
  );
};

export default TweetForm;

'use client';

import { useState, useEffect } from 'react';
import { useParams, useRouter } from 'next/navigation';
import Link from 'next/link';
import { findUserById } from '../../api/users';
import { deleteTweet } from '../../api/tweets';
import { UserPageData } from '../../_interfaces/UserPageData';
import Header from '../../_components/Header';
import Footer from '../../_components/Footer';
import TweetList from '../../_components/TweetList';

export default function UserPage() {
  const params = useParams();
  const router = useRouter();
  const userId = Number(params.userId);
  const [userData, setUserData] = useState<UserPageData | null>(null);

  useEffect(() => {
    const fetchUser = async () => {
      try {
        const data = await findUserById(userId);
        setUserData(data);
      } catch (error) {
        console.error('ユーザー情報の取得に失敗しました:', error);
      }
    };
    fetchUser();
  }, [userId]);

  const handleDeleteTweet = async (tweetId: number) => {
    if (!window.confirm('本当に削除しますか？')) return;
    try {
      await deleteTweet(tweetId);
      setUserData((prev) => {
        if (!prev) return prev;
        return {
          ...prev,
          tweets: prev.tweets.filter((tweet) => tweet.id !== tweetId),
        };
      });
    } catch (error) {
      console.error('ツイートの削除に失敗しました:', error);
    }
  };

  if (!userData) {
    return <div className="loading">Loading...</div>;
  }

  return (
    <>
      <Header />
      <div className="contents row">
        <div className="container">
          <h3>{userData.nickname}</h3>
          <p>投稿数: {userData.tweets.length}</p>
        </div>
        <TweetList tweets={userData.tweets} onDeleteTweet={handleDeleteTweet} />
      </div>
      <Footer />
    </>
  );
}

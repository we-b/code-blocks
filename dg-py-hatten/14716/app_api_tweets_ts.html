import axios from 'axios';
import { TweetData } from '../_interfaces/TweetData';

const api = axios.create({
  baseURL: 'http://localhost:8000/api',
  withCredentials: true,
});

export const findAllTweets = async (): Promise<TweetData[]> => {
  const response = await api.get('/tweets/');
  return response.data;
};

export const createTweet = async (data: {
  text: string;
  image: string;
}): Promise<void> => {
  await api.post('/tweets/', data);
};

export const findTweetById = async (id: number): Promise<TweetData> => {
  const response = await api.get(`/tweets/${id}/`);
  return response.data;
};

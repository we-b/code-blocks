'use client';

import { useState, useEffect } from 'react';
import { useParams } from 'next/navigation';
import Link from 'next/link';
import { findTweetById } from '../../api/tweets';
import { useAuthContext } from '../../context/AuthContext';
import { TweetData } from '../../_interfaces/TweetData';
import Header from '../../_components/Header';
import Footer from '../../_components/Footer';

export default function TweetDetailPage() {
  const params = useParams();
  const tweetId = Number(params.tweetId);
  const { user } = useAuthContext();
  const [tweet, setTweet] = useState<TweetData | null>(null);

  useEffect(() => {
    const fetchTweet = async () => {
      try {
        const data = await findTweetById(tweetId);
        setTweet(data);
      } catch (error) {
        console.error('ツイートの取得に失敗しました:', error);
      }
    };
    fetchTweet();
  }, [tweetId]);

  if (!tweet) {
    return <div className="loading">Loading...</div>;
  }

  const isOwner = user?.isAuthenticated && user.id === tweet.user.id;

  return (
    <>
      <Header />
      <div className="contents row">
        <div className="content_post" style={{ backgroundImage: tweet.image ? `url(${tweet.image})` : 'none' }}>
          {isOwner && (
            <div className="more">
              <span><img src="/images/arrow_top.png" alt="Arrow Top" /></span>
              <ul className="more_list">
                <li>
                  <Link href={`/tweets/${tweet.id}/edit`}>編集</Link>
                </li>
                <li>
                  <form action={`/`} method="post">
                    <input type="submit" className="delete-btn" value="削除" />
                  </form>
                </li>
              </ul>
            </div>
          )}
          <p>{tweet.text}</p>
          <span className="name">
            <Link href={`/users/${tweet.user.id}`}>
              <span>投稿者</span><span>{tweet.user.nickname}</span>
            </Link>
          </span>
        </div>
      </div>
      <Footer />
    </>
  );
}

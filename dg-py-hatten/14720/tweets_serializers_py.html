from rest_framework import serializers

from .models import Tweet
from comments.serializers import CommentSerializer
from users.serializers import UserSerializer


class TweetListSerializer(serializers.ModelSerializer):
    """ツイート一覧用のシリアライザ（ユーザー情報をネスト）"""

    user = UserSerializer(read_only=True)

    class Meta:
        model = Tweet
        fields = ['id', 'text', 'image', 'user', 'created_at']


class TweetCreateUpdateSerializer(serializers.ModelSerializer):
    """ツイート作成・更新用のシリアライザ"""

    class Meta:
        model = Tweet
        fields = ['text', 'image']


class TweetDetailSerializer(serializers.ModelSerializer):
    """ツイート詳細用のシリアライザ（コメント情報付き）"""

    user = UserSerializer(read_only=True)
    comments = CommentSerializer(many=True, read_only=True)

    class Meta:
        model = Tweet
        fields = ['id', 'text', 'image', 'user', 'comments', 'created_at']

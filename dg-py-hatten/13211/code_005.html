import { render, screen } from '@testing-library/react';
import { expect, test, vi, describe } from 'vitest';
import Home from '../app/page';
import React from 'react';

describe('Homeコンポーネント', () => {
  test('APIリクエスト成功時に投稿内容が表示される', async () => {
    global.fetch = vi.fn(() =>
      Promise.resolve({
        ok: true,
        json: () => Promise.resolve({ posts: [{ id: 1, content: 'Hello World' }] }),
      } as Response)
    );

    render(<Home />);

    const postContent = await screen.findByText('Hello World');
    expect(postContent).toBeDefined();
  });

  test('APIリクエスト失敗時にエラーメッセージが表示される', async () => {
    global.fetch = vi.fn(() => Promise.reject(new Error("APIリクエスト失敗")));

    render(<Home />);

    const errorMessage = await screen.findByText('投稿を取得できませんでした');
    expect(errorMessage).toBeDefined();
  });
});

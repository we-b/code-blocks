'use client';

import { useState, useEffect } from 'react';
import { useSearchParams } from 'next/navigation';
import { searchTweets, deleteTweet } from '../api/tweets';
import { TweetData } from '../_interfaces/TweetData';
import Header from './Header';
import Footer from './Footer';
import TweetList from './TweetList';
import SearchForm from './SearchForm';

const SearchPageClient = () => {
  const searchParams = useSearchParams();
  const query = searchParams.get('query') || '';
  const [tweets, setTweets] = useState<TweetData[]>([]);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    const fetchResults = async () => {
      setIsLoading(true);
      try {
        if (query) {
          const results = await searchTweets(query);
          setTweets(results);
        } else {
          setTweets([]);
        }
      } catch (error) {
        console.error('検索に失敗しました:', error);
      } finally {
        setIsLoading(false);
      }
    };
    fetchResults();
  }, [query]);

  const handleDeleteTweet = async (tweetId: number) => {
    if (!window.confirm('本当に削除しますか？')) return;
    try {
      await deleteTweet(tweetId);
      setTweets(tweets.filter((tweet) => tweet.id !== tweetId));
    } catch (error) {
      console.error('ツイートの削除に失敗しました:', error);
    }
  };

  return (
    <>
      <Header />
      <SearchForm defaultKeyword={query} />
      {isLoading ? (
        <div className="loading">Loading...</div>
      ) : tweets.length === 0 ? (
        <div className="contents row">
          <p>「{query}」に一致するツイートは見つかりませんでした</p>
        </div>
      ) : (
        <TweetList tweets={tweets} onDeleteTweet={handleDeleteTweet} />
      )}
      <Footer />
    </>
  );
};

export default SearchPageClient;

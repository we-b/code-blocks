'use client';

import { useForm } from 'react-hook-form';
import { useRouter } from 'next/navigation';

interface SearchFormData {
  keyword: string;
}

interface SearchFormProps {
  defaultKeyword?: string;
}

const SearchForm = ({ defaultKeyword = '' }: SearchFormProps) => {
  const router = useRouter();
  const { register, handleSubmit } = useForm<SearchFormData>({
    defaultValues: { keyword: defaultKeyword },
  });

  const onSubmit = (data: SearchFormData) => {
    if (!data.keyword.trim()) return;
    router.push(`/tweets/search?query=${encodeURIComponent(data.keyword)}`);
  };

  return (
    <div className="contents row">
      <form onSubmit={handleSubmit(onSubmit)} className="search-form">
        <input type="text" {...register('keyword')} placeholder="キーワードを入力" />
        <input type="submit" value="検索" />
      </form>
    </div>
  );
};

export default SearchForm;

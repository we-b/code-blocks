import { test, expect } from '@playwright/test';
import { prisma, resetDatabase } from "./utils/testUtils";
import { createPost } from "./createPost"; // 追記

test.afterAll(async () => {
  await resetDatabase();
  await prisma.$disconnect();
});

test('投稿を編集して、内容が更新されていることが確認できる', async ({ page }) => {
  await createPost(page, "テスト投稿"); // 投稿の処理を削除しこの行を追記

  const editButton = page.locator('button:has-text("編集")').last();
  await editButton.click();

  const editInput = page.locator('input[name="update-content"]')
  await expect(editInput).toHaveValue('テスト投稿');

  await editInput.fill('編集後のテスト投稿');

  const updateButton = page.locator('button:has-text("更新")')
  await updateButton.click();

  const updatedText = page.locator('text=編集後のテスト投稿')
  await expect(updatedText).toBeVisible();
});

test('空白文字を入力すると編集に失敗し、エラーメッセージが確認できる', async ({ page }) => {
  await createPost(page, "テスト投稿"); // 投稿の処理を削除しこの行を追記

  const editButton = page.locator('button:has-text("編集")').last();
  await editButton.click();

  const editInput = page.locator('input[name="update-content"]')
  await expect(editInput).toHaveValue('テスト投稿');

  await editInput.fill('');

  const updateButton = page.locator('button:has-text("更新")')
  await updateButton.click();

  const errorMessage = page.locator('.error-message:has-text("更新に失敗しました")');
  await expect(errorMessage).toBeVisible();
});

import { test, expect } from "@playwright/test";

test("入力フォームに値を入力すると、データの保存ができる", async ({ page }) => {
  await page.goto("http://localhost:3000/");

  const pageTitle = page.locator('text=トップページ');
  await expect(pageTitle).toBeVisible();

  const inputForm = page.locator('input[name="content"]');
  await expect(inputForm).toBeVisible();

  await inputForm.fill('テスト投稿');

  const postButton = page.locator('button:has-text("投稿する")');
  await postButton.click();

  // 以下の記述を追加する
  const postedText = page.locator('text=テスト投稿').last();
  await expect(postedText).toBeVisible();
});

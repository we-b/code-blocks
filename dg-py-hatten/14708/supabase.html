await supabase.from("posts").select("*");
// => postsテーブルの全てのデータを取得する

await supabase.from("posts").select("*").eq("id", 1);
// => postsテーブルのidが1のデータを取得する

await supabase.from("posts").insert({ content: "新規登録" });
// => postsテーブルのcontentカラムに"新規登録"を登録する

await supabase.from("posts").update({ content: "更新" }).eq("id", 1);
// => postsテーブルのidが1のデータのcontentを"更新"に変更する

await supabase.from("posts").delete().eq("id", 1);
// => postsテーブルのidが1のデータを削除する

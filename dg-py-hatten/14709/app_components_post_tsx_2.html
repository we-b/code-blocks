"use client";

import { ResponsePost } from "../_type/ResponsePost";
import { useState } from "react";

interface PostProps {
  post: ResponsePost;
}

export const Post: React.FC<PostProps> = ({ post }) => {
  const [isEditing, setIsEditing] = useState(false);

  return(
    <li className="post">
      {isEditing ? (
        <>
          <div className="form">
            <input
              type="text"
              name="update-content"
              className="input-form"
            />
            <button className="btn">更新</button>
          </div>
        </>
      ) : (
        <>
          <p>{post.content}</p>
          <div className="btn-list">
            <button className="btn" onClick={() => setIsEditing(true)}>編集</button>
          </div>
        </>
      )}
    </li>
  )
}

"use client";

import { ResponsePost } from "../_type/ResponsePost";
import { useState, useRef } from "react";

interface PostProps {
  post: ResponsePost;
}

export const Post: React.FC<PostProps> = ({ post }) => {
  const [isEditing, setIsEditing] = useState(false);
  const content = useRef<HTMLInputElement>(null);
  const [editContent, setEditContent] = useState(post.content);
  const [error, setError] = useState<string | null>(null);

  // 編集処理
  const handleUpdate = async (e: React.MouseEvent<HTMLButtonElement>) => {
    e.preventDefault();

    if (!content.current?.value) {
      setError("内容を入力してください");
      return;
    }

    try {
      const response = await fetch(`/api/posts/${post.id}`, {
        method: "PUT",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ content: content.current.value }),
      });

      if (!response.ok) {
        throw new Error("更新に失敗しました");
      }

      const data = await response.json();
      setEditContent(data.post.content);
      setIsEditing(false);
      setError(null);
    } catch (error) {
      console.log("APIリクエストエラー", error);
      setError("更新に失敗しました");
    }
  };

  return(
    <li className="post">
      {error && <div className="error-message">{error}</div>}
      {isEditing ? (
        <>
          <div className="form">
            <input
              type="text"
              name="update-content"
              className="input-form"
              ref={content}
              defaultValue={editContent}
            />
            <button className="btn" onClick={handleUpdate}>更新</button>
          </div>
        </>
      ) : (
        <>
          <p>{post.content}</p>
          <div className="btn-list">
            <button className="btn" onClick={() => setIsEditing(true)}>編集</button>
          </div>
        </>
      )}
    </li>
  )
}

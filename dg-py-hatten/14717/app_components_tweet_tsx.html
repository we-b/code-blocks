'use client';
import Link from 'next/link';
import { TweetData } from '../_interfaces/TweetData';
import { useAuthContext } from '../context/AuthContext';

interface TweetProps {
  tweet: TweetData;
  onDeleteTweet: (id: number) => void;
}

const Tweet = ({ tweet, onDeleteTweet }: TweetProps) => {
  const { user } = useAuthContext();
  const isOwner = user?.isAuthenticated && user.id === tweet.user.id;

  return (
    <div className="content_post" style={{ backgroundImage: tweet.image ? `url(${tweet.image})` : 'none' }}>
      <div className="more">
        <span><img src="/images/arrow_top.png" alt="Arrow Top" /></span>
        <ul className="more_list">
          <li>
            <Link href={`/tweets/${tweet.id}`}>詳細</Link>
          </li>
          {isOwner && (
            <>
              <li>
                <Link href={`/tweets/${tweet.id}/edit`} className="update-btn">編集</Link>
              </li>
              <li>
                <Link href="#" onClick={(e) => { e.preventDefault(); onDeleteTweet(tweet.id); }}>削除</Link>
              </li>
            </>
          )}
        </ul>
      </div>
      <p>{tweet.text}</p>
      <span className="name">
        <Link href={`/users/${tweet.user.id}`}>
          <span>投稿者</span><span>{tweet.user.nickname}</span>
        </Link>
      </span>
    </div>
  );
};

export default Tweet;

SELECT H.hub_id, 
COUNT(M1.hub_id) AS 修理が必要なPC,
COUNT(M2.hub_id) AS 修理が不要なPC
FROM hub H
INNER JOIN manage_pc M1
ON H.hub_id = M1.hub_id AND M1.repair = '必要'
INNER JOIN Hub H2
ON H.hub_id = H2.hub_id
INNER JOIN manage_pc M2
ON M2.hub_id = M2.hub_id AND M2.repair = '不要'
WHERE H.hub_id = 1
GROUP BY H.hub_id;

@Nested
@DisplayName("POST /tasks - 優先度機能のテスト")
class PriorityTests {

    @Test
    @DisplayName("優先度「high」を指定してタスクを作成できること")
    void shouldCreateTaskWithHighPriority() throws Exception {
        Map<String, Object> request = new HashMap<>();
        request.put("title", "重要なタスク");
        request.put("content", "急ぎの対応が必要");
        request.put("status", "todo");
        request.put("genre_id", testGenre.getId());
        request.put("priority", "high");

        mockMvc.perform(post("/tasks")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content(objectMapper.writeValueAsString(request)))
                .andExpect(status().isCreated())
                .andExpect(jsonPath("$.title").value("重要なタスク"))
                .andExpect(jsonPath("$.priority").value("high"));
    }

    @Test
    @DisplayName("優先度「low」を指定してタスクを作成できること")
    void shouldCreateTaskWithLowPriority() throws Exception {
        Map<String, Object> request = new HashMap<>();
        request.put("title", "急がないタスク");
        request.put("content", "時間のある時に対応");
        request.put("status", "todo");
        request.put("genre_id", testGenre.getId());
        request.put("priority", "low");

        mockMvc.perform(post("/tasks")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content(objectMapper.writeValueAsString(request)))
                .andExpect(status().isCreated())
                .andExpect(jsonPath("$.title").value("急がないタスク"))
                .andExpect(jsonPath("$.priority").value("low"));
    }

    @Test
    @DisplayName("優先度を指定しない場合、デフォルト値「medium」が設定されること")
    void shouldDefaultToMediumPriority() throws Exception {
        Map<String, Object> request = new HashMap<>();
        request.put("title", "通常のタスク");
        request.put("content", "普通の優先度");
        request.put("status", "todo");
        request.put("genre_id", testGenre.getId());
        // priority を指定しない

        mockMvc.perform(post("/tasks")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content(objectMapper.writeValueAsString(request)))
                .andExpect(status().isCreated())
                .andExpect(jsonPath("$.priority").value("medium"));
    }

    @Test
    @DisplayName("タスク作成後のレスポンスJSONに priority が含まれること")
    void shouldIncludePriorityInResponse() throws Exception {
        Map<String, Object> request = new HashMap<>();
        request.put("title", "レスポンス確認タスク");
        request.put("status", "todo");
        request.put("genre_id", testGenre.getId());
        request.put("priority", "high");

        mockMvc.perform(post("/tasks")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content(objectMapper.writeValueAsString(request)))
                .andExpect(status().isCreated())
                .andExpect(jsonPath("$.priority").exists())
                .andExpect(jsonPath("$.priority").value("high"));
    }
}

public class AdService {

  public void displayAd(String adType, int imageCount) {
    displayHeader();
    displayBody(adType, imageCount);
    displayFooter();
  }

  private void displayHeader() {
    System.out.println("株式会社〇〇");
  }

  private void displayBody(String adType, int imageCount) {
    switch (adType) {
      case "gif":
        if (imageCount == 2) {
          System.out.println("GIF広告: 2つの画像を表示");
        } else {
          System.out.println("GIF広告: 1つの画像を表示");
        }
        break;
      case "image":
        System.out.println("画像広告を表示");
        break;
      case "movie":
        System.out.println("動画広告を表示");
        break;
      default:
        System.out.println("デフォルト画像を表示");
        break;
    }
  }

  private void displayFooter() {
    System.out.println("詳しくはこちら");
  }
}

public class Main {
  public static void main(String[] args) {
      AdService adService = new AdService();
      adService.displayAd("gif", 1);         // GIF広告を表示
      adService.displayAd("image", 0);       // 画像広告を表示
      adService.displayAd("movie", 0);       // ムービー広告を表示
      adService.displayAd("unknown", 0);     // 未知の広告形式
  }
}

public class Cooking {
  private String type;

  public Cooking(String type) {
    this.type = type;
  }

  public void prepareDish() {
    gatherIngredients();

    // 条件分岐を使って、料理の種類に応じた調理方法を呼び出す
    if (type.equals("カレー")) {
      cookCurry();
    } else if (type.equals("パスタ")) {
      cookPasta();
    }

    serve();
  }

  private void gatherIngredients() {
    System.out.println("材料を集める");
  }

  private void cookCurry() {
    System.out.println("食材を煮込む");
  }

  private void cookPasta() {
    System.out.println("パスタを茹でる");
    System.out.println("トッピングと和える");
  }

  private void serve() {
    System.out.println("盛り付ける");
  }
}

public class Main {
  public static void main(String[] args) {
    Cooking curryCooking = new Cooking("カレー");
    curry.prepareDish();

    Cooking pastaCooking = new Cooking("パスタ");
    pasta.prepareDish();
  }
}

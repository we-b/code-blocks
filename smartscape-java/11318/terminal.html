import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    try (Scanner scanner = new Scanner(System.in)) {
      System.out.println("作りたい料理を選んでください:");
      System.out.println("1: カレー");
      System.out.println("2: パスタ");
      int choice = scanner.nextInt();
      
      Cooking selectedCooking = null;
      
      switch (choice) {
        case 1:
          selectedCooking = new CurryCooking();
          break;
        case 2:
          selectedCooking = new PastaCooking();
          break;
        default:
          System.out.println("無効な選択です。");
          return; // プログラムを終了
      }
        
      selectedCooking.prepareDish();
    }
  }
}

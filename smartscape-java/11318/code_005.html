public class Product {
  private String name;

  public Product(String name) {
    this.name = name;
  }

  public void sell() {
    if (name.equals("自転車")) {
      System.out.println("商品をお買い求めですか？");
      System.out.println("商品の購入にはカードもしくは現金が使えます");
      System.out.println("商品の値段は20000円です");
    } else if (name.equals("ゲームカセット")) {
      System.out.println("商品をお買い求めですか？");
      System.out.println("商品の購入にはカードもしくは現金が使えます");
      System.out.println("商品の値段は5000円です");
    }
  }
}

public class Main {
  public static void main(String[] args) {
    Product bicycle = new Product("自転車");
    bicycle.sell();
    
    Product gameCassette = new Product("ゲームカセット");
    gameCassette.sell();
  }
}

// 大きなインターフェース
public interface CompleteCar {
  void 走る();
  void 自動走行();
  void 衝突回避();
}

// 全ての機能を実装したクラス
public class AutonomousCar implements CompleteCar {
  public void 走る() {
    System.out.println("通常走行します");
  }

  public void 自動走行() {
    System.out.println("自動走行を開始します。");
  }

  public void 衝突回避() {
    System.out.println("衝突回避機能が有効になりました。");
  }
}

// 通常の車クラス
public class RegularCar implements CompleteCar {
  public void 走る() {
    System.out.println("走行します");
  }

  // 自動走行や衝突回避機能の実装はないが、文法的に実装する必要がある
  public void 自動走行() {
    throw new UnsupportedOperationException("この車は自動走行機能を持っていません。");
  }

  public void 衝突回避() {
    throw new UnsupportedOperationException("この車は衝突回避機能を持っていません。");
  }
}

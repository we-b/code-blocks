public abstract class Robot {
  public abstract void operate();
}

public class CleaningRobot extends Robot {
  @Override
  public void operate() {
    System.out.println("掃除を開始します。");
  }
}

public class CookingRobot extends Robot {
  @Override
  public void operate() {
    System.out.println("料理を開始します。");
  }
}

public abstract class RobotFactory {
  public Robot createRobot() {
    return createRobotType();
  }

  protected abstract Robot createRobotType();
}

public class CleaningRobotFactory extends RobotFactory {
  @Override
  protected Robot createRobotType() {
    Robot cleaningRobot = new CleaningRobot();
    // cleaningRobotに部屋の間取りデータを読み込む処理
    return cleaningRobot;
  }
}

public class CookingRobotFactory extends RobotFactory {
  @Override
  protected Robot createRobotType() {
    Robot cookingRobot = new CookingRobot();
    // cookingRobotにメニューやレシピのデータを読み込む処理
    return cookingRobot;
  }
}

public class Main {
  public static void main(String[] args) {
    RobotFactory cleaningRobotFactory = new CleaningRobotFactory();
    Robot cleaningRobot = cleaningRobotFactory.createRobot();
    cleaningRobot.operate();
    
    RobotFactory cookingRobotFactory = new CookingRobotFactory();
    Robot cookingRobot = cookingRobotFactory.createRobot();
    cookingRobot.operate();
  }
}

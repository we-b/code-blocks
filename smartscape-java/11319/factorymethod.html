// プロダクト
public abstract class Component {
  void operation();
}

public class ComponentA implements Component {
  @Override
  public void operation() {
    System.out.println("ComponentAの操作を実行中");
  }
}

public class ComponentB implements Component {
  @Override
  public void operation() {
    System.out.println("ComponentBの操作を実行中");
  }
}

// ファクトリー
public abstract class Factory {
  public abstract Component createComponent();
}

public class ComponentAFactory extends Factory {
  @Override
  public Component 【 ① 】{
    return new ComponentA();
  }
}

public class ComponentBFactory extends Factory {
  @Override
  public Component  【 ② 】 {
    return new ComponentB();
  }
}

// メインクラスで実行
public class Main {
  public static void main(String[] args) {
    Factory factoryA = new ComponentAFactory();
    Component componentA = factoryA.createComponent();
    componentA.operation();  // ComponentAの操作を実行中

    Factory factoryB = new ComponentBFactory();
    Component componentB = factoryB.createComponent();
    componentB.operation();  // ComponentBの操作を実行中
  }
}

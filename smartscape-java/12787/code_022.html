// health: 健康状態 => {weight: 体重(kg)、fat: 脂肪率(%)}

const reducer = (state, action) => {
  switch (action.type) {
    case '食事':
      return {weight: state.weight + 5, fat: state.fat + 5 };
    case 'サウナ':
      return {weight: state.weight - 5, fat: state.fat - 5 };
    case '筋トレ':
      return {weight: state.weight + 5, fat: state.fat - 5 };
  }
}

const [health, dispatch] = useReducer(reducer, {weight: 60, fat: 15});
console.log('初期状態: ' + health)

dispatch({type: '食事'})   
console.log('食事後: ' + health)
dispatch({type: 'サウナ'})
console.log('サウナ後: ' + health)
dispatch({type: '筋トレ'}) 
console.log('筋トレ後: ' + health)


// 出力結果
初期状態: {weight: 60, fat: 15}
食事後: {weight: 65, fat: 20}
サウナ後: {weight: 60, fat: 15}
筋トレ後: {weight: 65, fat: 10}

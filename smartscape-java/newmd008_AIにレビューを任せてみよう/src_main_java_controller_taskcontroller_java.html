// ※コンストラクタに TaskDuplicateService を追加し、フィールドに保持する
    private final TaskDuplicateService taskDuplicateService;

    // タスク複製
    @PostMapping("/{id}/duplicate")
    public ResponseEntity<?> duplicate(@PathVariable Long id) {
        try {
            Task duplicated = taskDuplicateService.duplicate(id);

            Map<String, Object> taskMap = new LinkedHashMap<>();
            taskMap.put("id", duplicated.getId());
            taskMap.put("title", duplicated.getTitle());
            taskMap.put("content", duplicated.getContent());
            taskMap.put("status", duplicated.getStatus());
            taskMap.put("priority", duplicated.getPriority());

            if (duplicated.getGenre() != null) {
                Map<String, Object> genreMap = new LinkedHashMap<>();
                genreMap.put("id", duplicated.getGenre().getId());
                genreMap.put("name", duplicated.getGenre().getName());
                taskMap.put("genre", genreMap);
            } else {
                taskMap.put("genre", null);
            }

            taskMap.put("created_at", duplicated.getCreatedAt());
            taskMap.put("updated_at", duplicated.getUpdatedAt());

            return ResponseEntity.ok(taskMap);
        } catch (RuntimeException e) {
            Map<String, String> error = new HashMap<>();
            error.put("error", "タスクが見つかりません");
            return ResponseEntity.status(HttpStatus.NOT_FOUND).body(error);
        }
    }

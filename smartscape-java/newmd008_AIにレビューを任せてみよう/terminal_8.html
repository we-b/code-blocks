  1. TaskDuplicateService.java

  @Service
  public class TaskDuplicateService {

      private static final String TITLE_SUFFIX = "(コピー)";
      private static final String INITIAL_STATUS = "todo";

      private final TaskRepository taskRepository;

      public TaskDuplicateService(TaskRepository taskRepository) {
          this.taskRepository = taskRepository;
      }

      public Task duplicate(Long taskId) {
          Task original = taskRepository.findById(taskId)
                  .orElseThrow(() -> new ResponseStatusException(
                      HttpStatus.NOT_FOUND, "Task not found"));

          Task duplicated = buildDuplicate(original);
          return taskRepository.save(duplicated);
      }

      private Task buildDuplicate(Task original) {
          Task duplicated = new Task();
          duplicated.setTitle(original.getTitle() + TITLE_SUFFIX);
          duplicated.setContent(original.getContent());
          duplicated.setGenre(original.getGenre());
          duplicated.setPriority(original.getPriority());
          duplicated.setStatus(INITIAL_STATUS);
          return duplicated;
      }
  }

import java.util.regex.Pattern;

public class FullNameValidation {
    public static void main(String[] args) {
        String fullName = "山田はな子";

        // 正規表現のパターン
        String regex = "^[ぁ-んァ-ヶ一-龥々ー]+$";
        boolean isValid = Pattern.matches(regex, fullName);

        if (isValid) {
            System.out.println("全角文字のみです: " + fullName);
        } else {
            System.out.println("全角文字以外が含まれています: " + fullName);
        }
    }
}

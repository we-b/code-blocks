// 自動車クラスの定義
class Car {
    // 属性（フィールド）
    String brand; // ブランド名
    int speed;    // 現在のスピード

    // コンストラクタ（初期化メソッド）: 新しい車を生成するときに呼び出され、ブランド名を設定します
    public Car(String brand) {
        this.brand = brand;
        this.speed = 0; // 初期スピードは0
    }

    // メソッド: 車のスピードを上げる処理
    public void accelerate(int increment) {
        speed += increment; // スピードを上げる
        System.out.println(brand + "のスピードが " + speed + " km/h になりました。");
    }

    // メソッド: 車を止める処理
    public void brake() {
        speed = 0; // スピードを止める
        System.out.println(brand + "が止まりました。");
    }
}

public class Main {
    public static void main(String[] args) {
        // 自動車クラスからインスタンスを生成する
        Car car1 = new Car("トヨタ");
        Car car2 = new Car("ホンダ");

        // メソッドを使って動作を実行する
        car1.accelerate(50); // 出力: トヨタのスピードが 50 km/h になりました。
        car2.accelerate(60); // 出力: ホンダのスピードが 60 km/h になりました。

        car1.brake(); // 出力: トヨタが止まりました。
    }
}

public class ImageCache {
    // キー: 画像ID, 値: 画像データ(byte配列)
    private static Map<String, byte[]> cache = new HashMap<>();

    public static void addImage(String imageId, byte[] imageData) {
        // 画像サイズが大きい場合、大量のメモリを消費する可能性がある
        cache.put(imageId, imageData);
        System.out.println(imageId + " をキャッシュに追加しました。現在のキャッシュサイズ: " + cache.size());
    }

    public static byte[] getImage(String imageId) {
        return cache.get(imageId);
    }

    // 問題点: キャッシュから削除する仕組みがない！
}

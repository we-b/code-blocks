package in.tech_camp.task_checker_back.service;

import in.tech_camp.task_checker_back.entity.GenreEntity;
import in.tech_camp.task_checker_back.entity.TaskEntity;
import in.tech_camp.task_checker_back.repository.GenreRepository;
import in.tech_camp.task_checker_back.repository.TaskRepository;
import org.springframework.stereotype.Service;

import java.util.*;

@Service
public class TaskService {

    private final TaskRepository taskRepository;
    private final GenreRepository genreRepository;

    public TaskService(TaskRepository taskRepository, GenreRepository genreRepository) {
        this.taskRepository = taskRepository;
        this.genreRepository = genreRepository;
    }

    /**
     * タスクを作成する
     * @param requestBody リクエストボディ
     * @return 作成結果（TaskEntityまたはエラーリスト）
     */
    public Map<String, Object> createTask(Map<String, Object> requestBody) {
        Map<String, Object> result = new HashMap<>();

        // バリデーション
        List<String> errors = new ArrayList<>();

        if (requestBody.get("name") == null
                || requestBody.get("name").toString().trim().isEmpty()) {
            errors.add("名前は必須です");
        }

        Integer status = 0;
        if (requestBody.get("status") != null) {
            try {
                status = Integer.parseInt(requestBody.get("status").toString());
                if (status < 0 || status > 5) {
                    errors.add("ステータスは 0〜5 の整数で指定してください");
                }
            } catch (NumberFormatException e) {
                errors.add("ステータスは 0〜5 の整数で指定してください");
            }
        }

        if (requestBody.get("priority") != null) {
            String priority = requestBody.get("priority").toString();
            if (!priority.equals("low") && !priority.equals("medium")
                    && !priority.equals("high")) {
                errors.add("優先度は low, medium, high のいずれかを指定してください");
            }
        }

        if (!errors.isEmpty()) {
            result.put("success", false);
            result.put("errors", errors);
            return result;
        }

        // エンティティの組み立て
        TaskEntity task = new TaskEntity();
        task.setName(requestBody.get("name").toString().trim());

        if (requestBody.get("explanation") != null) {
            task.setExplanation(requestBody.get("explanation").toString());
        }

        task.setStatus(status);

        if (requestBody.get("priority") != null) {
            task.setPriority(requestBody.get("priority").toString());
        }

        // ジャンルの関連付け
        if (requestBody.get("genreId") != null) {
            Integer genreId;
            try {
                genreId = Integer.parseInt(requestBody.get("genreId").toString());
            } catch (NumberFormatException e) {
                result.put("success", false);
                result.put("errors", List.of("genreId は数値で指定してください"));
                return result;
            }

            Optional<GenreEntity> genreOpt = genreRepository.findById(genreId.longValue());
            if (genreOpt.isEmpty()) {
                result.put("success", false);
                result.put("errors", List.of("指定されたジャンルが見つかりません"));
                return result;
            }
            task.setGenreId(genreId);
        }

        // 保存
        TaskEntity savedTask = taskRepository.save(task);

        result.put("success", true);
        result.put("task", savedTask);
        return result;
    }
}

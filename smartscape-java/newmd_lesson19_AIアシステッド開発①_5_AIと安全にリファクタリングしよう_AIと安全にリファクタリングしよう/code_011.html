package com.example.taskchecker.service;

import com.example.taskchecker.entity.Genre;
import com.example.taskchecker.entity.Task;
import com.example.taskchecker.repository.GenreRepository;
import com.example.taskchecker.repository.TaskRepository;
import org.springframework.stereotype.Service;

import java.util.*;

@Service
public class TaskService {

    private final TaskRepository taskRepository;
    private final GenreRepository genreRepository;

    public TaskService(TaskRepository taskRepository, GenreRepository genreRepository) {
        this.taskRepository = taskRepository;
        this.genreRepository = genreRepository;
    }

    /**
     * タスクを作成する
     * @param requestBody リクエストボディ
     * @return 作成結果（Taskまたはエラーリスト）
     */
    public Map<String, Object> createTask(Map<String, Object> requestBody) {
        Map<String, Object> result = new HashMap<>();

        // バリデーション
        List<String> errors = new ArrayList<>();

        if (requestBody.get("title") == null
                || requestBody.get("title").toString().trim().isEmpty()) {
            errors.add("タイトルは必須です");
        }

        String status = "todo";
        if (requestBody.get("status") != null) {
            status = requestBody.get("status").toString();
            if (!status.equals("todo") && !status.equals("doing")
                    && !status.equals("done")) {
                errors.add("ステータスは todo, doing, done のいずれかを指定してください");
            }
        }

        if (requestBody.get("priority") != null) {
            String priority = requestBody.get("priority").toString();
            if (!priority.equals("low") && !priority.equals("medium")
                    && !priority.equals("high")) {
                errors.add("優先度は low, medium, high のいずれかを指定してください");
            }
        }

        if (!errors.isEmpty()) {
            result.put("success", false);
            result.put("errors", errors);
            return result;
        }

        // エンティティの組み立て
        Task task = new Task();
        task.setTitle(requestBody.get("title").toString().trim());

        if (requestBody.get("content") != null) {
            task.setContent(requestBody.get("content").toString());
        }

        task.setStatus(status);

        if (requestBody.get("priority") != null) {
            task.setPriority(requestBody.get("priority").toString());
        }

        // ジャンルの関連付け
        if (requestBody.get("genre_id") != null) {
            Long genreId;
            try {
                genreId = Long.parseLong(requestBody.get("genre_id").toString());
            } catch (NumberFormatException e) {
                result.put("success", false);
                result.put("errors", List.of("genre_id は数値で指定してください"));
                return result;
            }

            Optional<Genre> genreOpt = genreRepository.findById(genreId);
            if (genreOpt.isEmpty()) {
                result.put("success", false);
                result.put("errors", List.of("指定されたジャンルが見つかりません"));
                return result;
            }
            task.setGenre(genreOpt.get());
        }

        // 保存
        Task savedTask = taskRepository.save(task);

        result.put("success", true);
        result.put("task", savedTask);
        return result;
    }
}

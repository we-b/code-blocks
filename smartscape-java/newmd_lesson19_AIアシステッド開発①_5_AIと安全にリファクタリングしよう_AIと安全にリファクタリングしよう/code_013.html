@PostMapping
public ResponseEntity<?> create(@RequestBody Map<String, Object> requestBody) {
    Map<String, Object> result = taskService.createTask(requestBody);

    if (!(boolean) result.get("success")) {
        Map<String, Object> errorResponse = new HashMap<>();
        errorResponse.put("errors", result.get("errors"));
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(errorResponse);
    }

    TaskEntity savedTask = (TaskEntity) result.get("task");

    // レスポンス組み立て
    Map<String, Object> taskMap = new LinkedHashMap<>();
    taskMap.put("id", savedTask.getId());
    taskMap.put("name", savedTask.getName());
    // ... （レスポンスの組み立て）

    return ResponseEntity.status(HttpStatus.CREATED).body(taskMap);
}

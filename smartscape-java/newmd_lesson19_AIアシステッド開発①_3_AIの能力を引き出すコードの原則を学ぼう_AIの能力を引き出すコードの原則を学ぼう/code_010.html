// OK例：ガード節で早期リターンし、ネストを浅く保つ
public String processUser(User user) {
    // userが存在しない場合は、ここで処理を終了
    if (user == null) {
        return "ユーザーが見つかりません";
    }

    // userがアクティブでない場合は、ここで処理を終了
    if (!user.isActive()) {
        return "ユーザーが無効です";
    }

    // 権限がない場合は、ここで処理を終了
    if (!user.hasPermission()) {
        return "権限がありません";
    }

    // ここからがメインの処理
    return "処理が完了しました: " + user.getName();
}

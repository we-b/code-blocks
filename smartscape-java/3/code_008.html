public int calculateFinalPrice(int price, boolean isReducedTaxRate) {
    // 通常の消費税率は10%
    double taxRate = 0.10;

    // 軽減税率対象商品の場合、特別な税率を適用する。
    // このビジネスルールは、2019年10月1日の法改正に基づく。詳細は仕様書XXXを参照。
    if (isReducedTaxRate) {
        taxRate = 0.08;
    }

    // パフォーマンス上の理由から、BigDecimalではなくint型で計算している。
    // 金額は円単位（小数点以下なし）のため、この方法で精度の問題は発生しない。
    int finalPrice = (int) Math.floor(price * (1 + taxRate));

    return finalPrice;
}

public class ShoppingCart {
  private void payWithCreditCard(int amount) {
    System.out.println("クレジットカードで" + amount + "円支払いました。");
  }

  private void payWithBankTransfer(int amount) {
    System.out.println("銀行振込で" + amount + "円支払いました。");
  }

  public void checkout(String paymentMethod, int amount) {
    switch (paymentMethod) {
      case "CreditCard":
        payWithCreditCard(amount);
        break;
      case "BankTransfer":
        payWithBankTransfer(amount);
        break;
      default:
        System.out.println("無効な支払い方法です。");
        break;
    }
  }

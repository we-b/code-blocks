// ============================================================
    // POST /tasks/{id}/duplicate - タスク複製API
    // ============================================================
    @Nested
    @DisplayName("POST /tasks/{id}/duplicate - タスク複製")
    class DuplicateTests {

        // ----------------------------------------------------------
        // 6. 正常系 - HTTPレスポンス
        // ----------------------------------------------------------
        @Nested
        @DisplayName("正常系")
        class SuccessTests {

            @Test
            @DisplayName("正常にリクエストすると HTTP 200 OK が返される")
            void shouldReturn200Ok() throws Exception {
                Task original = new Task("元のタスク", "説明", "done", testGenre);
                original.setPriority("high");
                Task saved = taskRepository.save(original);

                mockMvc.perform(post("/tasks/" + saved.getId() + "/duplicate")
                                .contentType(MediaType.APPLICATION_JSON))
                        .andExpect(status().isOk());
            }

            @Test
            @DisplayName("レスポンスに複製されたタスクの情報が含まれる")
            void shouldContainDuplicatedTaskInResponse() throws Exception {
                Task original = new Task("元のタスク", "説明", "doing", testGenre);
                original.setPriority("high");
                Task saved = taskRepository.save(original);

                mockMvc.perform(post("/tasks/" + saved.getId() + "/duplicate")
                                .contentType(MediaType.APPLICATION_JSON))
                        .andExpect(status().isOk())
                        .andExpect(jsonPath("$.title").value("元のタスク(コピー)"))
                        .andExpect(jsonPath("$.status").value("todo"))
                        .andExpect(jsonPath("$.priority").value("high"));
            }

            @Test
            @DisplayName("タスクが1件追加される")
            void shouldIncrementTaskCount() throws Exception {
                Task original = new Task("元のタスク", "説明", "todo", testGenre);
                Task saved = taskRepository.save(original);
                long countBefore = taskRepository.count();

                mockMvc.perform(post("/tasks/" + saved.getId() + "/duplicate")
                                .contentType(MediaType.APPLICATION_JSON))
                        .andExpect(status().isOk());

                assertThat(taskRepository.count()).isEqualTo(countBefore + 1);
            }

            @Test
            @DisplayName("元のタスクは削除されない")
            void shouldNotDeleteOriginalTask() throws Exception {
                Task original = new Task("元のタスク", "説明", "todo", testGenre);
                Task saved = taskRepository.save(original);

                mockMvc.perform(post("/tasks/" + saved.getId() + "/duplicate")
                                .contentType(MediaType.APPLICATION_JSON))
                        .andExpect(status().isOk());

                assertThat(taskRepository.existsById(saved.getId())).isTrue();
            }
        }

        // ----------------------------------------------------------
        // 8. 異常系 - HTTPエラー
        // ----------------------------------------------------------
        @Nested
        @DisplayName("異常系")
        class ErrorTests {

            @Test
            @DisplayName("存在しない task_id を指定すると HTTP 404 が返される")
            void shouldReturn404ForNonExistentId() throws Exception {
                mockMvc.perform(post("/tasks/999999/duplicate")
                                .contentType(MediaType.APPLICATION_JSON))
                        .andExpect(status().isNotFound());
            }

            @Test
            @DisplayName("タスクは作成されない")
            void shouldNotCreateTaskForNonExistentId() throws Exception {
                long countBefore = taskRepository.count();

                mockMvc.perform(post("/tasks/999999/duplicate")
                                .contentType(MediaType.APPLICATION_JSON))
                        .andExpect(status().isNotFound());

                assertThat(taskRepository.count()).isEqualTo(countBefore);
            }

            @Test
            @DisplayName("負の値の task_id を指定すると HTTP 404 が返される")
            void shouldReturn404ForNegativeId() throws Exception {
                mockMvc.perform(post("/tasks/-1/duplicate")
                                .contentType(MediaType.APPLICATION_JSON))
                        .andExpect(status().isNotFound());
            }
        }
    }

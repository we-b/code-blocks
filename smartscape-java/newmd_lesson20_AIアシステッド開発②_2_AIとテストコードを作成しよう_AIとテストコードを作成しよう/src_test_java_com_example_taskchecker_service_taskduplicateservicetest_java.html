package com.example.taskchecker.service;

import com.example.taskchecker.entity.Genre;
import com.example.taskchecker.entity.Task;
import com.example.taskchecker.repository.GenreRepository;
import com.example.taskchecker.repository.TaskRepository;
import org.junit.jupiter.api.*;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;

import static org.assertj.core.api.Assertions.*;

@SpringBootTest
class TaskDuplicateServiceTest {

    @Autowired
    private TaskDuplicateService taskDuplicateService;

    @Autowired
    private TaskRepository taskRepository;

    @Autowired
    private GenreRepository genreRepository;

    private Genre testGenre;

    @BeforeEach
    void setUp() {
        taskRepository.deleteAll();
        genreRepository.deleteAll();
        testGenre = genreRepository.save(new Genre("テストジャンル"));
    }

    // ============================================================
    // 1. 正常系 - 属性の引き継ぎ
    // ============================================================
    @Nested
    @DisplayName("属性の引き継ぎ")
    class AttributeInheritanceTests {

        @Test
        @DisplayName("title が正しく引き継がれ、末尾に「(コピー)」が追加される")
        void shouldCopyTitleWithSuffix() {
            Task original = createTask("買い物", "内容", "todo", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getTitle()).isEqualTo("買い物(コピー)");
        }

        @Test
        @DisplayName("content がそのまま引き継がれる")
        void shouldCopyContent() {
            Task original = createTask("タスク", "詳細な説明文です", "todo", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getContent()).isEqualTo("詳細な説明文です");
        }

        @Test
        @DisplayName("genre がそのまま引き継がれる")
        void shouldCopyGenre() {
            Task original = createTask("タスク", "内容", "todo", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getGenre().getId()).isEqualTo(testGenre.getId());
        }

        @Test
        @DisplayName("priority が low の場合、そのまま引き継がれる")
        void shouldCopyPriorityLow() {
            Task original = createTask("タスク", "内容", "todo", "low");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getPriority()).isEqualTo("low");
        }

        @Test
        @DisplayName("priority が medium の場合、そのまま引き継がれる")
        void shouldCopyPriorityMedium() {
            Task original = createTask("タスク", "内容", "todo", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getPriority()).isEqualTo("medium");
        }

        @Test
        @DisplayName("priority が high の場合、そのまま引き継がれる")
        void shouldCopyPriorityHigh() {
            Task original = createTask("タスク", "内容", "todo", "high");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getPriority()).isEqualTo("high");
        }
    }

    // ============================================================
    // 2. 正常系 - 特別な処理
    // ============================================================
    @Nested
    @DisplayName("特別な処理")
    class SpecialProcessingTests {

        @Test
        @DisplayName("status が done のタスクを複製すると、新タスクは todo になる")
        void shouldResetStatusFromDone() {
            Task original = createTask("タスク", "内容", "done", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getStatus()).isEqualTo("todo");
        }

        @Test
        @DisplayName("status が doing のタスクを複製すると、新タスクは todo になる")
        void shouldResetStatusFromDoing() {
            Task original = createTask("タスク", "内容", "doing", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getStatus()).isEqualTo("todo");
        }

        @Test
        @DisplayName("status が todo のタスクを複製すると、新タスクも todo になる")
        void shouldKeepStatusTodo() {
            Task original = createTask("タスク", "内容", "todo", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getStatus()).isEqualTo("todo");
        }
    }

    // ============================================================
    // 3. 正常系 - 境界値・エッジケース
    // ============================================================
    @Nested
    @DisplayName("境界値・エッジケース")
    class EdgeCaseTests {

        @Test
        @DisplayName("title が非常に長い文字列でも、末尾に「(コピー)」が追加される")
        void shouldHandleLongTitle() {
            String longTitle = "あ".repeat(100);
            Task original = createTask(longTitle, "内容", "todo", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getTitle()).isEqualTo(longTitle + "(コピー)");
        }

        @Test
        @DisplayName("title に既に「(コピー)」が含まれるタスクを複製すると、「(コピー)(コピー)」になる")
        void shouldAppendCopySuffixToAlreadyCopiedTitle() {
            Task original = createTask("タスク(コピー)", "内容", "todo", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getTitle()).isEqualTo("タスク(コピー)(コピー)");
        }

        @Test
        @DisplayName("content が null のタスクを複製すると、新タスクの content も null になる")
        void shouldHandleNullContent() {
            Task original = createTask("タスク", null, "todo", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getContent()).isNull();
        }

        @Test
        @DisplayName("content が空文字のタスクを複製すると、新タスクの content も空文字になる")
        void shouldHandleEmptyContent() {
            Task original = createTask("タスク", "", "todo", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getContent()).isEqualTo("");
        }
    }

    // ============================================================
    // 4. 正常系 - データの独立性
    // ============================================================
    @Nested
    @DisplayName("データの独立性")
    class DataIndependenceTests {

        @Test
        @DisplayName("複製後、元タスクと新タスクのIDが異なる")
        void shouldHaveDifferentId() {
            Task original = createTask("タスク", "内容", "todo", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            assertThat(duplicated.getId()).isNotEqualTo(original.getId());
        }

        @Test
        @DisplayName("複製によってタスクが1件増加する")
        void shouldIncrementTaskCount() {
            Task original = createTask("タスク", "内容", "todo", "medium");
            long countBefore = taskRepository.count();

            taskDuplicateService.duplicate(original.getId());

            assertThat(taskRepository.count()).isEqualTo(countBefore + 1);
        }

        @Test
        @DisplayName("複製後、元タスクを更新しても新タスクに影響しない")
        void shouldBeIndependentFromOriginal() {
            Task original = createTask("タスク", "元の説明", "todo", "medium");

            Task duplicated = taskDuplicateService.duplicate(original.getId());

            // 元タスクを更新
            original.setTitle("更新後のタスク");
            original.setContent("更新後の説明");
            taskRepository.save(original);

            // 複製したタスクは影響を受けない
            Task reloaded = taskRepository.findById(duplicated.getId()).orElseThrow();
            assertThat(reloaded.getTitle()).isEqualTo("タスク(コピー)");
            assertThat(reloaded.getContent()).isEqualTo("元の説明");
        }
    }

    // ============================================================
    // 5. 異常系
    // ============================================================
    @Nested
    @DisplayName("異常系")
    class ErrorTests {

        @Test
        @DisplayName("存在しないタスクIDを指定すると例外がスローされる")
        void shouldThrowExceptionForNonExistentId() {
            assertThatThrownBy(() ->
                taskDuplicateService.duplicate(999999L)
            ).isInstanceOf(RuntimeException.class);
        }
    }

    // ============================================================
    // ヘルパーメソッド
    // ============================================================
    private Task createTask(String title, String content,
                            String status, String priority) {
        Task task = new Task();
        task.setTitle(title);
        task.setContent(content);
        task.setStatus(status);
        task.setPriority(priority);
        task.setGenre(testGenre);
        return taskRepository.save(task);
    }
}

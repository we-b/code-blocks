const { PrismaClient } = require('@prisma/client');
const prisma = new PrismaClient();

async function transferMoney(senderAccountNumber, receiverAccountNumber, amount) {
  try {
    // $transactionメソッドを使用してトランザクションを開始
    await prisma.$transaction(async (tx) => {
      // 1. 送金元の口座情報を取得
      const sender = await tx.bankAccount.findUnique({
        where: { accountNumber: senderAccountNumber },
      });

      if (!sender) {
        throw new Error("送金元の口座が見つかりません。");
      }

      // 2. 送金先の口座情報を取得
      const receiver = await tx.bankAccount.findUnique({
        where: { accountNumber: receiverAccountNumber },
      });

      if (!receiver) {
        throw new Error("送金先の口座が見つかりません。");
      }

      // 3. 残高不足のチェック
      if (sender.balance < amount) {
        throw new Error("送金元の口座の残高が不足しています。");
      }

      // 4. 送金元の残高を減らす更新
      await tx.bankAccount.update({
        where: { accountNumber: senderAccountNumber },
        data: { balance: { decrement: amount } },
      });

      // 5. 送金先の残高を増やす更新
      await tx.bankAccount.update({
        where: { accountNumber: receiverAccountNumber },
        data: { balance: { increment: amount } },
      });
    });

    console.log("送金が完了しました。");

  } catch (error) {
    // トランザクション内でエラーが発生した場合（throw new Errorなど）、
    // 自動的にロールバックされ、catchブロックに処理が移ります。
    console.error("送金に失敗しました:", error.message);
  } finally {
    await prisma.$disconnect();
  }
}

// 関数の実行例
transferMoney('1234567', '7654321', 10000);

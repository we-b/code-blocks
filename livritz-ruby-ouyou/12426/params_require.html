params  #=> 中身は以下の内容
{
  "utf8" => "✓",
  "authenticity_token" => "token",
  "モデルA" => {
    "image" => "image.jpg",
    "text" => "sample text"
  },
  "commit" => "SEND",
  "controller" => "hoges",
  "action" => "create"
}

params.require(:モデルA)
#=> { "image" => "image.jpg", "text" => "sample text" }

params[:モデルA]  # 結果は同じだが、もしモデルAが存在しない場合はエラーになってくれない
#=> { "image" => "image.jpg", "text" => "sample text" }

params.require(:commit)  # モデルのファイル名以外のキーも指定できる
#=> "SEND"

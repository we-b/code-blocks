// 人間クラスを定義する
class Human {
    // 属性（フィールド）
    String name; // 名前
    int age;     // 年齢

    // コンストラクタ（初期化メソッド）: 新しいインスタンスを作成するときに呼び出され、オブジェクトを初期化します
    public Human(String name, int age) {
        this.name = name;  // nameフィールドに値を代入
        this.age = age;    // ageフィールドに値を代入
    }

    // メソッド（動作）: オブジェクトが行う具体的な操作
    public void introduce() {
        System.out.println("こんにちは、私の名前は " + name + " です。年齢は " + age + " 歳です。");
    }

    // 誕生日の際に年齢を1つ増やす動作を行うメソッド
    public void haveBirthday() {
        age++; // 年齢を1つ増やす
        System.out.println("誕生日おめでとう！年齢が " + age + " 歳になりました。");
    }
}

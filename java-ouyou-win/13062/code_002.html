public class Main {
    public static void main(String[] args) {
        // 人間クラスから新しいインスタンス（具体的な人物）を生成する
        Human person1 = new Human("山田太郎", 30); // 山田太郎という人物を生成
        Human person2 = new Human("鈴木花子", 25); // 鈴木花子という人物を生成

        // メソッドを使って動作を実行する
        person1.introduce(); // 出力: こんにちは、私の名前は 山田太郎 です。年齢は 30 歳です。
        person2.introduce(); // 出力: こんにちは、私の名前は 鈴木花子 です。年齢は 25 歳です。

        // 山田太郎が誕生日を迎える
        person1.haveBirthday(); // 出力: 誕生日おめでとう！年齢が 31 歳になりました。
    }
}

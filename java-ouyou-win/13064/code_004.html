public class ContinueExample {
    public static void main(String[] args) {
        int[] data = {10, 5, 0, 3};
        for (int value : data) {
            try {
                int result = 100 / value;
                System.out.println("Result: " + result);
            } catch (ArithmeticException e) {
                System.out.println("エラー発生: " + e.getMessage());
            }
        }
    }
}

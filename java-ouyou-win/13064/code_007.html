public class TransactionLikeExample {
    public static void main(String[] args) {
        try {
            processStep1();
            processStep2(); // ここで例外が発生したら、以降は実行されない
            processStep3();
        } catch (Exception e) {
            System.out.println("処理中断: " + e.getMessage());
        }
    }

    public static void processStep1() {
        System.out.println("ステップ1 実行中");
    }

    public static void processStep2() throws Exception {
        throw new Exception("ステップ2で失敗しました");
    }

    public static void processStep3() {
        System.out.println("ステップ3 実行中");
    }
}

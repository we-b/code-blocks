public class Human {
  private String body;
  private String personality;

  public Human(String body, String personality) {
    this.body = body;
    this.personality = personality;
  }
}

public class ChildA extends Human {
  public ChildA(String body, String personality) {
    super(body, personality);
  }

  public void outputIndividuality() {
    System.out.println("この子は野球が得意です。");
  }
}

public class ChildB extends Human {
  public ChildB(String body, String personality) {
    super(body, personality);
  }

  public void outputIndividuality() {
    System.out.println("この子はお笑いが好きで、みんなをいつも笑わせています。");
  }
}

package com.example;

public class Example {
  public int publicValue;           // どこからでもアクセス可能

  private int privateValue;         // このクラス内のみアクセス可能

  protected int protectedValue;    // 同パッケージ(com.example)かサブクラスからアクセス可能

  int packagePrivateValue;          // 同パッケージ内のみアクセス可能（デフォルト）

  public int getPrivateValue() {     // メソッドも同様
    return privateValue;
  }
}

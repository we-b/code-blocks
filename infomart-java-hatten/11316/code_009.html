public abstract class Product {
  private String name;
  private int price;
  private int weight;

  public abstract String getUniqueInfo();

  public String getName() {
    return name;
  }
    
  public int getPrice() {
    return price;
  }

  public int getWeight() {
    return weight;
  }
}

@Data
public class Pc extends Product {
  private PcParts parts;

  @Override
  public String getUniqueInfo() {
    return "スペック表\n" +
               "CPU: " + parts.getCpu() + "\n" +
               "RAM: " + parts.getRam() + "GB\n" +
               "ROM: " + parts.getRom() + "GB";
  }
}

@Data
public class PcParts {
  private String cpu;
  private int ram;
  private int rom;

  public String getCpu() {
    return cpu;
  }

  public int getRam() {
    return ram;
  }

  public int getRom() {
    return rom;
  }
}

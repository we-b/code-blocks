public interface Curry {
  String getDescription();
  int cost();
}

public class SimpleCurry implements Curry {
  @Override
  public String getDescription() {
    return "Basic Curry";
  }

  @Override
  public int cost() {
    return 500;
  }
}

public abstract class CurryDecorator implements Curry {
  protected Curry decoratedCurry;

  public CurryDecorator(Curry curry) {
    this.decoratedCurry = curry;
  }

  @Override
  public String getDescription() {
    return decoratedCurry.getDescription();
  }

  @Override
  public int cost() {
    return decoratedCurry.cost();
  }
}

public class CheeseDecorator extends CurryDecorator {
  public CheeseDecorator(Curry curry) {
    super(curry);
  }

  @Override
  public String getDescription() {
    return decoratedCurry.getDescription() + ", with Cheese";
  }

  @Override
  public int cost() {
    return decoratedCurry.cost() + 200;
  }
}

public class VegetableDecorator extends CurryDecorator {
  public VegetableDecorator(Curry curry) {
    super(curry);
  }

  @Override
  public String getDescription() {
    return decoratedCurry.getDescription() + ", with Vegetables";
  }

  @Override
  public int cost() {
    return decoratedCurry.cost() + 150;
  }
}

public class DecoratorPatternExample {
  public static void main(String[] args) {
    Curry basicCurry = new SimpleCurry();
    System.out.println(basicCurry.getDescription() + " Cost: " + basicCurry.cost());

    Curry cheeseCurry = new CheeseDecorator(basicCurry);
    System.out.println(cheeseCurry.getDescription() + " Cost: " + cheeseCurry.cost());

    Curry vegetableCurry = new VegetableDecorator(basicCurry);
    System.out.println(vegetableCurry.getDescription() + " Cost: " + vegetableCurry.cost());

    Curry cheeseVegetableCurry = new VegetableDecorator(cheeseCurry);
    System.out.println(cheeseVegetableCurry.getDescription() + " Cost: " + cheeseVegetableCurry.cost());
  }
}

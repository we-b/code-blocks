@Controller
public class DataController {

  @GetMapping("/process")
  public ResponseEntity<String> processData(@RequestParam String format) {
    DataProcessor processor;
    
    // リクエストパラメータに基づいて適切なプロセッサを選ぶ
    if ("csv".equalsIgnoreCase(format)) {
        processor = new CsvDataProcessor();
    } else if ("json".equalsIgnoreCase(format)) {
        processor = new JsonDataProcessor();
    } else {
        return ResponseEntity.badRequest().body("Invalid data format");
    }

    // 生成したプロセッサを使用して処理を実行
    processor.process();
    return ResponseEntity.ok(format.toUpperCase() + " data processed");
  }
}

import Header from '@/app/_components/Header';

interface LoginCredentials {

  password: string;
}

const LoginPage = () => {
  const { register, handleSubmit, formState: { errors } } = useForm<LoginCredentials>();

  const onSubmit = (formData: LoginCredentials) => {

  };

  return (
    <>

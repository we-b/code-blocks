// ConcreteSubjectクラス
public class PCShop implements Subject {
  private List<Observer> observers;

（省略）

  @Override
  public void notifyObservers(String softwareUpdate) {
    for (Observer observer : observers) {
      observer.downloadSoftwareUpdate(softwareUpdate);
    }
  }

  // 新しいソフトウェアの通知を受け取る
  public void noticeSoftwareUpdate(String softwareUpdate) {
    System.out.println("ソフトウェア更新が通知されました: " + softwareUpdate);
    notifyObservers(softwareUpdate);
  }
}

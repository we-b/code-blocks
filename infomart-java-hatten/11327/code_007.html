public final class Hashtag {
    private final String name;
    private final List<HashtagUser> users;
    private final SpreadTypeRatios ratios;

    public Hashtag(String name, List<HashtagUser> users, SpreadTypeRatios ratios) {
        this.name = name;
        this.users = users;
        this.ratios = ratios;
    }

    public SpreadResult calculateSpread(SpreadCount spreadCount) {
        WeightedSum weightedSum = calcWeightedSum(ratios);
        SpreadCount base = spreadCount.divide(weightedSum);

        Map<HashtagUser, SpreadCount> resultMap = users.stream()
            .collect(Collectors.toMap(
                user -> user,
                user -> base.times(ratios.ratioFor(user.getSpreadType()))
            ));

        return new SpreadResult(resultMap);
    }

    private WeightedSum calcWeightedSum(SpreadTypeRatios ratios) {
        List<Double> userRatios = users.stream()
            .map(user -> ratios.ratioFor(user.getSpreadType()))
            .toList();

        if (userRatios.isEmpty()) {
            throw new IllegalArgumentException("User ratios cannot be empty");
        }

        return WeightedSum.from(userRatios.get(0), 
            userRatios.subList(1, userRatios.size()).toArray(new Double[0]));
    }

    public String getName() {
        return name;
    }

    public List<HashtagUser> getUsers() {
        return users;
    }

    public SpreadTypeRatios getRatios() {
        return ratios;
    }
}

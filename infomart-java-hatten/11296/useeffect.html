import React, { useState, useEffect } from "react";

function App() {
  const [count, setCount] = useState<number>(0);
  
  useEffect(() => {
    console.log("マウント");
    return () => {
      console.log("アンマウント");
    };
  }, []);

  useEffect(() => {
    console.log("アップデート");
  }, [count]);

  return (
    <div
      onClick={() => {
        setCount(count + 1);
      }}
    >
      <h1>Hello World</h1>
    </div>
  );
}

export default App;

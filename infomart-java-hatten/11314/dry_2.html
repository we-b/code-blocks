import java.util.Scanner;

public class PrefectureSelector {

  // 都道府県を選択する共通メソッド
  public static String selectPrefecture(String prompt) {
    String[] prefecture = {"北海道", "青森", "...(省略)...", "沖縄"};
    Scanner scanner = new Scanner(System.in);
    System.out.println(prompt + "（0-" + (prefecture.length - 1) + "）:");
    int num = scanner.nextInt();
    return prefecture[num];
  }

  public static void selectBirthplace() {
    String birthplace = selectPrefecture("出生地を選択してください");
    System.out.println("出生地: " + birthplace);
  }

  public static void selectAddress() {
    String address = selectPrefecture("住所を選択してください");
    System.out.println("住所: " + address);
  }

  public static void selectWorkAddress() {
    String workAddress = selectPrefecture("勤務先住所を選択してください");
    System.out.println("勤務先住所: " + workAddress);
  }

  public static void main(String[] args) {
    selectBirthplace();
    selectAddress();
    selectWorkAddress();
  }
}

await prisma.post.findMany();
// => postテーブルの全てのデータを取得する

await prisma.post.findUnique({where: { id:1}});
// => postテーブルのidが1のデータを取得する

await prisma.post.create({data: {content: "新規登録"});
// => postテーブルのcontentカラムに”新規登録”を登録する

await prisma.post.update({where: {id:1}, data: {content:”更新”}});
// => postテーブルのidが1のデータのcontentを”更新”に変更する

await prisma.post.delete({ where: {id:1}});
// => postテーブルのidが1のデータを削除する

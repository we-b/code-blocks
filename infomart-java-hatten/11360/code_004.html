public class Main {
  public static void main(String[] args) {
    Coffee coffee = new SimpleCoffee;
    System.out.println("Simple Coffee Cost: " + coffee.cost());

    Coffee milkCoffee = new MilkDecorator(coffee);
    System.out.println("Coffee with Milk Cost: " + milkCoffee.cost());

    Coffee milkAndSugarCoffee = new SugarDecorator(milkCoffee);
    System.out.println("Coffee with Milk and Sugar Cost: " + milkAndSugarCoffee.cost());
  }
}

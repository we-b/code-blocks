# ☓：if文のネストが深くなり、読みにくくなっている
if hogehoge == 2:
    if fugafuga == 3:
        if foo == 4:
            print("Hello, World")

# ◯：条件をまとめてネストを浅くしている
if hogehoge == 2 and fugafuga == 3 and foo == 4:
    print("Hello, World")

# ◎：条件式（三項演算子）を使用している
result = "Hello, World" if hogehoge == 2 and fugafuga == 3 and foo == 4 else None

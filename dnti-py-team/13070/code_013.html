# ☓：GearクラスがWheelに関係するrim、tireの情報を持っている

# 自転車のギアを表すGearクラスが、「車輪のリムの直径、タイヤの厚み」まで責任を持ってしまっている
# Wheelクラスを作り、rim, tireの定義を移すべき

class Gear:
    def __init__(self, chainring, cog, rim, tire):
        self.chainring = chainring
        self.cog = cog
        self.rim = rim
        self.tire = tire
    def ratio(self):
        return self.chainring / self.cog
    def gear_inches(self):
        return self.ratio() * (self.rim + (self.tire * 2))

# 下記のコードはギアに関する情報しか渡していないにも関わらずエラーになる
print(Gear(52, 11).ratio())
# TypeError: __init__() missing 2 required positional arguments: 'rim' and 'tire'


# ◯：Wheelクラスを作り、車輪に関する定義を移す

class Wheel:
    def __init__(self, rim, tire):
        self.rim = rim
        self.tire = tire
    def diameter(self):
        return self.rim + (self.tire * 2)

class Gear:
    def __init__(self, chainring, cog, wheel=None):
        self.chainring = chainring
        self.cog = cog
        self.wheel = wheel
    def ratio(self):
        return self.chainring / self.cog
    def gear_inches(self):
        return self.ratio() * self.wheel.diameter()


wheel = Wheel(26, 1.5)
print(Gear(52, 11, wheel).gear_inches())
# -> 137.09090909090907

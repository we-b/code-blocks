# X：Userモデルに定義されたchange_attributesメソッドが、Tweet, Commentまで変更してしまっている
# models.py
class User(models.Model):
    @classmethod
    def change_attributes(cls, user, tweet, comment):
        user.family_name = "Hoge"
        user.first_name = "Fuga"
        user.save()
        tweet.body = "変更後のツイート"
        tweet.save()
        comment.body = "変更後のコメント"
        comment.save()

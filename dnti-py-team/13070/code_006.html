# X：makeビューはcreateとやっていることが変わらない
def create(request):
    form = UserForm(request.POST)
    if form.is_valid():
        form.save()
        return redirect('user-list')
def make(request):
    form = UserForm(request.POST)
    if form.is_valid():
        form.save()
        return redirect('user-list')

# ◯：Djangoのジェネリックビューを使い、標準的なCRUD操作を実装する
from django.views.generic import CreateView
class UserCreateView(CreateView):
    model = User
    form_class = UserForm
    success_url = reverse_lazy('user-list')

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSheet();
  const params = JSON.parse(e.postData.contents);

  // スプレッドシートの全データを取得
  const data = sheet.getDataRange().getValues();

  // 送られてきた商品名で一致する行を探す
  for (let i = 1; i < data.length; i++) { // 1行目はヘッダーなのでi=1から開始
    if (data[i][1] === params.product) { // 2列目(B列)の商品名と比較
      // 見つかった行の3列目(C列)の在庫数を更新
      sheet.getRange(i + 1, 3).setValue(params.newStock);
      break; // 更新したらループを抜ける
    }
  }

  // Difyに成功したことを伝えるメッセージを返す
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    message: '在庫を更新しました'
  })).setMimeType(ContentService.MimeType.JSON);
}

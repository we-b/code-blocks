function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSheet();
  const data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues();

  const productName = e.parameter.product_name || 'ノートパソコン';
  const result = data.find(row => row[1] === productName);

  if (result) {
    return ContentService.createTextOutput(JSON.stringify({
      product: result[1],
      price: result[3],
      stock: result[2],
      supplier: result[5]
    })).setMimeType(ContentService.MimeType.JSON);
  }

  return ContentService.createTextOutput(JSON.stringify({
    error: '商品が見つかりません'
  })).setMimeType(ContentService.MimeType.JSON);
}

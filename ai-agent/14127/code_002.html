function getProductInfo() {
  const sheet = SpreadsheetApp.getActiveSheet();
  const data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues();

  // 特定の商品を検索(例: ノートパソコン)
  const targetProduct = 'ノートパソコン';
  const result = data.find(row => row[1] === targetProduct);

  if (result) {
    const message =
      `【商品情報】\n\n` +
      `商品名: ${result[1]}\n` +
      `単価: ${result[3]}円\n` +
      `在庫数: ${result[2]}個`;

    Browser.msgBox(message);
  }
}

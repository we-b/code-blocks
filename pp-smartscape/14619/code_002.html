Patch(
    '備品貸し出し申請リスト',
    Defaults('備品貸し出し申請リスト'),
    {
        申請日:Today(),
        申請者メールアドレス:User().Email,
        申請者名: Name.Text,
        部署名: Group.Text,
        備品名: Equipment.Selected.Value,
        個数: Value(Num.Text),
        利用開始日: StartDate.SelectedDate,
        利用終了日: EndDate.SelectedDate, 
        備考: Memo.Text
    }
);
Reset(Name);
Reset(Group);
Reset(Equipment);
Reset(Num);
Reset(StartDate);
Reset(EndDate);
Reset(Memo);
Notify("申請が送信されました！", NotificationType.Success);

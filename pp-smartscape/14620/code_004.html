# リストに保存する処理
Patch(
    'SharePointリスト名',
    Defaults('SharePointリスト名'),
    {
      　リスト上のカラム名: 保存したい値, 
        リスト上のカラム名: 保存したい値, 
        ....

        プルダウンリストで選択した値は、コントロール名.Selected.Value で取得できます
        テキスト入力コントロールに入力された値は、コントロール名.Name.Text で取得できます
    }
);

# コントロールをリセットする処理
Reset(リセットしたいコントロール名);

// Stringは不変オブジェクトのため、連結するたびに新しいオブジェクトが生成されメモリを消費する

    result += name + ", ";
}

// ◯：StringBuilderを使用して連結している
StringBuilder sb = new StringBuilder();
for (String name : names) {
    sb.append(name).append(", ");
}
String result = sb.toString();

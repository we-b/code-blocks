// 注文を表すOrderクラスが「税率の計算、配送料の計算」まで責任を持ってしまっている
// TaxクラスやShippingクラスを作り、それぞれの計算ロジックを移すべき

public class Order {
    private int price;
    private double taxRate;
    private int distance;

    public int getTotalPrice() {
        int tax = (int)(price * taxRate);
        int shippingFee = distance * 10;
        return price + tax + shippingFee;
    }
}

// ◯：TaxクラスとShippingクラスを作り、それぞれの計算ロジックを移す

public class Order {
    private int price;
    private Tax tax;
    private Shipping shipping;

    public int getTotalPrice() {
        return price + tax.calculate(price) + shipping.getFee();
    }
}

// ☓：if文のネストが深くなり、読みにくくなっている
if (hogehoge == 2) {
    if (fugafuga == 3) {
        if (foo == 4) {
            System.out.println("Hello, World");
        }
    }
}

// ◯：条件をまとめてネストを浅くしている
if (hogehoge == 2 && fugafuga == 3 && foo == 4) {
    System.out.println("Hello, World");
}

// ◎：早期リターンを使ってネストをなくしている
if (hogehoge != 2 || fugafuga != 3 || foo != 4) {
    return;
}
System.out.println("Hello, World");

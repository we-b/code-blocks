<!-- ☓：レコードの数だけ同じビューを繰り返す部分で、th:eachのみ使用している -->
<div th:each="post : ${posts}">
    <div class="post-container">
        <p th:text="${post.title}"></p>
        <p th:text="${post.body}"></p>
        <p th:text="${post.user.name}"></p>
    </div>
</div>

<!-- ◯：繰り返しをThymeleafのfragmentで定義している -->
<!-- 部分テンプレートを使用する理由は再利用性が高まり、変更箇所が一箇所にまとまるから -->
<div th:each="post : ${posts}">
    <div th:replace="~{fragments/post :: postCard(post=${post})}"></div>
</div>

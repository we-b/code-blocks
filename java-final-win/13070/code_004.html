// ☓：Controllerにビジネスロジックを直接記述している
// 例）UserControllerにユーザーの集計処理が含まれている

@GetMapping("/users/stats")
public String userStats(Model model) {
    List<User> users = userRepository.findAll();
    long activeCount = users.stream().filter(u -> u.isActive()).count();
    double averageAge = users.stream().mapToInt(User::getAge).average().orElse(0);
    model.addAttribute("activeCount", activeCount);
    model.addAttribute("averageAge", averageAge);
    return "users/stats";
}

// ◯：Serviceクラスにビジネスロジックを定義している
@GetMapping("/users/stats")
public String userStats(Model model) {
    UserStats stats = userService.getUserStats();
    model.addAttribute("stats", stats);
    return "users/stats";
}

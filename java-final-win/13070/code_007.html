// 1つのUserが複数のPostを持っている例
// JOINを指定しないとPostの数だけUserをSELECTするSQLが発行されるため、レコードが増えるほど重くなる

public List<Post> findAll() {
    return postRepository.findAll();
}

// ◯：@QueryアノテーションでJOIN FETCHを指定している
// JOIN FETCHを使用することでSQLの発行回数を1回に抑え、動作を軽くできる

@Query("SELECT p FROM Post p JOIN FETCH p.user")
List<Post> findAllWithUser();

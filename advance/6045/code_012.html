class FileManager
 def initialize(path)
   # path名のファイルを作成。
   @file = File.open(path, "w")
 end

 def write_line(line)
   @file.print(line)
   @file.print("\n")
 end

end

# ↓箇条書きの機能を追加
class BulletWriter
 def initialize(real_writer)
     # ↓いずれのクラスでも、引数をインスタンス変数化しなければならない。
   @real_writer = real_writer
 end

 def write_line(text)
   @real_writer.write_line("- #{text}")
 end
end

bullet_writer = BulletWriter.new(FileManager.new('bullet.txt'))
bullet_writer.write_line('箇条書きの1つ目です。')
bullet_writer.write_line('箇条書きの2つ目です。')
bullet_writer.write_line('箇条書きの3つ目です。')

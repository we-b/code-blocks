# 従業員クラス
class Employee
  attr_reader :name, :salary

  def initialize(name, salary, manager, accountant)
    @name = name
    @salary = salary
    @manager = manager
    # この行が追加されている
    @accountant = accountant
  end

  def salary=(new_salary)
    @salary = new_salary
    @manager.update(self)
    @accountant.update(self)
  end
end

# 上司クラス
class Manager
  def update(changed_employee)
    puts "#{changed_employee.name}さんの給料は#{changed_employee.salary}円です！"
  end
end

# 経理担当クラス
class Accountant
  def update(changed_employee)
    puts "#{changed_employee.name}さんの給料は#{changed_employee.salary}円です!"
  end
end

takashi = Employee.new('takashi', 10000, Manager.new, Accountant.new)
takashi.salary = 9999999

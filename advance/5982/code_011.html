let youtube_ids = ["youtube_sample_id", "youtube_sample_id", "youtube_sample_id"]
let tiktok_ids = ["tiktok_sample_id", "tiktok_sample_id", "tiktok_sample_id"]

let movie_urls = []
let movies = [
 { genre: "youtube", ids: youtube_ids },
 { genre: "tiktok", ids: tiktok_ids }
]

function youtube_url(id){
 return `https://www.youtube.com/embed/${id}`
}

function tiktok_url(id){
 return `https://www.tiktok.com/@${id}`
}

movies.forEach(movie => {
 switch (movie.genre) {
   case "youtube":
     movie.ids.forEach(id => { movie_urls.push(youtube_url(id)) })
   case "tiktok":
     movie.ids.forEach(id => { movie_urls.push(tiktok_url(id)) })
 }
})

console.log(movie_urls)

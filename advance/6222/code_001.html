# 生物クラス
class Creature
  # 生物は全て呼吸を行う
  def breathing
    puts '酸素を吸い二酸化炭素を排出します'
  end
end

# 動物クラス
class Animal < Creature
  # 動物は移動できる
  def move
    puts '自由に移動する'
  end
end

# 植物クラス
class Plant < Creature
  # 植物は光合成ができる
  def photosynthesis
    puts '二酸化炭素を吸い酸素を排出します'
  end
end


require 'forwardable'
# ミドリムシクラス
class Euglena < Creature
  extend Forwardable
  def_delegators :@animal, :move
  def_delegators :@plant, :photosynthesis

  def initialize(animal, plant)
    @animal = animal
    @plant = plant
  end
end

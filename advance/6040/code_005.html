class Nation
 def initialize(country_name,currency)
   puts "#{country_name}の貨幣は#{currency}です。"
 end
end

class Japan < Nation
 def initialize(country_name,currency)
   @country_name = country_name
   @jp_currency = currency
   super(@country_name,@currency)
 end
end

class USA < Nation
  def initialize(country_name,currency)
    @country_name = country_name
    @currency = currency
    super(@country_name,@currency)
  end
end

Japan.new("日本","日本円")
USA.new("アメリカ","米ドル")

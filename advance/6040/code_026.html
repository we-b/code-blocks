# 商品の抽象クラス
class Product
  attr_accessor :name, :price, :weight
​
  def initialize(name: '', price: 0, weight: 0 )
    @name = name
    @price = price
    @weight = weight
  end
  # 商品の固有情報の出力を必須化
  def unique_info
    raise 'Called abstract method: unique_info'
  end
end
​
# 抽象クラスであるProductによって生成するパソコンの具象クラス
class PC < Product
  attr_accessor :parts
​
  def initialize(info, parts)
    super(info)
    @parts = parts
  end
  # 商品の固有情報
  def unique_info
    @spec_sheet = "
    スペック表
    CPU:#{parts.cpu}
    RAM:#{parts.ram}GB
    ROM:#{parts.rom}GB
    "
  end
end
# パソコンのパーツ

    @rom = rom

end

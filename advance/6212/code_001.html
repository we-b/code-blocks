module OrderConfirmationBaseInterfaceTest
  def test_responds_to_transaction
    assert_respond_to(@object, :transaction)
  end
end

class OrderConfirmationTest < Minitest::Test
  include OrderConfirmationBaseInterfaceTest


  def setup
    @food = Minitest::Mock.new
    @input = 0
    @oc = @object = OrderConfirmation.new([@food], 0)
  end

  def test_transaction
    @food.expect(:name, 'テック豚丼')
    @food.expect(:name, 'テック豚丼')
    @food.expect(:price, 440)
    assert_output(stdout =
                   "メニューを選び、投入する金額を入力してください。\n" +
                   "[0] 商品名：テック豚丼 税込み価格：440円\n" +
                   "テック豚丼が選択されました\n") { @oc.transaction }
    @food.verify
  end
end

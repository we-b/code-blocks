class Mail
  def initialize
    @name = "田中 太郎"
    @mail_addr = "sample@test.com"
    @company = "株式会社テックキャンプ "
    @addr = "東京都渋谷区1-1-1"
    @line = "■----------------------------------■"
  end

  def output_mail
    output_greeting
    output_body
    output_info
  end

  private
  def output_greeting
    puts "お世話になっております、#{@name}です。"
  end

  def output_body
    raise 'メール本文を設定してください。'
  end

  def output_info
    puts @line
    puts "#{@name}\n#{@mail_addr}\n#{@company}\n#{@addr}"
    puts @line
  end
end

class DoneMail < Mail
  private
  def output_body
    puts "この度は、ご契約ありがとうございました。"
  end
end

class IntroductionMail < Mail
  private
  def output_body
    puts "この度は、商品のご紹介です。弊社で取り扱っている「〇〇」は多くのお客様にご使用されています。"
    puts "体験会を10/10 10:00より行っておりますので、是非ご参加ください。"
  end
end

doneMail = DoneMail.new
doneMail.output_mail

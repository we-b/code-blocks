require './TechBurger'
require 'minitest/autorun'

module FoodInterfaceTest
  def test_implements_the_name_interface
    assert_respond_to(@object, :name)
  end

  def test_implements_the_price_interface
    assert_respond_to(@object, :price)
  end
end

class TechBurgerTest < Minitest::Test
  include FoodInterfaceTest
  def setup
    @teriyaki = @object = TechBurger.new(name: 'テリヤキバーガーセット', price: 600)
  end

  def test_name
    assert_equal 'テリヤキバーガーセット', @teriyaki.name
  end

  def test_price
    assert_equal 660, @teriyaki.price
  end
end

class TechGyudonTest < Minitest::Test
  include FoodInterfaceTest
  def setup
    @butadon = @object = TechGyudon.new(name: 'テック豚丼', price: 400)
  end

  def test_name
    assert_equal 'テック豚丼', @butadon.name
  end

  def test_price
    assert_equal 440, @butadon.price
  end
end

class OrderConfirmationTest < Minitest::Test
  def setup
    @food = Minitest::Mock.new
    @sales = Minitest::Mock.new
    @input = 0
  end

  def test_transaction
    oc = OrderConfirmation.new([@food], 0, @sales)
    @food.expect(:name, 'テック豚丼')
    @food.expect(:name, 'テック豚丼')
    @food.expect(:price, 440)
    @sales.expect(:update, 440, [@food])
    assert_output(stdout =
                   "メニューを選び、投入する金額を入力してください。\n" +
                   "[0] 商品名：テック豚丼 税込み価格：440円\n" +
                   "テック豚丼が選択されました\n") { oc.transaction }
    @sales.verify
    @food.verify
  end
end

class SearchService
 class << self
   def search(ele)
     # ele = {
     #   obj: 検索対象のモデル,
     #   keyword: 検索ワード,
     #   keyword_column: ワード検索する対象のカラム,
     #   id: 検索id,
     #   id_column: id検索する対象のカラム
     # }
     if !ele[:keyword].empty? && !ele[:id].zero?
       integration_search(ele)
     elsif !ele[:keyword].empty?
       keyword_search(ele)
     elsif !ele[:id].zero?
       id_search(ele)
     else
       ele[:obj].all
     end
   end

   private

   def integration_search(ele)
     ele[:obj].where("#{ele[:keyword_column]} LIKE(?)", "%#{ele[:keyword]}%").where("#{ele[:id_column]} = ?", ele[:id])
   end

   def keyword_search(ele)
     ele[:obj].where("#{ele[:keyword_column]} LIKE(?)", "%#{ele[:keyword]}%")
   end

   def id_search(ele)
     ele[:obj].where("#{ele[:id_column]} = ?", ele[:id])
   end
 end
end

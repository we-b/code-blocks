class Tweet < ApplicationRecord
 validates :text, presence: true
 belongs_to :user
 belongs_to :genre
 has_many :comments
 def self.search(search, genre_id)
   if search != '' && genre_id != '0'
     Tweet.where('text LIKE(?)', "%#{search}%").where(genre_id: genre_id)
   elsif search != ''
     Tweet.where('text LIKE(?)', "%#{search}%")
   elsif genre_id != '0'
     Tweet.where(genre_id: genre_id)
   else
     Tweet.all
   end
 end
end

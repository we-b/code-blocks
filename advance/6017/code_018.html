# 抽象クラスHuman
class Human
# 抽象メソッドにraiseで例外（=エラー）を定義
 def talk
   raise "Called abstract method: talk"
 end

 def walk
   raise "Called abstract method: walk"
 end

 def run
   raise "Called abstract method: run"
 end
end


# 具象クラスTeacherは、抽象クラスHumanを継承
class Japanese < Human
  # ↓抽象メソッドと同名のメソッドを定義する。
   def talk
     puts "話します"
   end

   def walk
     puts "歩きます"
   end

   def run
     puts "走ります"
   end


mr_tanaka = Japanese.new
mr_tanaka.talk

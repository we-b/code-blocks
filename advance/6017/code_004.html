attr_reader :name, :gender, :vitality, :power, :defense

  def initialize(args)
    @name = args[:name] || nil
    @gender = args[:gender] || nil
    @vitality = args[:vitality].to_i || nil
    @power = args[:power].to_i || nil
    @defense = args[:defense].to_i || nil
  end

  def attack
    puts "物理攻撃！相手に#{power}HPのダメージ！！"
  end

  def defense
    puts "身を守る！#{defense}HPのダメージを軽減する！！"
  end

class Wizard
  attr_reader :name, :gender, :vitality, :power, :defense

  def initialize(args)
    @name = args[:name]
    @gender = args[:gender]
    @vitality = args[:vitality]
    @power = args[:power]
    @defense = args[:defense] 
  end

  def attack
    puts "物理攻撃！相手に#{power}HPのダメージ！！"
  end

  def defense
    puts "身を守る！#{defense}HPのダメージを軽減する！！"
  end

  def run_attack_magic
    puts "こうげきじゅもんをとなえた！ 相手に20HPダメージ。"
  end
end

require './TechBurger'
require 'minitest/autorun'

class TechBurgerTest < Minitest::Test
  def setup
    @teriyaki = TechBurger.new(name: 'テリヤキバーガーセット', price: 600)
  end

  def test_name
    assert_equal 'テリヤキバーガーセット', @teriyaki.name
  end

  def test_price
    assert_equal 660, @teriyaki.price
  end
end

class OrderConfirmationTest < Minitest::Test
  def setup
    foods = [
      { name: 'テリヤキバーガーセット', price: 600 },
      { name: 'ビッグテックバーガーセット', price: 800 },
      { name: 'テックキッズセット', price: 380 }
    ]
    @foods = foods.map { |f| TechBurger.new({ name: f[:name], price: f[:price] }) }
    @sales = Minitest::Mock.new
    @input = rand(1)
  end

  def test_transaction
    oc = OrderConfirmation.new(@foods, @sales, @input)
    @sales.expect(:update, 660, [@foods[@input]])
    oc.transaction
    @sales.verify
  end
end

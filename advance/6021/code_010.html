class User
 def initialize(info)
   @name = info[:name]
 end

 def name
   @name
 end
end

class Tweet
 def initialize(contents)
   @poster_name = contents[:name]
   @text = contents[:text]
   @image = contents[:image]
 end

 def poster_name
   @poster_name
 end

 def text
   @text
 end

 def image
   @image
 end
end

class TweetForm
 def initialize
   @contents = { name: '', text: '', image: '' }
 end

 def input(user, data)
   contents[:name] = user.name
   contents[:text] = data[:text]
   contents[:image] = data[:image]
 end

 def create
   Tweet.new(contents)
 end

 def contents
   @contents
 end
end

tanaka = User.new({ name: 'tanaka' })

form = TweetForm.new
form.input(tanaka, { text: 'テスト投稿', image: 'テスト画像のURL' })
first_tweet = form.create

puts first_tweet.poster_name
puts first_tweet.text
puts first_tweet.image

class ConcreteDecoratorA < Decorator
  def do_something
    component.str = 'デコレータパターンを適用した例です'
  end
end

class ConcreteDecoratorB < Decorator
  def do_something
    component.str = "こちらもまたデコレータパターンを適用させた例です\n"
    component.str += 'コンポジションにより実現します。同様のメソッド(do_something)ですが各デコレータに応じて結果が変わっています'
  end
end

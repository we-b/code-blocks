  ---
  Step 2: Service クラスの基本設計

  責務の明確化：

  | 責務           | 担当           |
  |--------------|--------------|
  | HTTPリクエストの受付 | Controller   |
  | パラメータの変換・正規化 | Service      |
  | バリデーション実行    | Model（既存のまま） |
  | データベースへの保存   | Service      |
  | 成功/失敗の判定     | Service      |
  | レスポンスの生成     | Controller   |

  ---
  Step 3: Service クラスのインターフェース設計

  # 使用イメージ
  service = Tasks::CreateService.new(params)
  result = service.call

  result.success?  # => true / false
  result.task      # => Task インスタンス
  result.errors    # => エラーメッセージ（失敗時）

  設計ポイント：
  - call メソッドで実行（Rubyの慣習）
  - 結果オブジェクトを返す（成功/失敗を明示）
  - 例外ではなく戻り値で制御

  Step 5: 各ファイルの実装内容

  5-1. app/services/service_result.rb
  class ServiceResult
    attr_reader :task, :errors

    def initialize(success:, task: nil, errors: [])
      @success = success
      @task = task
      @errors = errors
    end

    def success?
      @success
    end

    def failure?
      !@success
    end
  end

  5-2. app/services/tasks/create_service.rb
  class Tasks::CreateService
    def initialize(params)
      @params = params
    end

    def call
      task = Task.new(task_attributes)

      if task.save
        ServiceResult.new(success: true, task: task)
      else
        ServiceResult.new(success: false, task: task, errors: task.errors.full_messages)
      end
    end

    private

    def task_attributes
      {
        name: @params[:name],
        explanation: @params[:explanation],
        status: @params[:status],
        priority: @params[:priority],
        genre_id: @params[:genreId],
        deadline_date: @params[:deadlineDate]
      }
    end
  end

  5-3. app/controllers/tasks_controller.rb（修正後）
  def create
    result = Tasks::CreateService.new(params).call
    @result = result.task
    tasks_all
  end

  Step 6: テストの作成

  spec/services/tasks/create_service_spec.rb
  RSpec.describe Tasks::CreateService do
    describe '#call' do
      context '有効なパラメータの場合' do
        it 'タスクを作成し、成功を返す' do
          params = { name: 'テスト', status: 'todo', genreId: 1 }
          result = described_class.new(params).call

          expect(result.success?).to be true
          expect(result.task).to be_persisted
        end
      end

      context '無効なパラメータの場合' do
        it '失敗を返し、エラーメッセージを含む' do
          params = { name: '' }  # nameが必須の場合
          result = described_class.new(params).call

          expect(result.failure?).to be true
          expect(result.errors).not_to be_empty
        end
      end
    end
  end

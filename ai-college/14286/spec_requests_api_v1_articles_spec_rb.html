require 'rails_helper'

RSpec.describe 'Api::V1::Articles', type: :request do
  describe 'GET /api/v1/articles' do
    context '記事が存在する場合' do
      before do
        create_list(:article, 3, :published)
      end

      it '記事一覧を取得できること' do
        get '/api/v1/articles'

        expect(response).to have_http_status(:ok)
        json = JSON.parse(response.body)
        expect(json.length).to eq(3)
      end
    end

    context '記事が存在しない場合' do
      it '空の配列を返すこと' do
        get '/api/v1/articles'

        expect(response).to have_http_status(:ok)
        json = JSON.parse(response.body)
        expect(json).to eq([])
      end
    end
  end

  describe 'GET /api/v1/articles/:id' do
    context '記事が存在する場合' do
      let(:article) { create(:article) }

      it '記事の詳細を取得できること' do
        get "/api/v1/articles/#{article.id}"

        expect(response).to have_http_status(:ok)
        json = JSON.parse(response.body)
        expect(json['id']).to eq(article.id)
        expect(json['title']).to eq(article.title)
        expect(json['body']).to eq(article.body)
      end
    end

    context '記事が存在しない場合' do
      it '404を返すこと' do
        get '/api/v1/articles/999999'

        expect(response).to have_http_status(:not_found)
      end
    end
  end

  describe 'POST /api/v1/articles' do
    context '正しいパラメータの場合' do
      let(:valid_params) do
        {
          article: {
            title: 'テスト記事',
            body: 'これはテスト記事の本文です。'
          }
        }
      end

      it '記事を作成できること' do
        expect {
          post '/api/v1/articles', params: valid_params
        }.to change(Article, :count).by(1)

        expect(response).to have_http_status(:created)
        json = JSON.parse(response.body)
        expect(json['title']).to eq('テスト記事')
      end
    end

    context '不正なパラメータの場合' do
      let(:invalid_params) do
        {
          article: {
            title: '',
            body: ''
          }
        }
      end

      it '422を返すこと' do
        post '/api/v1/articles', params: invalid_params

        expect(response).to have_http_status(:unprocessable_entity)
        json = JSON.parse(response.body)
        expect(json['errors']).to be_present
      end
    end
  end

  describe 'PATCH /api/v1/articles/:id' do
    let(:article) { create(:article, title: '元のタイトル') }

    context '正しいパラメータの場合' do
      let(:update_params) do
        {
          article: {
            title: '更新後のタイトル'
          }
        }
      end

      it '記事を更新できること' do
        patch "/api/v1/articles/#{article.id}", params: update_params

        expect(response).to have_http_status(:ok)
        json = JSON.parse(response.body)
        expect(json['title']).to eq('更新後のタイトル')

        article.reload
        expect(article.title).to eq('更新後のタイトル')
      end
    end

    context '記事が存在しない場合' do
      it '404を返すこと' do
        patch '/api/v1/articles/999999', params: { article: { title: 'test' } }

        expect(response).to have_http_status(:not_found)
      end
    end
  end

  describe 'DELETE /api/v1/articles/:id' do
    context '記事が存在する場合' do
      let!(:article) { create(:article) }

      it '記事を削除できること' do
        expect {
          delete "/api/v1/articles/#{article.id}"
        }.to change(Article, :count).by(-1)

        expect(response).to have_http_status(:no_content)
      end
    end

    context '記事が存在しない場合' do
      it '404を返すこと' do
        delete '/api/v1/articles/999999'

        expect(response).to have_http_status(:not_found)
      end
    end
  end
end

require 'rails_helper'

RSpec.describe Comment, type: :model do
  describe 'バリデーション' do
    let(:article) { create(:article) }

    context 'authorが空の場合' do
      it 'バリデーションに失敗すること' do
        comment = build(:comment, article: article, author: nil)
        expect(comment).not_to be_valid
        expect(comment.errors[:author]).to include("can't be blank")
      end
    end

    context 'bodyが空の場合' do
      it 'バリデーションに失敗すること' do
        comment = build(:comment, article: article, body: nil)
        expect(comment).not_to be_valid
        expect(comment.errors[:body]).to include("can't be blank")
      end
    end

    context 'articleが存在しない場合' do
      it 'バリデーションに失敗すること' do
        comment = build(:comment, article: nil)
        expect(comment).not_to be_valid
        expect(comment.errors[:article]).to include("must exist")
      end
    end

    context '全ての属性が正しく設定されている場合' do
      it 'バリデーションに成功すること' do
        comment = build(:comment, article: article)
        expect(comment).to be_valid
      end
    end
  end

  describe 'アソシエーション' do
    describe 'belongs_to :article' do
      it '記事に属していること' do
        article = create(:article)
        comment = create(:comment, article: article)
        expect(comment.article).to eq(article)
      end
    end

    describe 'dependent: :destroy' do
      it '親の記事が削除されるとコメントも削除されること' do
        article = create(:article)
        comment = create(:comment, article: article)

        expect {
          article.destroy
        }.to change(Comment, :count).by(-1)
      end
    end
  end
end

competitor-analysis/.claude/skills/competitor-analysis/scripts/generate-image.py を作成してください。

仕様:
- コマンドライン引数: -p "プロンプト" -o "出力パス"
- 環境変数 GOOGLE_CLOUD_PROJECT / GOOGLE_CLOUD_LOCATION を読み取り、Vertex AI 経由で認証する（APIキー文字列は使わない）
- google-genai ライブラリを使用し、genai.Client(vertexai=True, project=..., location=...) で初期化
- モデル: gemini-3-pro-image-preview
- generate_content の config に response_modalities=["TEXT", "IMAGE"] を設定
- レスポンスのpartsからimage部分を取得してファイルに保存
- Pillowを使って画像を保存
- エラー時は分かりやすいメッセージを表示して終了

参考コード:
#!/usr/bin/env python3
"""
画像生成スクリプト
Vertex AI（Gemini画像生成API）を使用して画像を生成する

使用例:
    python generate-image.py -p "競合分析のインフォグラフィック" -o "data/output/analysis.png"
"""

import argparse
import os
import sys

from PIL import Image
from google import genai
from google.genai import types


def generate_image(prompt: str, output_path: str) -> str:
    """
    Gemini APIを使用して画像を生成する

    Args:
        prompt: 画像の説明
        output_path: 出力ファイルパス

    Returns:
        生成された画像のファイルパス
    """
    project = os.environ.get("GOOGLE_CLOUD_PROJECT")
    location = os.environ.get("GOOGLE_CLOUD_LOCATION", "global")
    if not project:
        print("エラー: 環境変数 GOOGLE_CLOUD_PROJECT が設定されていません。")
        sys.exit(1)

    # APIキー文字列ではなく、gcloud認証（ADC）＋プロジェクトIDで Vertex AI に接続する
    client = genai.Client(vertexai=True, project=project, location=location)

    models = ["gemini-3-pro-image-preview", "gemini-3.1-flash-image-preview"]

    for model in models:
        try:
            print(f"モデル {model} で生成を試みます...")
            response = client.models.generate_content(
                model=model,
                contents=prompt,
                config=types.GenerateContentConfig(
                    response_modalities=["TEXT", "IMAGE"],
                ),
            )

            for part in response.parts:
                if part.inline_data is not None:
                    image = part.as_image()
                    os.makedirs(os.path.dirname(os.path.abspath(output_path)), exist_ok=True)
                    image.save(output_path)
                    print(f"画像を保存しました: {output_path}")
                    return output_path

            print(f"エラー: {model} のレスポンスに画像が含まれていませんでした。")

        except Exception as e:
            print(f"{model} での生成に失敗しました: {e}")
            if model != models[-1]:
                print("次のモデルにフォールバックします...")

    print("エラー: すべてのモデルで画像生成に失敗しました。")
    sys.exit(1)


def main():
    parser = argparse.ArgumentParser(description="Gemini APIを使用して画像を生成する")
    parser.add_argument(
        "-p", "--prompt",
        type=str,
        required=True,
        help="画像の説明（日本語可）",
    )
    parser.add_argument(
        "-o", "--output",
        type=str,
        required=True,
        help="出力ファイルパス（例: output/image.png）",
    )

    args = parser.parse_args()

    generate_image(prompt=args.prompt, output_path=args.output)


if __name__ == "__main__":
    main()

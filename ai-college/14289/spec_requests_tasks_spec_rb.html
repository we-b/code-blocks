require 'rails_helper'

RSpec.describe "Tasks API", type: :request do
  let!(:genre) { Genre.create!(name: 'テストジャンル') }

  describe "POST /tasks" do
    context 'priorityパラメータを指定してタスクを作成' do
      it '優先度「高」でタスクを作成できる' do
        task_params = {
          name: '重要なタスク',
          explanation: 'これは重要なタスクです',
          genreId: genre.id,
          deadlineDate: '2025-12-31',
          status: 0,
          priority: 'high'
        }

        post '/tasks', params: task_params, as: :json

        expect(response).to have_http_status(:success)

        created_task = Task.last
        expect(created_task.name).to eq('重要なタスク')
        expect(created_task.priority).to eq('high')
        expect(created_task.high?).to be true
      end

      it '優先度「中」でタスクを作成できる' do
        task_params = {
          name: '通常のタスク',
          explanation: 'これは通常のタスクです',
          genreId: genre.id,
          deadlineDate: '2025-12-31',
          status: 0,
          priority: 'medium'
        }

        post '/tasks', params: task_params, as: :json

        expect(response).to have_http_status(:success)

        created_task = Task.last
        expect(created_task.name).to eq('通常のタスク')
        expect(created_task.priority).to eq('medium')
        expect(created_task.medium?).to be true
      end

      it '優先度「低」でタスクを作成できる' do
        task_params = {
          name: '急がないタスク',
          explanation: 'これは急がないタスクです',
          genreId: genre.id,
          deadlineDate: '2025-12-31',
          status: 0,
          priority: 'low'
        }

        post '/tasks', params: task_params, as: :json

        expect(response).to have_http_status(:success)

        created_task = Task.last
        expect(created_task.name).to eq('急がないタスク')
        expect(created_task.priority).to eq('low')
        expect(created_task.low?).to be true
      end
    end

    context 'レスポンスJSONにpriorityが含まれている' do
      it 'タスク作成後のレスポンスにpriorityフィールドが含まれる' do
        task_params = {
          name: 'APIテストタスク',
          explanation: 'レスポンス確認用',
          genreId: genre.id,
          deadlineDate: '2025-12-31',
          status: 0,
          priority: 'high'
        }

        post '/tasks', params: task_params, as: :json

        expect(response).to have_http_status(:success)

        json_response = JSON.parse(response.body)
        expect(json_response).to be_an(Array)

        created_task = json_response.find { |task| task['name'] == 'APIテストタスク' }
        expect(created_task).not_to be_nil
        expect(created_task['priority']).to eq('high')
      end

      it 'priorityを指定しない場合、デフォルト値「medium」が設定される' do
        task_params = {
          name: 'デフォルト優先度タスク',
          explanation: 'デフォルト値確認用',
          genreId: genre.id,
          deadlineDate: '2025-12-31',
          status: 0
        }

        post '/tasks', params: task_params, as: :json

        expect(response).to have_http_status(:success)

        json_response = JSON.parse(response.body)
        created_task = json_response.find { |task| task['name'] == 'デフォルト優先度タスク' }

        expect(created_task).not_to be_nil
        expect(created_task['priority']).to eq('medium')
      end
    end
  end
end

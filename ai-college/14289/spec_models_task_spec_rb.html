require 'rails_helper'

RSpec.describe Task, type: :model do
  # テストデータを事前に作成
  let(:genre) { Genre.create!(name: 'テストジャンル') }

  describe 'バリデーション' do
    context '優先度が正常な場合' do
      it '優先度を「低」に設定できること' do
        task = Task.create(name: 'テストタスク', genre: genre, priority: :low)
        expect(task).to be_valid
        expect(task.priority).to eq('low')
      end

      it '優先度を「中」に設定できること' do
        task = Task.create(name: 'テストタスク', genre: genre, priority: :medium)
        expect(task).to be_valid
        expect(task.priority).to eq('medium')
      end

      it '優先度を「高」に設定できること' do
        task = Task.create(name: 'テストタスク', genre: genre, priority: :high)
        expect(task).to be_valid
        expect(task.priority).to eq('high')
      end
    end
  end

  describe 'デフォルト値' do
    context '新規タスク作成時' do
      it '優先度を指定しない場合、デフォルト値が「中」になること' do
        task = Task.create(name: 'テストタスク', genre: genre)
        expect(task.priority).to eq('medium')
      end
    end
  end
end

const DISCOUNT_RATES = {
  PREMIUM_LOYAL: 0.3,
  PREMIUM_REGULAR: 0.2,
  PREMIUM_NEW: 0.1,
  FREQUENT_BUYER: 0.05,
  DEFAULT: 0
};

function getDiscount(user) {
  if (user.isPremium && user.purchaseCount > 10) {
    return user.membershipYears > 5
      ? DISCOUNT_RATES.PREMIUM_LOYAL
      : DISCOUNT_RATES.PREMIUM_REGULAR;
  }
  if (user.isPremium) return DISCOUNT_RATES.PREMIUM_NEW;
  if (user.purchaseCount > 20) return DISCOUNT_RATES.FREQUENT_BUYER;
  return DISCOUNT_RATES.DEFAULT;
}

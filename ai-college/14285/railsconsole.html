# 1. 記事を作成
article = Article.create!(title: 'テスト記事', body: 'テスト本文', status: :draft)

# 2. コメントを作成
comment1 = article.comments.create!(author_name: 'ユーザー1', body: 'コメント1')
comment2 = article.comments.create!(author_name: 'ユーザー2', body: 'コメント2')

# 3. 記事からコメントを取得
article.comments
# => [#<Comment id: 1, ...>, #<Comment id: 2, ...>]

# 4. コメントから記事を取得
comment1.article
# => #<Article id: 1, ...>

# 5. 記事を削除
article.destroy
Comment.count
# => 0（コメントも削除される）

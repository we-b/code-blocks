class BankAccount {
    private double balance; // 直接アクセスできないようにする

    // コンストラクタで初期残高を設定
    public BankAccount(double initialBalance) {
        if (initialBalance < 0) {
            throw new IllegalArgumentException("初期残高は0以上である必要があります");
        }
        this.balance = initialBalance;
    }

    // ゲッターメソッド（残高を取得する）
    public double getBalance() {
        return balance;
    }

    // 預金メソッド
    public void deposit(double amount) {
        if (amount <= 0) {
            throw new IllegalArgumentException("預金額は正の値である必要があります");
        }
        balance += amount;
    }

    // 引き出しメソッド
    public void withdraw(double amount) {
        if (amount <= 0) {
            throw new IllegalArgumentException("引き出し額は正の値である必要があります");
        }
        if (balance >= amount) {
            balance -= amount;
        } else {
            throw new IllegalArgumentException("残高不足です");
        }
    }
}

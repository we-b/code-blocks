import java.util.regex.Pattern;

public class PostalCodeValidation {
    public static void main(String[] args) {
        String postalCode = "150-0044";

        // 正規表現のパターン
        String regex = "^\\d{3}-\\d{4}$";
        boolean isValid = Pattern.matches(regex, postalCode);

        if (isValid) {
            System.out.println("郵便番号の形式が正しいです: " + postalCode);
        } else {
            System.out.println("郵便番号の形式が正しくありません: " + postalCode);
        }
    }
}

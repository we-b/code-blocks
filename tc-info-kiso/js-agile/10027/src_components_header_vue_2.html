<script setup>
import FormModal from './FormModal.vue'
import { ref } from 'vue'

const showModal = ref(false);
const body = ref('');

const openModal = (component) => {
  showModal.value = !showModal.value;
  body.value = component
}
</script>

<template>
  <div class="header-list">
    <button class="add-list" @click="openModal('goods')">リスト追加</button>
    <button @click="openModal('genre')">ジャンル追加</button>
    <Modal v-model="showModal">
      <FormModal :isOpen="showModal" :body="body" @close-modal="showModal = false"/>
    </Modal>
  </div>
</template>

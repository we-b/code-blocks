// ☓：if文のネストが深くなり、読みにくくなっている
if (hogehoge === 2) {
  if (fugafuga === 3) {
    if (foo === 4) {
      console.log("Hello, World");
    }
  }
}

// ◯：条件をまとめてネストを浅くしている
if (hogehoge === 2 && fugafuga === 3 && foo === 4) {
  console.log("Hello, World");
}

// ◎：1行で書けるものは1行にまとめている
if (hogehoge === 2 && fugafuga === 3 && foo === 4) console.log("Hello, World");

// ◎：三項演算子を使用している
hogehoge === 2 && fugafuga === 3 && foo === 4 ? console.log("Hello, World") : false;

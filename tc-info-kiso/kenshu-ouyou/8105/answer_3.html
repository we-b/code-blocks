it 'ログインに失敗し、再びサインインページに戻ってくる' do
  @user = FactoryBot.create(:user)
  visit root_path
  expect(page).to have_current_path(new_user_session_path)
  fill_in 'user_email', with: 'test'
  fill_in 'user_password', with: 'test'
  click_on('Log in')
  expect(page).to have_current_path(new_user_session_path)
end

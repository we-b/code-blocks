import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class RollerCoaster {
  public static void main(String[] args) {
    int rideCount = 0;
    List<Map<String, String>> friends = new ArrayList<>();

    try (Scanner scanner = new Scanner(System.in)){

      Map<String, String> friend1 = new HashMap<>();
      System.out.println("お友達の名前は？");
      friend1.put("name", scanner.nextLine());
      System.out.println("お友達の身長は？");
      friend1.put("height", scanner.nextLine());
      friends.add(friend1);
      rideCount += 1;
  
      Map<String, String> friend2 = new HashMap<>();
      System.out.println("お友達の名前は？");
      friend2.put("name", scanner.nextLine());
      System.out.println("お友達の身長は？");
      friend2.put("height", scanner.nextLine());
      friends.add(friend2);
      rideCount += 1;
  
      Map<String, String> friend3 = new HashMap<>();
      System.out.println("お友達の名前は？");
      friend3.put("name", scanner.nextLine());
      System.out.println("お友達の身長は？");
      friend3.put("height", scanner.nextLine());
      friends.add(friend3);
      rideCount += 1;
  
      System.out.println("乗車するのは" + rideCount + "人です");
  
    }
  }
}

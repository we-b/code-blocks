class PC
  attr_reader :data, :config
  def initialize(data,config)
    @data = data
    @config = config
  end
end

class Printer
  def initialize(pc,num)
    @data = pc.data
    @config = pc.config
    @num = num
  end

  def print
    # Aクラスのhoge,fugaが無ければ以下の処理を実行できない。
    puts "#{@data}を#{@config}で#{@num}部印刷しました。"
  end
end

pc = PC.new("会議資料", "モノクロ")
printer = Printer.new(pc,100)
puts printer.print
# => “会議資料をモノクロで100部印刷しました。”

let movie_urls = []
let movies = [{ genre: "youtube", ids: ["youtube_sample_id", "youtube_sample_id", "youtube_sample_id"] },{ genre: "tiktok", ids: ["tiktok_sample_id", "tiktok_sample_id", "tiktok_sample_id"] }]

movies.forEach(movie => {
 if (movie.genre == "youtube") {
   movie.ids.forEach(id => { movie_urls.push(`https://www.youtube.com/embed/${id}`) })
 } else if (movie.genre == "tiktok") {
   movie.ids.forEach(id => { movie_urls.push(`https://www.tiktok.com/@${id}`) })
 }
})

console.log(movie_urls)

class TweetsController < ApplicationController

  def index
    @tweets = Tweet.all
  end

  def new
    @tweet = Tweet.new
  end

  def create
    Tweet.create(tweet_params)
  end

  def destroy
    tweet = Tweet.find(params[:id])
    tweet.destroy
  end

  def edit
    @tweet = Tweet.find(params[:id])
  end

  def update
    tweet = Tweet.find(params[:id])
    tweet.update(tweet_params)
  end

  def show
    @tweet = Tweet.find(params[:id])
  end

  private

  def tweet_params
    params.require(:tweet).permit(:name, :image, :text)
  end
end

上記のコードでは、以下の部分が重複しており、修正する際にすべて修正しなければいけません。
`@tweet = Tweet.find(params[:id])`が重複している箇所は以下です。

def edit
  @tweet = Tweet.find(params[:id])
end

def show
  @tweet = Tweet.find(params[:id])
end

class Item
  attr_reader :name, :price
  def initialize(name, price)
    @name = name
    @price = price
  end
end

class Customer
  attr_reader :purchase_items

  def initialize(name)
    @name = name
    @purchase_items = []
  end

  def buy(pay, item)
    return puts 'お金が足りません' if pay < item.price

    puts "#{@name}が#{item.name}を購入しました"
    purchase_items << item
  end

  def inquiry_purchase_items
    puts "#{@name}の購入商品一覧"
    purchase_items.each do |purchase_item|
      puts "商品名:#{purchase_item.name}"
    end
  end
end

items = [
  Item.new('おもちゃ', 1000),
  Item.new('なりきりセット', 10000)
]
customer = Customer.new('たかし')
customer.buy(2000, items[0])
customer.buy(1000, items[1])
customer.inquiry_purchase_items

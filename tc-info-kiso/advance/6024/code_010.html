class CPU
  attr_reader :name, :gender, :vitality

  def initialize(args)
    @name = args[:name]
    @gender = args[:gender]
    @vitality = args[:vitality]
    get_occupy(args)
  end

  def attack
    raise "Called abstract method: attack"
  end

  def defense
    raise "Called abstract method: defense"
  end

  def get_occupy
    nil
  end
end

class Player < CPU
  attr_reader :occupy

  def get_occupy(args)
    @occupy = args[:occupy]
  end

  def run_skill
    occupy.run_skill
  end

  def attack
    puts '物理攻撃！相手に10HPのダメージ！！'
  end

  def defense
    puts '身を守る！10HPのダメージを軽減する！！'
  end
end

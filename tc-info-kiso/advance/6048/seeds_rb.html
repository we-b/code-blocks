require 'securerandom'

 # ランダムのアルファベットを生成
 def random_string
   ('A'..'Z').to_a.sample(5).join
 end

 # ランダムのhex15桁を生成
 def password
   SecureRandom.hex(15)
 end

 # ランダムのアドレスを生成
 def mail_address
   "#{SecureRandom.hex(10)}@sample.com"
 end


10.times do
   params = { nickname: random_string, password: password, email: mail_address }
   User.create(params)
end

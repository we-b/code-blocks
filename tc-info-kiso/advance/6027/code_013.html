class SendMail
  attr_reader :mail
  def initialize(mail)
    @mail = mail
  end


  def write_mail
    mail.output_mail
  end
 end

class Mail
  def initialize(name, mail_addr, company)
    @name = name
    @mail_addr = mail_addr
    @company = company
    @line = "■----------------------------------■"
  end

 def output_mail
   puts "お世話になっております、#{@name}です。"
   puts "この度は、ご契約ありがとうございました。"
   puts @line
   puts "#{@name}\n#{@mail_addr}\n#{@company}"
   puts @line
 end
end

mail = Mail.new("樋口", "test@example.ne.jp", "株式会社XXXカンパニー")
SendMail.new(mail).write_mail

class Mail
  def initialize
    @name = "田中 太郎"
    @mail_addr = "test@example.ne.jp"
    @line = "■----------------------------------■"
  end

 def output_mail_content
   puts "お世話になっております、#{@name}です。"
   puts "この度は、ご契約ありがとうございました。"
   puts @line
   puts "#{@name}\n#{@mail_addr}"
   puts @line
 end
end

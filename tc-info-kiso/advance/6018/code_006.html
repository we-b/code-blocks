# 車に共通する抽象概念が定義されているCarクラス
class Car
　︙
end

# Carクラスを継承したPriusクラス
class Prius < Car

 # カーナビのモジュールを追加
 extend CarNavigation

end

Prius.navigate
# CarNavigationモジュールの振る舞いをクラスメソッドとして使用できるようになる

prius = Prius.new
prius.navigate
# undefined method `navigate' for ・・・ というエラーになる

prius2 = Prius.new
prius2.extend CarNavigation
prius2.navigate
# エラーにならない。モジュールの振る舞いをメソッドとして使用できるようになる

class Club
  attr_accessor :name

  def introduction(goal)
    puts "#{self.name}です。#{goal}を目指しています。部員募集中です!"
  end
end

class BaseballClub < Club
  def introduction(goal)
    super
  end
end

class BrassBandClub < Club
  def introduction(goal)
    super
  end
end


baseball_club = BaseballClub.new
baseball_club.name = '〇〇高校 硬式野球部'

brass_band_club = BrassBandClub.new
brass_band_club.name = '吹奏楽サークル'

baseball_club.introduction('甲子園出場')
#=> 〇〇高校 硬式野球部です。甲子園出場を目指しています。部員募集中です!

brass_band_club.introduction('県大会 金賞')
#=> 吹奏楽サークルです。県大会 金賞を目指しています。部員募集中です!

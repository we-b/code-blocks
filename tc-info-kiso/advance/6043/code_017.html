class Mail
  attr_reader :name, :mail_addr, :company, :addr, :line
  attr_accessor :elements
  # 中略
  def output_mail
    puts "お世話になっております、#{name}です。"
   construct_sentence
   puts line
   puts "#{name}\n#{mail_addr}\n#{company}\n#{addr}"
   puts line
  end

 private

 def construct_sentence
   elements[:main].output_contents
   elements[:sub].output_contents if elements.key?(:sub)
 end

end

class Agreement
  def output_contents
    puts ‘この度は、ご契約ありがとうございました。’
  end
end

class IntroductionProduct

end

class InfoCampaign
 def initialize(term)
   @term = term
 end
 def output_contents
   puts ‘\nまた、弊社では1月4日より「新年キャンペーン」を開催いたします。’
   puts ‘この機会に、ぜひお買い求めください。’
   puts ‘\nキャンペーンの詳細’
   puts "開催期間：#{@term}"
   puts ‘対象商品：〇〇 30％OFF(他のキャンペーンとは併用不可)’
 end
end

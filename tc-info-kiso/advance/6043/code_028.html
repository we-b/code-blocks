class Mail
  attr_reader :name, :mail_addr, :company, :addr, :line
  attr_accessor :elements
  def initialize(elements)
    @name = '田中 太郎'
    @mail_addr = 'sample@test.com'
    @company = '株式会社テックキャンプ '
    @addr = '東京都渋谷区1-1-1'
    @line = '■----------------------------------■'
    @elements = elements
  end
  def output_mail
    puts "お世話になっております、#{name}です。"
    construct_sentence
    puts line
    puts "#{name}\n#{mail_addr}\n#{company}\n#{addr}"
    puts line
  end

  private
  
  def construct_sentence
    elements.each do |element|
      element.output_contents
    end
  end
end

class Agreement
  def output_contents
    puts 'この度は、ご契約ありがとうございました。'
  end
end

class IntroductionProduct
  def output_contents
    puts 'この度は、商品のご紹介です。弊社で取り扱っている「〇〇」は多くのお客様にご使用されています。'
    puts '体験会を10/10 10:00より行っておりますので、是非ご参加ください。'
  end
end

class InfoVacation
  def output_contents
    puts "\n\n年末年始の休暇のお知らせ"
    puts '2020年12月29日 ～ 2021年1月3日'
    puts '新年の営業は1月4日午前9時からとなります。'
  end
end

class InfoCampaign
  def initialize(term)
    @term = term
  end
  def output_contents
    puts "\nまた、弊社では1月4日より「新年キャンペーン」を開催いたします。"
    puts 'この機会に、ぜひお買い求めください。'
    puts "\nキャンペーンの詳細"
    puts "開催期間：#{@term}"
    puts '対象商品：〇〇 30％OFF(他のキャンペーンとは併用不可)'
  end
end

term = '1月4日(月曜日) 〜 1月8日(金曜日)'
elements = [ Agreement.new, InfoCampaign.new(term),InfoVacation.new ]

mail = Mail.new(elements)
mail.output_mail
mail.elements = [ IntroductionProduct.new ]
mail.output_mail

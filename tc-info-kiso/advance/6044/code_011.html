module Subject
  def initialize
    @observers=[]
  end

  def add_observer(observer)
    @observers << observer
  end

  def notify_observers
    @observers.each do |observer|
      observer.update(self)
    end
  end
end

  # 従業員クラス
class Employee
  # Subject moduleをinclude
  include Subject
  attr_reader :name, :salary

  def initialize(name, salary)
    # Subject moduleのinitializeを行う
    super()
    @name = name
    @salary = salary
  end

  def salary=(new_salary)
    @salary = new_salary
    notify_observers
  end
end

# 上司クラス
class Manager
  def update(changed_employee)
    puts "上司：#{changed_employee.name}さんの給料は#{changed_employee.salary}円です！"
  end
end
  
  # 経理担当クラス
class Accountant
  def update(changed_employee)
    puts "経理部門：#{changed_employee.name}さんの給料は#{changed_employee.salary}円です!"
  end
end

takashi = Employee.new('takashi', 10000)
takashi.add_observer(Manager.new)
takashi.add_observer(Accountant.new)
takashi.salary = 9999999

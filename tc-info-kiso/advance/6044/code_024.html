class User < ApplicationRecord
  after_update :notify_password_update

  def notify_password_update
    AdminMailer.with(user: self.email).passwaord_update_mail.deliver_later
    # Slack通知用のオブジェクトを作成
    notifier = Slack::Notifier.new(Rails.application.config.slack_webhook_url)
    # Slackに通知する
    notifier.ping("#{self}さんがパスワードを変更しました！")
  end
end

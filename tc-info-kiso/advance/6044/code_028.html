class User < ApplicationRecord
  # ここに記述した処理をUserObserverに移動する
end

class UserObserver < ActiveRecord::Observer
  # ユーザーがパスワードを変更した時に、以下のメソッドが動作する
  def after_update(user)
    AdminMailer.with(user: user.email).passwaord_update_mail.deliver_later
    # Slack通知用のオブジェクトを作成する
    notifier = Slack::Notifier.new(Rails.application.config.slack_webhook_url)
    # Slackに通知する
    notifier.ping("#{user.name}さんがパスワードを変更しました！")
  end
end

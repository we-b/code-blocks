# 人間クラス
class Human
  attr_reader :vocal_cords, :mouth, :right_arm, :left_arm

  def initialize
    # 人間の各部位
    @vocal_cords = '声帯'
    @mouth = '口'
    @right_arm = '右腕'
    @left_arm = '左腕'
  end

  # 人間が話す際の一連の動作
  def speak
    neuro_transmission(vocal_cords) # 神経を通って声帯を動かすような命令が出る
    neuro_transmission(mouth) # 神経を通って口を動かすような命令が出る
    puts '喋ります'
  end

 # 人間が右手を動かす際の一連の動作
  def move_right_arm
    neuro_transmission(right_arm) # 神経を通って右手を動かすような命令が出る
    puts '右腕を動かします'
  end

  # 人間が左手を動かす際の一連の動作
  def move_left_arm
    neuro_transmission(left_arm) # 神経を通って左手を動かすような命令が出る
    puts '左腕を動かします'
  end

  # 体の各部位を動かす際に必ず呼ばれる命令
  # 各部位を動かすためには神経伝達物質が各部位に到達する必要がある
  def neuro_transmission(part)
    puts "神経伝達物質が#{part}に到達しました"
  end

  # 風邪をひいたときの症状
  def catch_a_cold
    immune_function # 免疫機能が働く
    puts '熱が出ます'
  end

  # 風邪をひいたときに呼び出されるメソッド
  def immune_function
    puts '免疫機能が働きます'
  end
end

human = Human.new
human.speak
human.move_left_arm
human.move_right_arm
human.catch_a_cold

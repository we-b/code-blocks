# 顧客クラス
class Customer
  def admission_fee # 入場料
    500
  end
end

# 受付クラス
class Reception
  # 入場料の請求
  def request_payment
    puts "#{customers.length}人入場するには#{sum}円必要です。"
  end

  # 入場料の計算
  def sum
    @sum = 0
    customers.each do |customer|
      @sum += customer.admission_fee
    end
    @sum
  end

  # インスタンス生成の分離
  def customers
    @customers ||= [Customer.new, Customer.new]
  end
end

reception = Reception.new
reception.request_payment

class Hero
  attr_reader :hero_type, :hit_point, :power

  def initialize(args)
    @hero_type = args.fetch(:hero_type)
    @hit_point = args.fetch(:hit_point)
    @power = args.fetch(:power)
  end

  def detail
    puts "#{hero_type}の攻撃力は#{power}でHPは#{hit_point}です。"
  end
end

class Enemy
  attr_reader :type, :hit_point, :power

  def initialize(args)
    @type = args.fetch(:type)
    @hit_point = args.fetch(:hit_point)
    @power = args.fetch(:power)
  end

  def enemy_detail
    puts "#{type}の攻撃力は#{power}でHPは#{hit_point}です。"
  end
end

class CharacterStatus
  attr_reader :characters

  def initialize(characters)
    @characters = characters
  end

  def check_status
    display_character
    character = select_character
    detail_character(character)
  end

  private

  def display_character
    puts '確認したいキャラクターを選択してください。'
    characters.each_with_index do |character, i|
      if character.kind_of?(Hero)
        puts "[#{i}] タイプ： #{character.hero_type}"
      else
        puts "[#{i}] タイプ： #{character.type}"
      end
    end
  end

  def select_character
    character = characters[gets.to_i]
    if character.kind_of?(Hero)
      puts "#{character.hero_type}が選択されました"
    else
      puts "#{character.type}が選択されました"
    end
    character
  end

  def detail_character(character)
    if character.kind_of?(Hero)
      character.detail
    else
      character.enemy_detail
    end
  end
end

heroes = [
 { hero_type: '剣士', hit_point: 200, power: 100 },
 { hero_type: '魔法使い', hit_point: 100, power: 30 },
 { hero_type: '僧侶', hit_point: 150, power: 50 }
]

@heroes = heroes.map { |b| Hero.new(hero_type: b[:hero_type], hit_point: b[:hit_point], power: b[:power]) }
cs = CharacterStatus.new(@heroes)
cs.check_status

enemies = [
 { type: 'スライム', hit_point: 5, power: 5 },
 { type: 'ゴーレム', hit_point: 500, power: 300 },
 { type: 'ドラゴン', hit_point: 700, power: 400 }
]

@enemies = enemies.map { |b| Enemy.new(type: b[:type], hit_point: b[:hit_point], power: b[:power]) }
cs = CharacterStatus.new(@enemies)
cs.check_status

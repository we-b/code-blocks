require 'forwardable'

class FileManager
  def initialize(path)
    # path名のファイルを作成。
    @file = File.open(path, "w")
  end

  def write_line(line)
    @file.print(line)
    @file.print("\n")
  end
end

class FileManageDecorator
  extend Forwardable

  # 第一引数にオブジェクトを参照している属性、第二引数以降に委譲させたいメソッド名を指定。
  def_delegator :@real_writer, :write_line
  def initialize(real_writer)
    @real_writer = real_writer
  end
end

class LineNumberWriter < FileManageDecorator
  def initialize(real_writer)
    super(real_writer)
    @line_number = 1
  end

  def write_line(text)
    @real_writer.write_line("#{@line_number}: #{text}")
    @line_number += 1
  end
end

class BlankLineWriter < FileManageDecorator
  def write_line(text)
    @real_writer.write_line("#{text}")
    @real_writer.write_line("")
  end
end

line_writer = LineNumberWriter.new(FileManager.new('line.txt'))
line_writer.write_line('1行目へ記述するテキストです。')
line_writer.write_line('2行目へ記述するテキストです。')
line_writer.write_line('3行目へ記述するテキストです。')

blank_writer = BlankLineWriter.new(FileManager.new('blank.txt'))
blank_writer.write_line('1行目に記述し、次は1行空行にします。')
blank_writer.write_line('3行目に記述し、次は1行空行にします。')
blank_writer.write_line('5行目に記述し、次は1行空行にします。')

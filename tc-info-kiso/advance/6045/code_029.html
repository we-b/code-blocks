# requireを使って有効化する。
require 'forwardable'

# 装飾されるクラスはとくに変化なし。
class Subject
  # 中略
end

class Decorator
  # Decoratorとなるクラスにextendで読み込ませる。
  extend Forwardable
  # def_delegatorでメソッドの委譲先を指定する。
  def_delegator :@real_writer, :write_line
  def initialize(real_writer)
    @real_writer = real_writer
  end
end

require 'rails_helper'

RSpec.describe Tasks::DuplicateService, type: :service do
  let(:genre) { Genre.create!(name: 'テストジャンル') }

  # ============================================================
  # 1. 正常系 - 属性の引き継ぎ
  # ============================================================
  describe '属性の引き継ぎ' do
    context 'name の引き継ぎ' do
      it 'name が正しく引き継がれ、末尾に「(コピー)」が追加される' do
        original_task = Task.create!(
          name: '買い物',
          genre: genre
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.name).to eq('買い物(コピー)')
      end
    end

    context 'explanation の引き継ぎ' do
      it 'explanation がそのまま引き継がれる' do
        original_task = Task.create!(
          name: 'テストタスク',
          explanation: '詳細な説明文です',
          genre: genre
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.explanation).to eq('詳細な説明文です')
      end
    end

    context 'genre_id の引き継ぎ' do
      it 'genre_id がそのまま引き継がれる' do
        original_task = Task.create!(
          name: 'テストタスク',
          genre: genre
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.genre_id).to eq(genre.id)
      end
    end

    context 'priority の引き継ぎ' do
      it 'priority が low の場合、そのまま引き継がれる' do
        original_task = Task.create!(
          name: 'テストタスク',
          genre: genre,
          priority: :low
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.priority).to eq('low')
        expect(result.task.low?).to be true
      end

      it 'priority が medium の場合、そのまま引き継がれる' do
        original_task = Task.create!(
          name: 'テストタスク',
          genre: genre,
          priority: :medium
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.priority).to eq('medium')
        expect(result.task.medium?).to be true
      end

      it 'priority が high の場合、そのまま引き継がれる' do
        original_task = Task.create!(
          name: 'テストタスク',
          genre: genre,
          priority: :high
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.priority).to eq('high')
        expect(result.task.high?).to be true
      end
    end
  end

  # ============================================================
  # 2. 正常系 - 特別な処理
  # ============================================================
  describe '特別な処理' do
    context 'status の初期化' do
      it 'status が「完了」のタスクを複製すると、新タスクは「未着手」になる' do
        original_task = Task.create!(
          name: 'テストタスク',
          genre: genre,
          status: 2  # 完了
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.status).to eq(0)  # 未着手
      end

      it 'status が「進行中」のタスクを複製すると、新タスクは「未着手」になる' do
        original_task = Task.create!(
          name: 'テストタスク',
          genre: genre,
          status: 1  # 進行中
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.status).to eq(0)  # 未着手
      end

      it 'status が「未着手」のタスクを複製すると、新タスクも「未着手」になる' do
        original_task = Task.create!(
          name: 'テストタスク',
          genre: genre,
          status: 0  # 未着手
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.status).to eq(0)  # 未着手
      end
    end

    context 'deadline_date のリセット' do
      it 'deadline_date が設定されているタスクを複製すると、新タスクの deadline_date は nil になる' do
        original_task = Task.create!(
          name: 'テストタスク',
          genre: genre,
          deadline_date: Date.new(2025, 12, 31)
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.deadline_date).to be_nil
      end

      it 'deadline_date が nil のタスクを複製すると、新タスクの deadline_date も nil になる' do
        original_task = Task.create!(
          name: 'テストタスク',
          genre: genre,
          deadline_date: nil
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.deadline_date).to be_nil
      end
    end
  end

  # ============================================================
  # 3. 正常系 - 境界値・エッジケース
  # ============================================================
  describe '境界値・エッジケース' do
    context 'name の境界値' do
      it 'name が空文字のタスクを複製すると、「(コピー)」のみの name になる' do
        original_task = Task.create!(
          name: '',
          genre: genre
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.name).to eq('(コピー)')
      end

      it 'name が非常に長い文字列でも、末尾に「(コピー)」が追加される' do
        long_name = 'あ' * 100
        original_task = Task.create!(
          name: long_name,
          genre: genre
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.name).to eq("#{long_name}(コピー)")
      end

      it 'name に既に「(コピー)」が含まれるタスクを複製すると、「(コピー)(コピー)」になる' do
        original_task = Task.create!(
          name: 'タスク(コピー)',
          genre: genre
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.name).to eq('タスク(コピー)(コピー)')
      end
    end

    context 'explanation の境界値' do
      it 'explanation が nil のタスクを複製すると、新タスクの explanation も nil になる' do
        original_task = Task.create!(
          name: 'テストタスク',
          genre: genre,
          explanation: nil
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.explanation).to be_nil
      end

      it 'explanation が空文字のタスクを複製すると、新タスクの explanation も空文字になる' do
        original_task = Task.create!(
          name: 'テストタスク',
          genre: genre,
          explanation: ''
        )

        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.explanation).to eq('')
      end
    end
  end

  # ============================================================
  # 4. 正常系 - データの独立性
  # ============================================================
  describe 'データの独立性' do
    it '複製後、元タスクと新タスクのIDが異なる' do
      original_task = Task.create!(
        name: 'テストタスク',
        genre: genre
      )

      result = described_class.new(original_task.id).call

      expect(result.success?).to be true
      expect(result.task.id).not_to eq(original_task.id)
    end

    it '複製後、新タスクの created_at は複製実行時の時刻になる' do
      original_task = Task.create!(
        name: 'テストタスク',
        genre: genre
      )
      original_created_at = original_task.created_at

      travel_to(1.day.from_now) do
        result = described_class.new(original_task.id).call

        expect(result.success?).to be true
        expect(result.task.created_at).to be > original_created_at
      end
    end

    it '複製後、元タスクを更新しても新タスクに影響しない' do
      original_task = Task.create!(
        name: 'テストタスク',
        explanation: '元の説明',
        genre: genre
      )

      result = described_class.new(original_task.id).call
      duplicated_task = result.task

      # 元タスクを更新
      original_task.update!(name: '更新後のタスク', explanation: '更新後の説明')

      # 複製したタスクは影響を受けない
      duplicated_task.reload
      expect(duplicated_task.name).to eq('テストタスク(コピー)')
      expect(duplicated_task.explanation).to eq('元の説明')
    end

    it '複製によってタスクが1件増加する' do
      original_task = Task.create!(
        name: 'テストタスク',
        genre: genre
      )

      expect {
        described_class.new(original_task.id).call
      }.to change(Task, :count).by(1)
    end
  end

  # ============================================================
  # 5. 異常系
  # ============================================================
  describe '異常系' do
    context '存在しないタスクIDを指定' do
      it 'ActiveRecord::RecordNotFound が発生する' do
        non_existent_id = 999999

        expect {
          described_class.new(non_existent_id).call
        }.to raise_error(ActiveRecord::RecordNotFound)
      end
    end

    context '不正なタスクIDを指定' do
      it 'nil を指定すると ActiveRecord::RecordNotFound が発生する' do
        expect {
          described_class.new(nil).call
        }.to raise_error(ActiveRecord::RecordNotFound)
      end
    end
  end
end

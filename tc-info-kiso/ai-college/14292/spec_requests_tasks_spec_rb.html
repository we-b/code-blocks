require 'rails_helper'

RSpec.describe "Tasks API", type: :request do
  let!(:genre) { Genre.create!(name: 'テストジャンル') }

  // 省略

  # ============================================================
  # POST /tasks/:id/duplicate - タスク複製API
  # ============================================================
  describe 'POST /tasks/:id/duplicate' do
    # ----------------------------------------------------------
    # 6. 正常系 - HTTPレスポンス
    # ----------------------------------------------------------
    describe '正常系' do
      let!(:original_task) do
        Task.create!(
          name: '元のタスク',
          explanation: 'これは元のタスクの説明です',
          genre: genre,
          priority: :high,
          status: 2,
          deadline_date: Date.new(2025, 12, 31)
        )
      end

      context 'HTTPレスポンス' do
        it '正常にリクエストすると HTTP 200 OK が返される' do
          post "/tasks/#{original_task.id}/duplicate", as: :json

          expect(response).to have_http_status(:ok)
        end

        it 'レスポンスにタスク一覧（配列）が含まれる' do
          post "/tasks/#{original_task.id}/duplicate", as: :json

          json_response = JSON.parse(response.body)
          expect(json_response).to be_an(Array)
        end

        it 'レスポンスに複製されたタスクの情報が含まれる' do
          post "/tasks/#{original_task.id}/duplicate", as: :json

          json_response = JSON.parse(response.body)
          duplicated_task = json_response.find { |task| task['name'] == '元のタスク(コピー)' }

          expect(duplicated_task).not_to be_nil
        end
      end

      context 'データベースへの影響' do
        it 'タスクが1件追加される' do
          expect {
            post "/tasks/#{original_task.id}/duplicate", as: :json
          }.to change(Task, :count).by(1)
        end

        it '元のタスクは削除されない' do
          post "/tasks/#{original_task.id}/duplicate", as: :json

          expect(Task.exists?(original_task.id)).to be true
        end
      end

    end

    # ----------------------------------------------------------
    # 8. 異常系 - HTTPエラー
    # ----------------------------------------------------------
    describe '異常系' do
      context '存在しないタスクID' do
        it '存在しない task_id を指定すると HTTP 404 Not Found が返される' do
          non_existent_id = 999999

          post "/tasks/#{non_existent_id}/duplicate", as: :json

          expect(response).to have_http_status(:not_found)
        end

        it 'タスクは作成されない' do
          non_existent_id = 999999

          expect {
            post "/tasks/#{non_existent_id}/duplicate", as: :json
          }.not_to change(Task, :count)
        end
      end

      context '不正なタスクID形式' do
        it '文字列の task_id を指定すると HTTP 404 が返される' do
          post '/tasks/invalid_id/duplicate', as: :json

          expect(response).to have_http_status(:not_found)
        end

        it '負の値の task_id を指定すると HTTP 404 が返される' do
          post '/tasks/-1/duplicate', as: :json

          expect(response).to have_http_status(:not_found)
        end

        it '0 の task_id を指定すると HTTP 404 が返される' do
          post '/tasks/0/duplicate', as: :json

          expect(response).to have_http_status(:not_found)
        end
      end
    end
  end
end

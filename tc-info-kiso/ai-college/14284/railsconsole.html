# ケース1: titleが空
article = Article.new(title: '', body: 'テスト本文')
article.valid?
# => false
article.errors.full_messages
# => ["タイトルを入力してください"]

# ケース2: titleが101文字
article = Article.new(title: 'あ' * 101, body: 'テスト本文')
article.valid?
# => false
article.errors.full_messages
# => ["タイトルは100文字以内で入力してください"]

# ケース3: titleが空白のみ
article = Article.new(title: '   ', body: 'テスト本文')
article.valid?
# => false
article.errors.full_messages
# => ["タイトルに空白文字のみは使用できません"]

# ケース4: bodyが空
article = Article.new(title: 'テストタイトル', body: '')
article.valid?
# => false
article.errors.full_messages
# => ["本文を入力してください"]

# ケース5: 有効なデータ
article = Article.new(title: 'テストタイトル', body: 'テスト本文')
article.valid?
# => true

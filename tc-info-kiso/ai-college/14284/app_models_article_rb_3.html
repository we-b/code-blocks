class Article < ApplicationRecord
  enum status: {
    draft: 0,
    published: 1,
    archived: 2
  }, _prefix: true

  validates :title,
            presence: { message: 'を入力してください' },
            length: {
              in: 1..100,
              message: 'は100文字以内で入力してください'
            },
            format: {
              without: /\A\s*\z/,
              message: 'に空白文字のみは使用できません'
            }

  validates :body, presence: { message: 'を入力してください' }
end

require 'rails_helper'

RSpec.describe Article, type: :model do
  describe 'バリデーション' do
    context 'titleが空の場合' do
      it 'バリデーションに失敗すること' do
        article = build(:article, title: nil)
        expect(article).not_to be_valid
        expect(article.errors[:title]).to include("can't be blank")
      end
    end

    context 'bodyが空の場合' do
      it 'バリデーションに失敗すること' do
        article = build(:article, body: nil)
        expect(article).not_to be_valid
        expect(article.errors[:body]).to include("can't be blank")
      end
    end

    context '全ての属性が正しく設定されている場合' do
      it 'バリデーションに成功すること' do
        article = build(:article)
        expect(article).to be_valid
      end
    end
  end

  describe 'enum status' do
    context '新規作成時' do
      it 'デフォルトでdraftになること' do
        article = create(:article)
        expect(article).to be_status_draft
      end
    end

    context 'ステータスの変更' do
      let(:article) { create(:article) }

      it 'publishedに変更できること' do
        article.status_published!
        expect(article).to be_status_published
      end

      it 'archivedに変更できること' do
        article.status_archived!
        expect(article).to be_status_archived
      end
    end

    context '無効なステータス' do
      it 'ArgumentErrorが発生すること' do
        expect {
          create(:article, status: :invalid_status)
        }.to raise_error(ArgumentError)
      end
    end
  end
end

require 'rails_helper'

RSpec.describe 'Api::V1::Comments', type: :request do
  let(:article) { create(:article) }

  describe 'GET /api/v1/articles/:article_id/comments' do
    context 'コメントが存在する場合' do
      before do
        create_list(:comment, 3, article: article)
      end

      it 'コメント一覧を取得できること' do
        get "/api/v1/articles/#{article.id}/comments"

        expect(response).to have_http_status(:ok)
        json = JSON.parse(response.body)
        expect(json.length).to eq(3)
      end
    end

    context 'コメントが存在しない場合' do
      it '空の配列を返すこと' do
        get "/api/v1/articles/#{article.id}/comments"

        expect(response).to have_http_status(:ok)
        json = JSON.parse(response.body)
        expect(json).to eq([])
      end
    end

    context '記事が存在しない場合' do
      it '404を返すこと' do
        get '/api/v1/articles/999999/comments'

        expect(response).to have_http_status(:not_found)
      end
    end
  end

  describe 'POST /api/v1/articles/:article_id/comments' do
    context '正しいパラメータの場合' do
      let(:valid_params) do
        {
          comment: {
            author: 'テストユーザー',
            body: 'これはテストコメントです。'
          }
        }
      end

      it 'コメントを作成できること' do
        expect {
          post "/api/v1/articles/#{article.id}/comments", params: valid_params
        }.to change(Comment, :count).by(1)

        expect(response).to have_http_status(:created)
        json = JSON.parse(response.body)
        expect(json['author']).to eq('テストユーザー')
        expect(json['article_id']).to eq(article.id)
      end
    end

    context '不正なパラメータの場合' do
      let(:invalid_params) do
        {
          comment: {
            author: '',
            body: ''
          }
        }
      end

      it '422を返すこと' do
        post "/api/v1/articles/#{article.id}/comments", params: invalid_params

        expect(response).to have_http_status(:unprocessable_entity)
        json = JSON.parse(response.body)
        expect(json['errors']).to be_present
      end
    end

    context '記事が存在しない場合' do
      it '404を返すこと' do
        post '/api/v1/articles/999999/comments', params: {
          comment: { author: 'test', body: 'test' }
        }

        expect(response).to have_http_status(:not_found)
      end
    end
  end

  describe 'DELETE /api/v1/articles/:article_id/comments/:id' do
    context 'コメントが存在する場合' do
      let!(:comment) { create(:comment, article: article) }

      it 'コメントを削除できること' do
        expect {
          delete "/api/v1/articles/#{article.id}/comments/#{comment.id}"
        }.to change(Comment, :count).by(-1)

        expect(response).to have_http_status(:no_content)
      end
    end

    context 'コメントが存在しない場合' do
      it '404を返すこと' do
        delete "/api/v1/articles/#{article.id}/comments/999999"

        expect(response).to have_http_status(:not_found)
      end
    end
  end
end

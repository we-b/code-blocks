import java.util.Scanner;

public class PrefectureSelector {
    public static void main(String[] args) {
        String[] prefectures = {"北海道", "青森", "...(省略)...", "沖縄"};

        selectPrefecture(prefectures, "birthplace");
        selectPrefecture(prefectures, "address");
        selectPrefecture(prefectures, "work_address");
    }

    public static void selectPrefecture(String[] prefectures, String category) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("番号を入力してください: ");
        int num = scanner.nextInt();

        if (num >= 0 && num < prefectures.length) {
            String selectedPrefecture = prefectures[num];
            switch (category) {
                case "birthplace":
                    System.out.println("出生地: " + selectedPrefecture);
                    break;
                case "address":
                    System.out.println("住所: " + selectedPrefecture);
                    break;
                case "work_address":
                    System.out.println("勤務地: " + selectedPrefecture);
                    break;
                default:
                    System.out.println("不明なカテゴリです");
            }
        } else {
            System.out.println("無効な番号です");
        }
    }
}

import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String[] prefecture = {"北海道", "青森", "...(省略)...", "沖縄"};

        selectBirthplace(scanner, prefecture);
        selectAddress(scanner, prefecture);
        selectWorkAddress(scanner, prefecture);
    }

    public static void selectBirthplace(Scanner scanner, String[] prefecture) {
        System.out.print("出身地の番号を選んでください: ");
        int num = scanner.nextInt();
        if (num >= 0 && num < prefecture.length) {
            String birthplace = prefecture[num];
            System.out.println("出身地: " + birthplace);
        } else {
            System.out.println("無効な番号です。");
        }
    }

    public static void selectAddress(Scanner scanner, String[] prefecture) {
        System.out.print("住所の番号を選んでください: ");
        int num = scanner.nextInt();
        if (num >= 0 && num < prefecture.length) {
            String address = prefecture[num];
            System.out.println("住所: " + address);
        } else {
            System.out.println("無効な番号です。");
        }
    }

    public static void selectWorkAddress(Scanner scanner, String[] prefecture) {
        System.out.print("勤務先住所の番号を選んでください: ");
        int num = scanner.nextInt();
        if (num >= 0 && num < prefecture.length) {
            String workAddress = prefecture[num];
            System.out.println("勤務先住所: " + workAddress);
        } else {
            System.out.println("無効な番号です。");
        }
    }
}

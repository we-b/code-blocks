<script setup>
import { ref, computed } from 'vue';

const inputMinutes = ref(0);
const time = ref(0);

  const minutes = Math.floor(time.value / 60);
  const seconds = ((time.value / 60) % 1) * 60;
  return `${minutes}:${seconds.toFixed(0).padStart(2, '0')}`;
});

function setTime() {
  time.value = inputMinutes.value * 60;
}

public abstract class AdTemplate {
  public void displayAd() {
    displayHeader();
    displayBody();
    displayFooter();
  }

  private void displayHeader() {
    System.out.println("株式会社〇〇");
  }

  private void displayFooter() {
    System.out.println("詳しくはこちら");
  }

  protected abstract void displayBody();
}

public class GifAd extends AdTemplate {
  private final int imageCount;

  public GifAd(int imageCount) {
      this.imageCount = imageCount;
  }

  @Override
  protected void displayBody() {
      if (imageCount == 2) {
          System.out.println("GIF広告: 2つの画像を表示");
      } else {
          System.out.println("GIF広告: 1つの画像を表示");
      }
  }
}

public class ImageAd extends AdTemplate {
  @Override
  protected void displayBody() {
      System.out.println("画像広告を表示");
  }
}

public class MovieAd extends AdTemplate {
  @Override
  protected void displayBody() {
      System.out.println("動画広告を表示");
  }
}

public class AdService {
  public void displayAd(String adType, int imageCount) {
    AdTemplate adTemplate;

    switch (adType) {
      case "gif":
        adTemplate = new GifAd(imageCount);
        break;
      case "image":
        adTemplate = new ImageAd();
        break;
      case "movie":
        adTemplate = new MovieAd();
        break;
      default:
        System.out.println("デフォルト画像を表示");
        return;
    }

    adTemplate.displayAd();
  }
}

public class Main {
  public static void main(String[] args) {
      AdService adService = new AdService();
      adService.displayAd("gif", 1);         // GIF広告を表示
      adService.displayAd("image", 0);       // 画像広告を表示
      adService.displayAd("movie", 0);       // ムービー広告を表示
      adService.displayAd("unknown", 0);     // 未知の広告形式
  }
}

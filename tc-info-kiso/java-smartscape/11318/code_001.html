public class CookingCurry {
  public void prepareDish() {
    gatherIngredients();
    cookCurry();
    serve();
  }
  
  private void gatherIngredients() {
    System.out.println("材料を集める");
  }

  private void cookCurry() {
    System.out.println("食材を煮込む");
  }

  private void serve() {
    System.out.println("盛り付ける");
  }
}

public class CookingPasta {
  public void prepareDish() {
    gatherIngredients();
    cookPasta();
    serve();
  }
  
  private void gatherIngredients() {
    System.out.println("材料を集める");
  }

  private void cookPasta() {
    System.out.println("パスタを茹でる");
    System.out.println("トッピングと和える");
  }

  private void serve() {
    System.out.println("盛り付ける");
  }
}

public class Main {
  public static void main(String[] args) {
    CookingCurry curry = new CookingCurry();
    curry.prepareDish();
    
    CookingPasta pasta = new CookingPasta();
    pasta.prepareDish();
  }
}

import java.util.ArrayList;
import java.util.List;

// 掲示板クラス
public class BulletinBoard {
  private List<String> messages;

  public BulletinBoard() {
    messages = new ArrayList<>();
  }

  public void postMessage(String message) {
    messages.add(message);
    System.out.println("新しいメッセージ: " + message);
  }

  public void displayMessages() {
    System.out.println("メッセージ一覧:");
    for (String message : messages) {
      System.out.println("- " + message);
    }
  }
}

public class Organization {
  private String name;
  private BulletinBoard bulletinBoard; // 組織に共有される掲示板

  public Organization(String name, BulletinBoard bulletinBoard) {
    this.name = name;
    this.bulletinBoard = bulletinBoard; // 共有掲示板を渡す
  }

  public void postMessage(String message) {
    bulletinBoard.postMessage(name + "からの: " + message);
  }
}

public class Main {
  public static void main(String[] args) {
    // 掲示板インスタンスを一つだけ作成
    BulletinBoard sharedBoard = new BulletinBoard();

    Organization salesOrg = new Organization("営業部", sharedBoard);
    Organization developmentOrg = new Organization("開発部", sharedBoard);

    // 各組織が同じ掲示板にメッセージを投稿
    salesOrg.postMessage("売上が前年同期比で10%増加しました！");
    developmentOrg.postMessage("新しいプロジェクトがスタートしました！");

    // 統一された掲示板に投稿されたメッセージを表示
    sharedBoard.displayMessages();
  }
}

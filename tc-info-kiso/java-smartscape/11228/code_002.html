import java.util.Scanner;

public class PrefectureSelector {
    public static void main(String[] args) {
        String[] prefectures = {"北海道", "青森", "...(省略)...", "沖縄"};
        
        selectBirthplace(prefectures);
        selectAddress(prefectures);
        selectWorkAddress(prefectures);
    }

    public static void selectBirthplace(String[] prefectures) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("番号を入力してください: ");
        int num = scanner.nextInt();
        System.out.println("出生地: " + prefectures[num]);
    }

    public static void selectAddress(String[] prefectures) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("番号を入力してください: ");
        int num = scanner.nextInt();
        System.out.println("住所: " + prefectures[num]);
    }

    public static void selectWorkAddress(String[] prefectures) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("番号を入力してください: ");
        int num = scanner.nextInt();
        System.out.println("勤務地: " + prefectures[num]);
    }
}

<!DOCTYPE html>
<head>
    <title>Vue.js App</title>
</head>
<body>
    <div id="app">
        <h1>{{ count }}</h1>
        <button v-on:click="decreaseCount">−</button>
        <button v-on:click="increaseCount">+</button>
    </div>
    <script src="https://unpkg.com/vue@3/dist/vue.global.js"></script>

    <script>
    const app = Vue.createApp({
        data() {
            return {
                count: 0
            };
        },
        methods: {
            increaseCount() {
                this.count += 1;
            },
            decreaseCount() {
                if (this.count > 0) {
                    this.count -= 1;
                }
            }
        }
    });
    app.mount('#app');
    </script>
</body>
</html>

package in.tech_camp.donation_app.form;

import jakarta.validation.*;
import org.junit.jupiter.api.*;
import java.util.Set;

import static org.junit.jupiter.api.Assertions.*;

class DonationAddressFormTest {

    private Validator validator;

    @BeforeEach
    void setUp() {
        ValidatorFactory factory = Validation.buildDefaultValidatorFactory();
        validator = factory.getValidator();
    }

    private DonationAddressForm createValidForm() {
        DonationAddressForm form = new DonationAddressForm();
        form.setPostalCode("123-4567");
        form.setPrefecture(1);
        form.setCity("千代田区");
        form.setHouseNumber("1-1");
        form.setBuildingName("東京ハイツ");
        form.setPrice(2000);
        return form;
    }

    @Nested
    @DisplayName("内容に問題ない場合")
    class ValidCases {

        @Test
        @DisplayName("すべての値が正しく入力されていれば有効であること")
        void allValid() {
            DonationAddressForm form = createValidForm();
            Set<ConstraintViolation<DonationAddressForm>> violations = validator.validate(form);
            assertTrue(violations.isEmpty());
        }

        @Test
        @DisplayName("cityは空でも有効であること")
        void cityCanBeEmpty() {
            DonationAddressForm form = createValidForm();
            form.setCity("");
            Set<ConstraintViolation<DonationAddressForm>> violations = validator.validate(form);
            assertTrue(violations.isEmpty());
        }

        @Test
        @DisplayName("houseNumberは空でも有効であること")
        void houseNumberCanBeEmpty() {
            DonationAddressForm form = createValidForm();
            form.setHouseNumber("");
            Set<ConstraintViolation<DonationAddressForm>> violations = validator.validate(form);
            assertTrue(violations.isEmpty());
        }

        @Test
        @DisplayName("buildingNameは空でも有効であること")
        void buildingNameCanBeEmpty() {
            DonationAddressForm form = createValidForm();
            form.setBuildingName("");
            Set<ConstraintViolation<DonationAddressForm>> violations = validator.validate(form);
            assertTrue(violations.isEmpty());
        }
    }

    @Nested
    @DisplayName("内容に問題がある場合")
    class InvalidCases {

        @Test
        @DisplayName("postalCodeが空だと無効であること")
        void postalCodeBlank() {
            DonationAddressForm form = createValidForm();
            form.setPostalCode("");
            Set<ConstraintViolation<DonationAddressForm>> violations = validator.validate(form);
            assertFalse(violations.isEmpty());
        }

        @Test
        @DisplayName("postalCodeがハイフンを含んだ正しい形式でないと無効であること")
        void postalCodeInvalidFormat() {
            DonationAddressForm form = createValidForm();
            form.setPostalCode("1234567");
            Set<ConstraintViolation<DonationAddressForm>> violations = validator.validate(form);
            assertFalse(violations.isEmpty());
        }

        @Test
        @DisplayName("prefectureを選択していないと無効であること")
        void prefectureNotSelected() {
            DonationAddressForm form = createValidForm();
            form.setPrefecture(0);
            Set<ConstraintViolation<DonationAddressForm>> violations = validator.validate(form);
            assertFalse(violations.isEmpty());
        }

        @Test
        @DisplayName("priceが空だと無効であること")
        void priceNull() {
            DonationAddressForm form = createValidForm();
            form.setPrice(null);
            Set<ConstraintViolation<DonationAddressForm>> violations = validator.validate(form);
            assertFalse(violations.isEmpty());
        }

        @Test
        @DisplayName("priceが1円未満では無効であること")
        void priceLessThanOne() {
            DonationAddressForm form = createValidForm();
            form.setPrice(0);
            Set<ConstraintViolation<DonationAddressForm>> violations = validator.validate(form);
            assertFalse(violations.isEmpty());
        }

        @Test
        @DisplayName("priceが1,000,000円を超過すると無効であること")
        void priceExceedsMillion() {
            DonationAddressForm form = createValidForm();
            form.setPrice(1000001);
            Set<ConstraintViolation<DonationAddressForm>> violations = validator.validate(form);
            assertFalse(violations.isEmpty());
        }
    }
}

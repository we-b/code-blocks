    GUNMA(10, "群馬県"), SAITAMA(11, "埼玉県"), CHIBA(12, "千葉県"),
    TOKYO(13, "東京都"), KANAGAWA(14, "神奈川県"), NIIGATA(15, "新潟県"),
    TOYAMA(16, "富山県"), ISHIKAWA(17, "石川県"), FUKUI(18, "福井県"),
    YAMANASHI(19, "山梨県"), NAGANO(20, "長野県"), GIFU(21, "岐阜県"),
    SHIZUOKA(22, "静岡県"), AICHI(23, "愛知県"), MIE(24, "三重県"),

    private final int id;
    private final String name;

    Prefecture(int id, String name) {
        this.id = id;
        this.name = name;
    }

    public int getId() { return id; }
    public String getName() { return name; }

    public static Prefecture fromId(int id) {
        for (Prefecture p : Prefecture.values()) {
            if (p.getId() == id) return p;
        }
        throw new IllegalArgumentException("Invalid prefecture id: " + id);
    }

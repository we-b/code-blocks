package in.tech_camp.donation_app.form;

import jakarta.validation.constraints.*;
import lombok.Data;

@Data
public class DonationAddressForm {

    // 寄付金額のバリデーション
    @NotNull(message = "金額を入力してください")
    @Min(value = 1, message = "金額は1円以上にしてください")
    @Max(value = 1000000, message = "金額は1,000,000円以下にしてください")
    private Integer price;

    // 郵便番号のバリデーション
    @NotBlank(message = "郵便番号を入力してください")
    @Pattern(regexp = "\\d{3}-\\d{4}", message = "郵便番号はハイフンを含む正しい形式で入力してください")
    private String postalCode;

    // 都道府県のバリデーション（0以外の値を許可）
    @Min(value = 1, message = "都道府県を選択してください")
    private int prefecture;

    // 任意項目
    private String city;
    private String houseNumber;
    private String buildingName;
}

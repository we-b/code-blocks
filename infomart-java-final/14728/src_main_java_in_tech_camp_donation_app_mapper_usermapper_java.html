package in.tech_camp.donation_app.mapper;

import in.tech_camp.donation_app.entity.User;
import org.apache.ibatis.annotations.*;

@Mapper
public interface UserMapper {
    @Select("SELECT * FROM users WHERE email = #{email}")
    User findByEmail(String email);

    @Insert("INSERT INTO users (email, password, name, name_reading, nickname) VALUES (#{email}, #{password}, #{name}, #{nameReading}, #{nickname})")
    @Options(useGeneratedKeys = true, keyProperty = "id")
    void insert(User user);
}

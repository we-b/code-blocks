package in.tech_camp.donation_app.service;

import in.tech_camp.donation_app.entity.*;
import in.tech_camp.donation_app.form.DonationAddressForm;
import in.tech_camp.donation_app.mapper.*;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
@RequiredArgsConstructor
public class DonationService {

    private final DonationMapper donationMapper;
    private final AddressMapper addressMapper;

    @Transactional
    public void save(DonationAddressForm form, Long userId) {
        // 寄付情報を保存し、自動生成されたIDを取得
        Donation donation = new Donation();
        donation.setPrice(form.getPrice());
        donation.setUserId(userId);
        donationMapper.insert(donation);

        // 住所を保存（donation_idに先ほど保存した寄付情報のIDを指定）
        Address address = new Address();
        address.setPostalCode(form.getPostalCode());
        address.setPrefecture(form.getPrefecture());
        address.setCity(form.getCity());
        address.setHouseNumber(form.getHouseNumber());
        address.setBuildingName(form.getBuildingName());
        address.setDonationId(donation.getId());
        addressMapper.insert(address);
    }
}

package in.tech_camp.donation_app.controller;

import in.tech_camp.donation_app.entity.*;
import in.tech_camp.donation_app.mapper.*;
import lombok.RequiredArgsConstructor;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;

@Controller
@RequiredArgsConstructor
public class DonationController {

    private final DonationMapper donationMapper;
    private final AddressMapper addressMapper;
    private final UserMapper userMapper;

    @GetMapping("/donations")
    public String index() {
        return "donations/index";
    }

    @GetMapping("/donations/new")
    public String newDonation(Model model) {
        model.addAttribute("prefectures", Prefecture.values());
        return "donations/new";
    }

    @PostMapping("/donations")
    public String create(@RequestParam String postalCode,
                         @RequestParam int prefecture,
                         @RequestParam(required = false) String city,
                         @RequestParam(required = false) String houseNumber,
                         @RequestParam(required = false) String buildingName,
                         @RequestParam int price,
                         @AuthenticationPrincipal UserDetails userDetails) {
        // ログイン中のユーザーを取得
        User currentUser = userMapper.findByEmail(userDetails.getUsername());

        // 寄付情報を保存
        Donation donation = new Donation();
        donation.setPrice(price);
        donation.setUserId(currentUser.getId());
        donationMapper.insert(donation);

        // 住所情報を保存
        Address address = new Address();
        address.setPostalCode(postalCode);
        address.setPrefecture(prefecture);
        address.setCity(city);
        address.setHouseNumber(houseNumber);
        address.setBuildingName(buildingName);
        address.setDonationId(donation.getId());
        addressMapper.insert(address);

        return "redirect:/donations";
    }
}

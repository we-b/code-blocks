package in.tech_camp.donation_app.controller;

import in.tech_camp.donation_app.entity.Prefecture;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;

@Controller
public class DonationController {

    private static final Logger logger = LoggerFactory.getLogger(DonationController.class);

    @GetMapping("/donations")
    public String index() {
        return "donations/index";
    }

    @GetMapping("/donations/new")
    public String newDonation(Model model) {
        model.addAttribute("prefectures", Prefecture.values());
        return "donations/new";
    }

    @PostMapping("/donations")
    public String create(@RequestParam String postalCode,
                         @RequestParam int prefecture,
                         @RequestParam(required = false) String city,
                         @RequestParam(required = false) String houseNumber,
                         @RequestParam(required = false) String buildingName,
                         @RequestParam int price) {
        logger.info("postalCode: {}", postalCode);
        logger.info("prefecture: {}", prefecture);
        logger.info("city: {}", city);
        logger.info("houseNumber: {}", houseNumber);
        logger.info("buildingName: {}", buildingName);
        logger.info("price: {}", price);
        return "redirect:/donations";
    }
}

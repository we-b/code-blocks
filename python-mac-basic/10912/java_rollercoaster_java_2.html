import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class RollerCoaster {
  public static void main(String[] args) {
    int rideCount = 0;
    List<Map<String, String>> friends = new ArrayList<>();

    try (Scanner scanner = new Scanner(System.in)){

      for (int i = 0; i < 3; i++) {
        Map<String, String> friend = new HashMap<>();
        System.out.println("お友達の名前は？");
        friend.put("name", scanner.nextLine());
        System.out.println("お友達の身長は？");
        friend.put("height", scanner.nextLine());
        friends.add(friend);
      }
  
      for (Map<String, String> friend : friends) {
        if (Integer.parseInt(friend.get("height")) >= 130) {
          System.out.println(friend.get("name") + "くんは乗車できます");
          rideCount++;
        } else {
          System.out.println(friend.get("name") + "くんは乗車できません！");
        }
      }  

      System.out.println("乗車するのは" + rideCount + "人です");
  
    }
  }
}

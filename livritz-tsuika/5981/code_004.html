function youtube_url(id){
 return `https://www.youtube.com/embed/${id}`
}

function tiktok_url(id){
 return `https://www.tiktok.com/@${id}`
}

let youtube_movies = ["19883950", "81293482", "80931242"]
let youtube_movie_urls = []
youtube_movies.forEach(id => { youtube_movie_urls.push(youtube_url(id)) })
console.log("youtube_moviesに格納されたYouTubeの動画を表示するURLを表示します")
console.log(youtube_movie_urls)

let tiktok_movies = ["31290843", "19841902", "92103929"]
let tiktok_movie_urls = []
tiktok_movies.forEach(id => { tiktok_movie_urls.push(tiktok_url(id)) })
console.log("tiktok_moviesに格納されたTikTokの動画を表示するURLを表示します")
console.log(tiktok_movie_urls)

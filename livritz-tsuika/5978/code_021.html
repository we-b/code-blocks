# 印刷機クラス
class Printer
def initialize(jobs)
  @jobs = jobs
end

# ① 印刷待ちリストに追加された順に印刷できるように末尾に追加するようにしている
def add_job(job)
  @jobs << job
  puts "#{job}を印刷待ちリストに追加します"
end

# ② 印刷待ちリストに追加された順に印刷できるように印刷待ちリストの先頭を印刷
def print
  puts "#{@jobs.shift}を印刷します"
  # ③ shiftメソッドを使用しないと元々の配列の先頭が消えない
end

def display_jobs
  # ④ TODO: 画面に表示させるフォーマットを整える
  puts '印刷待ちのページ'
  puts @jobs
end

end

jobs = ['１ページ目', '２ページ目', '３ページ目', '４ページ目']
printer = Printer.new(jobs)
printer.add_job('５ページ目')
printer.print
printer.display_jobs
